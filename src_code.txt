import { useState, useMemo } from "react";
import { Typography, Tabs, Space } from "antd";
import { formatFullDate } from "../../utility/format";
import { InventoryTable } from "../../components/inventory/InventoryTable";
import { useProducts } from "../../hooks/useProducts";
import { mockStockExits } from "../../mocks/stockExitList"; 

const { Title, Text } = Typography;

enum InventoryTabKey {
  MATERIAL = "material",
  TOOLS = "tools",
  OUTBOUND_BREAKDOWN = "outbound-breakdown",
  OUTBOUND_NON_BREAKDOWN = "outbound-non-breakdown",
}

export const InventoryPage = () => {
  const [activeTab, setActiveTab] = useState<string>(InventoryTabKey.MATERIAL);

  const { products, isLoading: isLoadingProducts } = useProducts();
  
  const [outboundData] = useState(mockStockExits);
  const isLoadingOutbound = false;
  
  const filteredMaterials = useMemo(() => {
    return products.filter(p => p.categoryId === 1 || p.categoryId === 2 || !p.categoryId); 
  }, [products]);

  const filteredTools = useMemo(() => {
    return products.filter(p => p.categoryId === 3 || p.categoryId === 4);
  }, [products]);

  const outboundBreakdown = useMemo(() => {
    return outboundData.slice(0, Math.ceil(outboundData.length / 2)); 
  }, [outboundData]);

  const outboundNonBreakdown = useMemo(() => {
    return outboundData.slice(Math.ceil(outboundData.length / 2));
  }, [outboundData]);

  const tabItems = [
    {
      key: InventoryTabKey.MATERIAL,
      label: "Stok Persediaan Material",
      children: (
        <InventoryTable 
          type="stock" 
          dataSource={filteredMaterials} 
          loading={isLoadingProducts} 
        />
      ),
    },
    {
      key: InventoryTabKey.TOOLS,
      label: "Stok Persediaan Alat Bantu",
      children: (
        <InventoryTable 
          type="stock" 
          dataSource={filteredTools} 
          loading={isLoadingProducts} 
        />
      ),
    },
    {
      key: InventoryTabKey.OUTBOUND_BREAKDOWN,
      label: "Pengeluaran (Breakdown)",
      children: (
        <InventoryTable 
          type="outbound" 
          dataSource={outboundBreakdown} 
          loading={isLoadingOutbound} 
        />
      ),
    },
    {
      key: InventoryTabKey.OUTBOUND_NON_BREAKDOWN,
      label: "Pengeluaran (Non-Breakdown)",
      children: (
        <InventoryTable 
          type="outbound" 
          dataSource={outboundNonBreakdown} 
          loading={isLoadingOutbound} 
        />
      ),
    },
  ];

  return (
    <Space direction="vertical" size="large" style={{ width: "100%" }}>
      <div>
        <Title level={3} style={{ marginBottom: 4 }}>
          Manajemen Persediaan & Pengeluaran
        </Title>
        <Text type="secondary">{formatFullDate(new Date())}</Text>
      </div>

      <Tabs
        activeKey={activeTab}
        onChange={setActiveTab}
        type="card"
        size="large"
        items={tabItems}
        destroyInactiveTabPane={false} 
      />
    </Space>
  );
};

export default InventoryPage;

interface FormValues {
  productId: number;
  supplierId: number | null;
  quantity: number;
  unitPrice: number;
  date: dayjs.Dayjs;
  note?: string;
}

export const StockEntryCreatePage = () => {
  const navigate = useNavigate();
  const [searchParams] = useSearchParams();
  const [form] = Form.useForm<FormValues>();
  const productSearchRef = useRef<HTMLInputElement>(null);

  const preselectedId = searchParams.get("productId");
  const [searchKeyword, setSearchKeyword] = useState("");
  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null);

  const { products, isLoading: isSearching } = useProductSearch(searchKeyword);
  const createMutation = useCreateStockEntry();

  // Auto-focus ke product search saat halaman dibuka
  useEffect(() => {
    setTimeout(() => productSearchRef.current?.focus(), 100);
  }, []);

  // Jika ada productId di URL, langsung set
  useEffect(() => {
    if (preselectedId) {
      const found = products.find((p) => p.id === Number(preselectedId));
      if (found) {
        setSelectedProduct(found);
        form.setFieldsValue({
          productId: found.id,
          unitPrice: found.costPrice,
        });
      }
    }
  }, [preselectedId, products, form]);

  // ⌨️ Ctrl+Enter → submit
  useKeyboardShortcut({
    key: "Enter",
    ctrlKey: true,
    handler: () => form.submit(),
  });

  // ⌨️ Esc → kembali
  useKeyboardShortcut({
    key: "Escape",
    handler: () => navigate(-1),
  });

  const handleProductChange = (productId: number) => {
    const product = products.find((p) => p.id === productId);
    if (!product) return;
    setSelectedProduct(product);
    setSearchKeyword("");
    form.setFieldsValue({
      productId: product.id,
      unitPrice: product.costPrice,
    });
    // Fokus ke quantity setelah pilih produk
    setTimeout(() => {
      form.getFieldInstance("quantity")?.focus?.();
    }, 50);
  };

  const handleSubmit = async (values: FormValues) => {
    await createMutation.mutateAsync({
      productId: values.productId,
      supplierId: values.supplierId ?? null,
      quantity: values.quantity,
      unitPrice: values.unitPrice,
      date: values.date.format("YYYY-MM-DD"),
      note: values.note,
    });

    form.resetFields();
    setSelectedProduct(null);
    // Balik ke search produk untuk entri berikutnya
    setTimeout(() => productSearchRef.current?.focus(), 100);
  };

  const quantity = Form.useWatch("quantity", form) ?? 0;
  const unitPrice = Form.useWatch("unitPrice", form) ?? 0;
  const total = quantity * unitPrice;

  return (
    <div style={{ maxWidth: 900, margin: "0 auto" }}>
      {/* Header */}
      <div style={{ marginBottom: 16, display: "flex", alignItems: "center", gap: 12 }}>
        <Button
          icon={<ArrowLeftOutlined />}
          onClick={() => navigate(-1)}
          type="text"
        />
        <div style={{ flex: 1 }}>
          <Title level={3} style={{ margin: 0 }}>
            Input Barang Masuk
          </Title>
          <Text type="secondary">
            Tekan <Tag>Ctrl</Tag> + <Tag>Enter</Tag> untuk simpan ·{" "}
            <Tag>Esc</Tag> untuk batal
          </Text>
        </div>
      </div>

      <Card style={{ borderRadius: 12 }}>
        <Form<FormValues>
          form={form}
          layout="vertical"
          onFinish={handleSubmit}
          initialValues={{ date: dayjs() }}
          requiredMark={false}
        >
          {/* Produk */}
          <Form.Item
            name="productId"
            label="Produk"
            rules={[{ required: true, message: "Pilih produk" }]}
          >
            <Select
              showSearch
              size="large"
              placeholder="Ketik nama atau kode produk..."
              loading={isSearching}
              onSearch={setSearchKeyword}
              filterOption={false}
              onChange={handleProductChange}
              autoFocus
              ref={productSearchRef as never}
              notFoundContent={isSearching ? "Mencari..." : "Tidak ditemukan"}
              options={products.map((p) => ({
                value: p.id,
                label: `${p.name} · ${p.code} (stok: ${p.stock} ${p.unit})`,
              }))}
            />
          </Form.Item>

          {/* Info produk terpilih */}
          {selectedProduct && (
            <Card
              size="small"
              style={{
                marginBottom: 16,

              }}
            >

                <Col span={8}>

                  <div>
                    <Text strong>
                      {selectedProduct.stock} {selectedProduct.unit}
                    </Text>
                  </div>
                </Col>
                <Col span={8}>
                  <Text type="secondary">Kategori</Text>
                  <div>
                    <Text strong>{selectedProduct.categoryName ?? "-"}</Text>
                  </div>
                </Col>
                <Col span={8}>
                  <Text type="secondary">Harga beli terakhir</Text>
                  <div>
                    <Text strong>{formatRupiah(selectedProduct.costPrice)}</Text>

import { useState, useEffect, useMemo } from "react";
import { Card, Col, Row, Statistic, Typography } from "antd";
import {
  AppstoreOutlined,
  InboxOutlined,
  ArrowDownOutlined,
  ArrowUpOutlined,
} from "@ant-design/icons";
import { SetupWizard } from "../../components/entry/SetupWizard";
import { QuickEntryPanel } from "../../components/dashboard/QuickPanel";
import { StockTable } from "../../components/dashboard/StockTable";
import { StockExitList } from "../../components/dashboard/StockExitList";
import { useQuickEntry } from "../../hooks/useQuickEntry";
import { useProducts } from "../../hooks/useProducts";
import { mockStockExits } from "../../mocks/stockExitList";
import { formatFullDate } from "../../utility/format";
import { COLORS } from "../../utility/constants";
import type { Product, StockExit } from "../../types";

const { Title, Text } = Typography;

interface StatCardProps {
  title: string;
  value: number;
  icon: React.ReactNode;
  color: string;
  suffix?: string;
  loading?: boolean;
}

const StatCard = ({
  title,
  value,
  icon,
  color,
  suffix,
  loading,
}: StatCardProps) => (
  <Card
    variant="borderless"
    style={{ borderRadius: 12 }}
    styles={{ body: { padding: 16 } }}
  >
    <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
      <div
        style={{
          width: 48,
          height: 48,
          borderRadius: 12,
          background: `${color}15`,
          color,
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          fontSize: 22,
          flexShrink: 0,
        }}
      >
        {icon}
      </div>
      <div style={{ flex: 1, minWidth: 0 }}>
        <Text type="secondary" style={{ fontSize: 13 }}>
          {title}
        </Text>
        <Statistic
          value={loading ? 0 : value}
          suffix={suffix}
          valueStyle={{
            color,
            fontWeight: 600,
            fontSize: 22,
            lineHeight: 1.2,
          }}
          loading={loading}
        />
      </div>
    </div>
  </Card>
);

export const DashboardPage = () => {
  const { products: fetchedProducts, isLoading } = useProducts();
  const [products, setProducts] = useState<Product[]>([]);
  const [exits, setExits] = useState<StockExit[]>(mockStockExits);

  useEffect(() => {
    if (fetchedProducts.length > 0) {
      setProducts(fetchedProducts);
    }
  }, [fetchedProducts]);

  const entry = useQuickEntry({
    onFinish: (exit) => {
      setProducts((prev) =>
        prev.map((p) => {
          const line = exit.items.find((i) => i.itemId === p.id);
          if (!line) return p;
          return { ...p, stock: Math.max(0, p.stock - line.quantity) };
        }),
      );
      setExits((prev) => [exit, ...prev]);
    },
  });

  const stats = useMemo(() => {
    const totalProducts = products.length;
    const totalStock = products.reduce((sum, p) => sum + (p.stock ?? 0), 0);
    const todayInbound = 0;
    const today = new Date().toISOString().split("T")[0];
    const todayOutbound = exits
      .filter((e) => e.date === today)
      .reduce((sum, e) => sum + e.items.reduce((s, i) => s + i.quantity, 0), 0);

    return { totalProducts, totalStock, todayInbound, todayOutbound };
  }, [products, exits]);

  return (
    <>
      <SetupWizard
        open={entry.flow.wizardOpen}
        step={
          entry.flow.step === "done"
            ? "workplace"
            : (entry.flow.step as "workplace" | "person")
        }
        title={entry.flow.currentWizardData?.title ?? ""}
        placeholder={entry.flow.currentWizardData?.placeholder ?? ""}
        options={entry.flow.currentWizardData?.options ?? []}
        onSelect={entry.flow.selectWizardValue}
        onBack={
          entry.flow.step !== "workplace" ? entry.flow.backWizard : undefined
        }
        onClose={() => entry.flow.closeWizard()}
      />

      <div style={{ marginBottom: 16 }}>
        <Title level={3} style={{ marginBottom: 4 }}>
          Dashboard
        </Title>
        <Text type="secondary">{formatFullDate(new Date())}</Text>
      </div>

      <QuickEntryPanel entry={entry} />

      <Row gutter={[16, 16]} style={{ marginBottom: 16 }}>
        <Col xs={24} sm={12} lg={6}>
          <StatCard
            title="Total Produk"
            value={stats.totalProducts}
            icon={<AppstoreOutlined />}
            color={COLORS.primary}
            suffix="item"
            loading={isLoading}
          />
        </Col>
        <Col xs={24} sm={12} lg={6}>
          <StatCard
            title="Total Stok"
            value={stats.totalStock}
            icon={<InboxOutlined />}
            color={COLORS.success}
            loading={isLoading}
          />
        </Col>
        <Col xs={24} sm={12} lg={6}>
          <StatCard
            title="Masuk Hari Ini"
            value={stats.todayInbound}
            icon={<ArrowDownOutlined />}
            color={COLORS.warning}
            loading={false}
          />
        </Col>
        <Col xs={24} sm={12} lg={6}>
          <StatCard
            title="Keluar Hari Ini"
            value={stats.todayOutbound}
            icon={<ArrowUpOutlined />}
            color={COLORS.danger}
            loading={false}
          />
        </Col>
      </Row>

      <Row gutter={[16, 16]}>
        <Col xs={24} lg={16}>
          <StockTable products={products} loading={isLoading} />
        </Col>
        <Col xs={24} lg={8}>
          <StockExitList exits={exits} />
        </Col>
      </Row>
    </>
  );
};

export default DashboardPage;

  for (let i = 29; i >= 0; i--) {
    const date = dayjs().subtract(i, "day").format("YYYY-MM-DD");

import { useCallback, useState } from "react";
import { mockPersons, mockWorkplaces } from "../mocks/entry";
import type { EntryLine, Person, SetupStep, Workplace } from "../types";
import type { Product } from "../types";

export type EntryPhase = "search" | "quantity";

export const useEntryFlow = () => {
  // ==========================================
  // Wizard state
  // ==========================================
  const [step, setStep] = useState<SetupStep>("workplace");
  const [workplace, setWorkplace] = useState<Workplace | null>(null);
  const [person, setPerson] = useState<Person | null>(null);
  const [wizardOpen, setWizardOpen] = useState(true);

  // ==========================================
  // Board state
  // ==========================================
  const [phase, setPhase] = useState<EntryPhase>("search");
  const [activeItem, setActiveItem] = useState<Product | null>(null);
  const [activeQty, setActiveQty] = useState(1);
  const [lines, setLines] = useState<EntryLine[]>([]);

  // ==========================================
  // Wizard data
  // ==========================================
  const wizardData = {
    workplace: {
      value: workplace,
      options: mockWorkplaces,
      placeholder: "Ketik lokasi pekerjaan...",
      title: "Pilih Lokasi Pekerjaan",
    },
    person: {
      value: person,
      options: mockPersons,
      placeholder: "Ketik nama pengambil...",
      title: "Pilih Nama",
    },
  } as const;

  const currentWizardData =
    step === "done"
      ? { value: null, title: "", placeholder: "", options: [] }
      : wizardData[step as "workplace" | "person"];

  // ==========================================
  // Wizard actions
  // ==========================================
  const openWizard = useCallback(() => setWizardOpen(true), []);
  const closeWizard = useCallback(() => setWizardOpen(false), []);

  const selectWizardValue = useCallback(
    (value: Person | Workplace) => {
      if (step === "workplace") {
        setWorkplace(value as Workplace);
        setPerson(null);
        setStep("person");
      } else if (step === "person") {
        setPerson(value as Person);
        setStep("done");
        setWizardOpen(false);
      }
    },
    [step],
  );

  const resetSession = useCallback(() => {
    setStep("workplace");
    setWorkplace(null);
    setPerson(null);
    setPhase("search");
    setActiveItem(null);
    setActiveQty(1);
    setLines([]);
    setWizardOpen(true);
  }, []);

  const backWizard = useCallback(() => {
    if (step === "person") {
      setPerson(null);
      setStep("workplace");
    }
  }, [step]);

  // ==========================================
  // Board actions
  // ==========================================
  const selectItem = useCallback((item: Product) => {
    setActiveItem(item);
    setActiveQty(1);
    setPhase("quantity");
  }, []);

  const confirmQuantity = useCallback(() => {
    if (!activeItem) return;
    setLines((prev) => {
      const idx = prev.findIndex((l) => l.itemId === activeItem.id);
      if (idx >= 0) {
        const copy = [...prev];
        copy[idx] = { ...copy[idx], quantity: copy[idx].quantity + activeQty };
        return copy;
      }
      return [
        ...prev,
        {
          itemId: activeItem.id,
          itemName: activeItem.name,
          unit: activeItem.unit,
          quantity: activeQty,
        },
      ];
    });
    setActiveItem(null);
    setActiveQty(1);
    setPhase("search");
  }, [activeItem, activeQty]);

  const cancelQuantity = useCallback(() => {
    setActiveItem(null);
    setActiveQty(1);
    setPhase("search");
  }, []);

  const removeLine = useCallback((itemId: number) => {
    setLines((prev) => prev.filter((l) => l.itemId !== itemId));
  }, []);

  const increaseQty = useCallback(() => setActiveQty((q) => q + 1), []);
  const decreaseQty = useCallback(
    () => setActiveQty((q) => Math.max(1, q - 1)),
    [],
  );

  return {
    step,
    person,
    workplace,
    currentWizardData,
    selectWizardValue,
    backWizard,
    wizardOpen,
    openWizard,
    closeWizard,
    phase,
    activeItem,
    activeQty,
    lines,
    selectItem,
    confirmQuantity,
    cancelQuantity,
    increaseQty,
    decreaseQty,
    setActiveQty,
    removeLine,
    resetSession,
    isReady: step === "done",
  };
};

  refetch: () => void;
}

const EMPTY_STATS: DashboardStats = {
  totalProducts: 0,
  totalStock: 0,
  todayInbound: 0,
  todayOutbound: 0,
};

export const useDashboardStats = (): UseDashboardStatsResult => {
  const query = useQuery<DashboardStats, Error, DashboardStats>({
    queryKey: ["dashboard-stats"],
    refetchInterval: USE_MOCK ? false : 60_000,
    queryFn: async (): Promise<DashboardStats> => {
      // === MODE MOCK ===
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return getMockDashboardStats();
      }

      // === MODE SUPABASE ===
      const today = new Date().toISOString().split("T")[0];

      const [productsRes, inboundRes, outboundRes] = await Promise.all([
        supabaseClient.from(TABLES.products).select("stok"),
        supabaseClient
          .from(TABLES.stockEntries)
          .select("jumlah")
          .eq("tanggal", today),
        supabaseClient
          .from(TABLES.stockExits)
          .select("items")
          .eq("date", today),
      ]);

      if (productsRes.error) throw productsRes.error;
      if (inboundRes.error) throw inboundRes.error;
      if (outboundRes.error) throw outboundRes.error;

      const totalProducts = productsRes.data?.length ?? 0;
      const totalStock =
        productsRes.data?.reduce(
          (sum, p: { stok: number | null }) => sum + (p.stok ?? 0),
          0,
        ) ?? 0;
      const todayInbound =
        inboundRes.data?.reduce(
          (sum, e: { jumlah: number | null }) => sum + (e.jumlah ?? 0),
          0,
        ) ?? 0;

      // Hitung total quantity dari items[] di stock exits
      const todayOutbound =
        outboundRes.data?.reduce(
          (sum, exit: { items: Array<{ quantity: number }> }) =>
            sum + (exit.items?.reduce((s, i) => s + (i.quantity ?? 0), 0) ?? 0),
          0,
        ) ?? 0;

      return { totalProducts, totalStock, todayInbound, todayOutbound };
    },
  });

  return {
    stats: query.data ?? EMPTY_STATS,
    isLoading: query.isLoading,
    refetch: query.refetch,
  };
};

export const useProductSearch = (keyword: string, limit = 8) => {
  const trimmed = keyword.trim();

  const { data, isLoading, refetch } = useQuery<Product[]>({
    queryKey: ["product-search", trimmed, limit],
    queryFn: async () => {
      // === MOCK ===
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 150));
        if (!trimmed) return mockProducts.slice(0, limit);
        const lower = trimmed.toLowerCase();
        return mockProducts
          .filter(
            (p) =>
              p.name.toLowerCase().includes(lower) ||
              p.code.toLowerCase().includes(lower),

      // === SUPABASE ===
      let query = supabaseClient

      const { data, error } = await query;
      if (error) throw error;
      return (data ?? []).map(mapProduct);
    },
    staleTime: 10_000,
  });

  return {
    products: data ?? [],
    isLoading,
    refetch,

import { useCallback, useEffect, useRef, useState } from "react";
import { App } from "antd";
import dayjs from "dayjs";
import { useProductSearch } from "./useProductSearch";
import { useEntryFlow } from "./useEntryFlow";
import type { Product, StockExit, StockExitItem } from "../types";

export interface PendingLine {
  itemId: number;
  itemName: string;
  unit: string;
  quantity: number;
}

interface UseQuickEntryParams {
  onFinish?: (exit: StockExit) => void;
}

export const useQuickEntry = ({ onFinish }: UseQuickEntryParams = {}) => {
  const { message } = App.useApp();
  const flow = useEntryFlow();

  const [keyword, setKeyword] = useState("");
  const [highlightedIdx, setHighlightedIdx] = useState(0);
  const [activeItem, setActiveItem] = useState<Product | null>(null);
  const [activeQty, setActiveQty] = useState(1);
  const [pendingLines, setPendingLines] = useState<PendingLine[]>([]);

  const searchRef = useRef<HTMLInputElement>(null);
  const qtyRef = useRef<HTMLInputElement>(null);

  const { products: searchResults } = useProductSearch(keyword, 8);
  const phase = activeItem ? "quantity" : "search";

  useEffect(() => {
    if (!flow.isReady) return;
    if (phase === "search") {
      setTimeout(() => searchRef.current?.focus(), 100);
    } else {
      setTimeout(() => qtyRef.current?.focus(), 100);
    }
  }, [phase, flow.isReady]);

  useEffect(() => setHighlightedIdx(0), [keyword]);

  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (e.ctrlKey && e.key === "Enter" && pendingLines.length > 0) {
        e.preventDefault();
        handleFinish();
      }
    };
    window.addEventListener("keydown", handler);
    return () => window.removeEventListener("keydown", handler);
  }, [pendingLines.length]);

  const commitItem = useCallback(() => {
    if (searchResults.length === 0) return;
    const chosen = searchResults[highlightedIdx] ?? searchResults[0];
    setActiveItem(chosen);
    setActiveQty(1);
    setKeyword("");
  }, [searchResults, highlightedIdx]);

  const handleSearchKeyDown = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === "ArrowDown") {
        e.preventDefault();
        setHighlightedIdx((i) => Math.min(i + 1, searchResults.length - 1));
      } else if (e.key === "ArrowUp") {
        e.preventDefault();
        setHighlightedIdx((i) => Math.max(i - 1, 0));
      } else if (e.key === "Enter") {
        e.preventDefault();
        commitItem();
      }
    },
    [searchResults.length, commitItem],
  );

  const confirmQuantity = useCallback(() => {
    if (!activeItem) return;
    setPendingLines((prev) => {
      const idx = prev.findIndex((l) => l.itemId === activeItem.id);
      if (idx >= 0) {
        const copy = [...prev];
        copy[idx] = {
          ...copy[idx],
          quantity: copy[idx].quantity + activeQty,
        };
        return copy;
      }
      return [
        ...prev,
        {
          itemId: activeItem.id,
          itemName: activeItem.name,
          unit: activeItem.unit,
          quantity: activeQty,
        },
      ];
    });
    setActiveItem(null);
    setActiveQty(1);
  }, [activeItem, activeQty]);

  const handleQtyKeyDown = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === "Enter") {
        e.preventDefault();
        confirmQuantity();
      } else if (e.key === "Escape") {
        e.preventDefault();
        setActiveItem(null);
        setActiveQty(1);
      } else if (e.key === "ArrowUp") {
        e.preventDefault();
        setActiveQty((q) => q + 1);
      } else if (e.key === "ArrowDown") {
        e.preventDefault();
        setActiveQty((q) => Math.max(1, q - 1));
      }
    },
    [confirmQuantity],
  );

  const removeLine = useCallback((itemId: number) => {
    setPendingLines((prev) => prev.filter((l) => l.itemId !== itemId));
  }, []);

  const resetEntry = useCallback(() => {
    setPendingLines([]);
    setActiveItem(null);
    setActiveQty(1);
    setKeyword("");
  }, []);

  const handleFinish = useCallback(() => {
    if (pendingLines.length === 0) return;
    if (!flow.person || !flow.workplace) return;

    const now = dayjs();
    const exit: StockExit = {
      id: Date.now(),
      timestamp: now.toISOString(),
      date: now.format("YYYY-MM-DD"),
      time: now.format("HH:mm"),
      team: "-",
      person: flow.person.name,
      workplace: flow.workplace.name,
      items: pendingLines.map<StockExitItem>((l) => ({
        itemId: l.itemId,
        itemName: l.itemName,
        unit: l.unit,
        quantity: l.quantity,
      })),
    };

    message.success(
      `Tersimpan: ${pendingLines.length} jenis barang di ${flow.workplace.name}`,
    );

    onFinish?.(exit);
    resetEntry();
    flow.resetSession();
  }, [pendingLines, flow, message, onFinish, resetEntry]);

  const handleCancel = useCallback(() => {
    resetEntry();
    flow.resetSession();
  }, [resetEntry, flow]);

  const totalPendingItems = pendingLines.reduce((s, l) => s + l.quantity, 0);

  return {
    flow,
    keyword,
    setKeyword,
    highlightedIdx,
    searchResults,
    searchRef,
    handleSearchKeyDown,
    commitItem,
    activeItem,
    activeQty,
    setActiveQty,
    qtyRef,
    handleQtyKeyDown,
    confirmQuantity,
    pendingLines,
    totalPendingItems,
    removeLine,
    phase,
    handleFinish,
    handleCancel,
  };
};

        total,
      }));
    },
  });

  return {
    items: data ?? [],
    isLoading,
    refetch,
  };

import { useEffect } from "react";

interface UseKeyboardShortcutOptions {
  key: string;
  ctrlKey?: boolean;
  shiftKey?: boolean;
  altKey?: boolean;
  metaKey?: boolean;
  enabled?: boolean;
  preventDefault?: boolean;
  /** Jika true, shortcut diabaikan saat user sedang mengetik di input/textarea */
  ignoreInputs?: boolean;
  handler: (event: KeyboardEvent) => void;
}

const isEditableElement = (el: EventTarget | null): boolean => {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName.toLowerCase();
  return (
    tag === "input" ||
    tag === "textarea" ||
    tag === "select" ||
    el.isContentEditable
  );
};

export const useKeyboardShortcut = ({
  key,
  ctrlKey = false,
  shiftKey = false,
  altKey = false,
  metaKey = false,
  enabled = true,
  preventDefault = true,
  ignoreInputs = false,
  handler,
}: UseKeyboardShortcutOptions) => {
  useEffect(() => {
    if (!enabled) return;

    const onKeyDown = (event: KeyboardEvent) => {
      if (event.key !== key) return;
      if (event.ctrlKey !== ctrlKey) return;
      if (event.shiftKey !== shiftKey) return;
      if (event.altKey !== altKey) return;
      if (event.metaKey !== metaKey) return;
      if (ignoreInputs && isEditableElement(event.target)) return;

      if (preventDefault) event.preventDefault();
      handler(event);
    };

    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  }, [
    key,
    ctrlKey,
    shiftKey,
    altKey,
    metaKey,
    enabled,
    preventDefault,
    ignoreInputs,
    handler,
  ]);
};

import { SearchOutlined } from "@ant-design/icons";
import { useKeyboardShortcut } from "../../hooks/useKeyboardShortcut";
import { useProductSearch } from "../../hooks/useProductSearch";
import type { Product } from "../../types";

const { Text } = Typography;

export const GlobalSearch = () => {
  const navigate = useNavigate();
  const [open, setOpen] = useState(false);
  const [keyword, setKeyword] = useState("");
  const [activeIndex, setActiveIndex] = useState(0);
  const inputRef = useRef<HTMLInputElement>(null);

  const { products, isLoading } = useProductSearch(keyword);

  // Reset state saat modal terbuka
  useEffect(() => {
    if (open) {
      setKeyword("");
      setActiveIndex(0);
      setTimeout(() => inputRef.current?.focus(), 50);
    }
  }, [open]);

  // Clamp active index saat hasil berubah
  useEffect(() => {
    if (activeIndex >= products.length) setActiveIndex(0);
  }, [products.length, activeIndex]);

  const handleSelect = useCallback(
    (product: Product) => {
      setOpen(false);
      navigate(`/stock-entries/create?productId=${product.id}`);
    },
    [navigate],
  );

  // ⌨️ Space → buka palette (kecuali sedang mengetik di input)
  useKeyboardShortcut({
    key: " ",
    ignoreInputs: true,
    enabled: !open,
    handler: () => setOpen(true),
  });

  // ⌨️ Ctrl+K → buka palette juga (opsional, umum dipakai)
  useKeyboardShortcut({
    key: "k",
    ctrlKey: true,
    enabled: !open,
    handler: () => setOpen(true),
  });

  const handleKeyDown = (event: React.KeyboardEvent) => {
    if (event.key === "ArrowDown") {
      event.preventDefault();
      setActiveIndex((i) => Math.min(i + 1, products.length - 1));
    } else if (event.key === "ArrowUp") {
      event.preventDefault();
      setActiveIndex((i) => Math.max(i - 1, 0));
    } else if (event.key === "Enter") {
      event.preventDefault();
      const selected = products[activeIndex];
      if (selected) handleSelect(selected);
    } else if (event.key === "Escape") {
      event.preventDefault();
      setOpen(false);
    }
  };

  return (
    <Modal
      open={open}
      onCancel={() => setOpen(false)}
      footer={null}
      closable={false}
      width={560}
      destroyOnClose
      styles={{ body: { padding: 0 } }}
      style={{ top: 80 }}
    >
      <div style={{ padding: 12, borderBottom: "1px solid #f0f0f0" }}>
        <Input
          ref={inputRef as never}
          size="large"
          prefix={<SearchOutlined />}
          placeholder="Cari produk berdasarkan nama atau kode..."
          value={keyword}
          onChange={(e) => setKeyword(e.target.value)}
          onKeyDown={handleKeyDown}
          variant="borderless"
          autoFocus
        />
      </div>

      <div style={{ maxHeight: 360, overflowY: "auto", padding: 8 }}>
        {isLoading && (
          <div style={{ textAlign: "center", padding: 24 }}>
            <Spin />
          </div>
        )}

        {!isLoading && products.length === 0 && (
          <Empty
            description="Produk tidak ditemukan"
            image={Empty.PRESENTED_IMAGE_SIMPLE}
            style={{ padding: 24 }}
          />
        )}

        {!isLoading &&
          products.map((product, index) => (
            <div
              key={product.id}
              onMouseEnter={() => setActiveIndex(index)}
              onClick={() => handleSelect(product)}
              style={{
                padding: "10px 12px",
                borderRadius: 8,
                cursor: "pointer",
                display: "flex",
                alignItems: "center",
                justifyContent: "space-between",
                gap: 12,
                background:
                  index === activeIndex ? "#e6f4ff" : "transparent",
              }}
            >
              <div style={{ minWidth: 0, flex: 1 }}>
                <div style={{ fontWeight: 500, fontSize: 14 }}>
                  {product.name}
                </div>
                <Text type="secondary" style={{ fontSize: 12 }}>
                  {product.code} · Stok: {product.stock} {product.unit}
                </Text>
              </div>
              <Tag color="blue" style={{ margin: 0 }}>
                Input
              </Tag>
            </div>
          ))}
      </div>

      <div
        style={{
          padding: "8px 16px",
          borderTop: "1px solid #f0f0f0",
          display: "flex",
          gap: 16,
          fontSize: 12,
          color: "#8c8c8c",
        }}
      >
        <span>↑↓ navigasi</span>
        <span>↵ pilih</span>
        <span>esc tutup</span>
      </div>
    </Modal>
  );
};

      {/* ====================================== */}

                data: p,

              onSelect={(_, opt) => {

import { useEffect, useRef, useState } from "react";
import {
  AutoComplete,
  Card,
  Input,
  Modal,
  Space,
  Steps,
  Tag,
  Typography,
} from "antd";
import { SearchOutlined } from "@ant-design/icons";
import type { Person, Workplace } from "../../types/entry";

const { Title, Text } = Typography;

interface Props {
  open: boolean;
  step: "workplace" | "person";
  title: string;
  placeholder: string;
  options: readonly (Person | Workplace)[];
  onSelect: (value: Person | Workplace) => void;
  onBack?: () => void;
  onClose: () => void;
}

const STEP_INDEX: Record<"workplace" | "person", number> = {
  workplace: 0,
  person: 1,
};

export const SetupWizard = ({
  open,
  step,
  title,
  placeholder,
  options,
  onSelect,
  onBack,
  onClose,
}: Props) => {
  const [keyword, setKeyword] = useState("");
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    if (open) {
      setKeyword("");
      setTimeout(() => inputRef.current?.focus(), 100);
    }
  }, [open, step]);

  const filtered = options.filter((o) =>
    o.name.toLowerCase().includes(keyword.trim().toLowerCase()),
  );

  const handleEnter = () => {
    if (filtered.length > 0) {
      onSelect(filtered[0]);
    }
  };

  return (
    <Modal
      open={open}
      onCancel={onClose}
      footer={null}
      closable
      maskClosable
      keyboard
      width={520}
      styles={{ body: { padding: 0 } }}
      style={{ top: 100 }}
    >
      <Card
        variant="borderless"
        style={{ borderRadius: 16 }}
        styles={{ body: { padding: 24 } }}
      >
        <Steps
          current={STEP_INDEX[step]}
          size="small"
          items={[{ title: "Lokasi" }, { title: "Nama" }]}
          style={{ marginBottom: 20 }}
        />

        <Title level={4} style={{ marginBottom: 4 }}>
          {title}
        </Title>
        <Text type="secondary">

        <div style={{ marginTop: 16 }}>
          <AutoComplete
            value={keyword}
            onChange={setKeyword}
            options={filtered.map((o) => ({
              value: o.name,
              label: o.name,
              data: o,
            }))}
            onSelect={(_, option) => {
              const data = (option as { data: Person | Workplace }).data;
              onSelect(data);
            }}
            style={{ width: "100%" }}
            placeholder={placeholder}
          >
            <Input
              ref={inputRef as never}
              size="large"
              prefix={<SearchOutlined />}
              onKeyDown={(e) => {
                if (e.key === "Enter") {
                  e.preventDefault();
                  handleEnter();
                } else if (e.key === "Escape" && onBack) {
                  e.preventDefault();
                  onBack();
                }
              }}
              autoFocus
            />
          </AutoComplete>
        </div>

        <Space style={{ marginTop: 16 }} size="small">
          <Tag>Enter</Tag>
          <Text type="secondary" style={{ fontSize: 12 }}>
            pilih item pertama
          </Text>
          {onBack && (

              <Tag>Esc</Tag>
              <Text type="secondary" style={{ fontSize: 12 }}>
                kembali
              </Text>
            </>
          )}
        </Space>
      </Card>
    </Modal>
  );
};

        <div
          style={{
            marginTop: 12,
            paddingTop: 12,
            borderTop: "1px dashed #f0f0f0",
          }}

import type {
  AuthProvider,
  AuthActionResponse,
  CheckResponse,
} from "@refinedev/core";
import { supabaseClient } from "./utility/supabaseClient";
import { USE_MOCK } from "./utility/config";

const MOCK_USER = {
  email: "admin@admin.com",
  password: "admin123",
  name: "Administrator",
};

const STORAGE_KEY = "mock-auth-user";

const getMockUser = () => {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
};

export const authProvider: AuthProvider = {
  login: async ({ email, password }): Promise<AuthActionResponse> => {
    if (USE_MOCK) {
      await new Promise((r) => setTimeout(r, 400));
      if (email === MOCK_USER.email && password === MOCK_USER.password) {
        localStorage.setItem(
          STORAGE_KEY,
          JSON.stringify({ email, name: MOCK_USER.name }),
        );
        return { success: true, redirectTo: "/dashboard" };
      }
      return {
        success: false,
        error: {
          name: "LoginError",
          message: "Email atau password salah",
        },
      };
    }

    const { data, error } = await supabaseClient.auth.signInWithPassword({
      email,
      password,
    });

    if (error) {
      return {
        success: false,
        error: { name: "LoginError", message: error.message },
      };
    }
    if (!data?.user) {
      return {
        success: false,
        error: { name: "LoginError", message: "User tidak ditemukan" },
      };
    }
    return { success: true, redirectTo: "/dashboard" };
  },

  logout: async (): Promise<AuthActionResponse> => {
    if (USE_MOCK) {
      localStorage.removeItem(STORAGE_KEY);
      return { success: true, redirectTo: "/login" };
    }
    const { error } = await supabaseClient.auth.signOut();
    if (error) {
      return {
        success: false,
        error: { name: "LogoutError", message: error.message },
      };
    }
    return { success: true, redirectTo: "/login" };
  },

  // ⬇️ INI YANG DIPERBAIKI: return type CheckResponse, bukan AuthActionResponse
  check: async (): Promise<CheckResponse> => {
    if (USE_MOCK) {
      const user = getMockUser();
      return user
        ? { authenticated: true }
        : { authenticated: false, redirectTo: "/login", logout: true };
    }
    const { data } = await supabaseClient.auth.getSession();
    return data.session
      ? { authenticated: true }
      : { authenticated: false, redirectTo: "/login", logout: true };
  },

  getIdentity: async () => {
    if (USE_MOCK) {
      const user = getMockUser();
      if (!user) return null;
      return {
        id: "mock-1",
        name: user.name,
        email: user.email,
        avatar: `https://ui-avatars.com/api/?name=${encodeURIComponent(
          user.name,
        )}&background=1677ff&color=fff`,
      };
    }
    const { data } = await supabaseClient.auth.getUser();
    if (!data.user) return null;
    return {
      id: data.user.id,
      name: data.user.email?.split("@")[0] ?? "User",
      email: data.user.email,
      avatar: `https://ui-avatars.com/api/?name=${encodeURIComponent(
        data.user.email ?? "U",
      )}&background=1677ff&color=fff`,
    };
  },

  getPermissions: async () => ["admin"],

  onError: async (error) => {
    if (error?.statusCode === 401) {
      return { logout: true, redirectTo: "/login", error };
    }
    return { error };
  },
};
