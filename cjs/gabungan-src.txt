import type { AuthProvider, AuthActionResponse, CheckResponse } from "@refinedev/core";
import { supabaseClient } from "./utility/supabaseClient";
import { USE_MOCK } from "./utility/config";

const STORAGE_KEY = "mock-auth-user";

const getMockUser = () => {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
};

export const authProvider: AuthProvider = {
  login: async ({ email, password }): Promise<AuthActionResponse> => {
    if (USE_MOCK) {
      await new Promise((r) => setTimeout(r, 400));
      if (email === "admin@admin.com" && password === "admin123") {
        localStorage.setItem(STORAGE_KEY, JSON.stringify({ email, name: "Administrator", role: "admin" }));
        return { success: true, redirectTo: "/dashboard" };
      }
      return { success: false, error: { name: "LoginError", message: "Email/password salah" } };
    }

    const { data, error } = await supabaseClient.auth.signInWithPassword({ email, password });
    if (error || !data?.user) {
      return { success: false, error: { name: "LoginError", message: error?.message || "User tidak ditemukan" } };
    }
    return { success: true, redirectTo: "/dashboard" };
  },

  logout: async (): Promise<AuthActionResponse> => {
    if (USE_MOCK) {
      localStorage.removeItem(STORAGE_KEY);
      return { success: true, redirectTo: "/dashboard" };
    }
    await supabaseClient.auth.signOut();
    return { success: true, redirectTo: "/dashboard" };
  },

  // ✅ PERBAIKAN: Tidak lagi di-hardcode true. Wajib cek session untuk rute admin.
  check: async (): Promise<CheckResponse> => {
    if (USE_MOCK) {
      return { authenticated: !!getMockUser(), redirectTo: "/login" };
    }
    const { data } = await supabaseClient.auth.getSession();
    return { 
      authenticated: !!data.session, 
      redirectTo: "/login" 
    };
  },

  getIdentity: async () => {
    if (USE_MOCK) {
      const user = getMockUser();
      return user 
        ? { id: "admin-1", name: user.name, email: user.email, role: "admin" }
        : { id: "public-user", name: "User (Public)", email: "", role: "user" };
    }
    const { data } = await supabaseClient.auth.getUser();
    if (data?.user) {
      return { id: data.user.id, name: data.user.email?.split("@")[0] ?? "Admin", email: data.user.email, role: "admin" };
    }
    return { id: "public-user", name: "User (Public)", email: "", role: "user" };
  },

  getPermissions: async () => {
    if (USE_MOCK) return getMockUser() ? ["admin"] : ["user"];
    const { data } = await supabaseClient.auth.getSession();
    return data.session ? ["admin"] : ["user"];
  },

  onError: async (error) => ({ error }),
};

import React, { Component, ErrorInfo, ReactNode } from "react";
import { Result, Button, Typography } from "antd";

const { Paragraph, Text } = Typography;

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  public state: State = { hasError: false, error: null };

  public static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  public componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error("Uncaught error:", error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div style={{ padding: 40, display: "flex", justifyContent: "center" }}>
          <Result
            status="500"
            title="Terjadi Kesalahan (500)"
            subTitle="Maaf, aplikasi mengalami masalah internal."
            extra={
              <Button type="primary" onClick={() => window.location.reload()}>
                Muat Ulang Halaman
              </Button>
            }
          >
            <div style={{ textAlign: "left", background: "#f5f5f5", padding: 16, borderRadius: 8 }}>
              <Text strong>Detail Error:</Text>
              <Paragraph type="danger" copyable style={{ marginTop: 8 }}>
                {this.state.error?.message}
              </Paragraph>
            </div>
          </Result>
        </div>
      );
    }
    return this.props.children;
  }
}

import { useState, useRef, useEffect, useCallback } from "react";
import { useNavigate } from "react-router-dom";
import { Input, Modal, Spin, Empty, Tag, Typography } from "antd";
import { SearchOutlined } from "@ant-design/icons";
import { useKeyboardShortcut } from "../../hooks/useKeyboardShortcut";
import { useDebounce } from "../../hooks/useDebounce";
import { useProductSearch } from "../../hooks/useProductSearch";
import type { Product } from "../../types";

const { Text } = Typography;

export const GlobalSearch = () => {
  const navigate = useNavigate();
  const [open, setOpen] = useState(false);
  const [keyword, setKeyword] = useState("");
  const [activeIndex, setActiveIndex] = useState(0);
  const inputRef = useRef<HTMLInputElement>(null);

  const debouncedKeyword = useDebounce(keyword, 250);
  const { products, isLoading } = useProductSearch(debouncedKeyword);

  useEffect(() => {
    if (open) {
      setKeyword("");
      setActiveIndex(0);
      setTimeout(() => inputRef.current?.focus(), 50);
    }
  }, [open]);

  useEffect(() => {
    if (activeIndex >= products.length) setActiveIndex(0);
  }, [products.length, activeIndex]);

  const handleSelect = useCallback(
    (product: Product) => {
      setOpen(false);
      navigate(`/products/create?productId=${product.id}`);
    },
    [navigate],
  );

  // ⌨️ Ctrl+K → buka palette pencarian global
  useKeyboardShortcut({
    key: "k",
    ctrlKey: true,
    enabled: !open,
    handler: () => setOpen(true),
  });

  const handleKeyDown = (event: React.KeyboardEvent) => {
    if (event.key === "ArrowDown") {
      event.preventDefault();
      setActiveIndex((i) => Math.min(i + 1, Math.max(0, products.length - 1)));
    } else if (event.key === "ArrowUp") {
      event.preventDefault();
      setActiveIndex((i) => Math.max(i - 1, 0));
    } else if (event.key === "Enter") {
      event.preventDefault();
      const selected = products[activeIndex];
      if (selected) handleSelect(selected);
    } else if (event.key === "Escape") {
      event.preventDefault();
      setOpen(false);
    }
  };

  return (
    <Modal
      open={open}
      onCancel={() => setOpen(false)}
      footer={null}
      closable={false}
      width={560}
      destroyOnClose
      styles={{ body: { padding: 0 } }}
      style={{ top: 80 }}
    >
      <div style={{ padding: 12, borderBottom: "1px solid #f0f0f0" }}>
        <Input
          ref={inputRef as never}
          size="large"
          prefix={<SearchOutlined />}
          placeholder="Cari produk berdasarkan nama atau kode (Ctrl+K)..."
          value={keyword}
          onChange={(e) => setKeyword(e.target.value)}
          onKeyDown={handleKeyDown}
          variant="borderless"
          autoFocus
        />
      </div>

      <div style={{ maxHeight: 360, overflowY: "auto", padding: 8 }}>
        {isLoading && (
          <div style={{ textAlign: "center", padding: 24 }}>
            <Spin />
          </div>
        )}

        {!isLoading && products.length === 0 && (
          <Empty
            description="Produk tidak ditemukan"
            image={Empty.PRESENTED_IMAGE_SIMPLE}
            style={{ padding: 24 }}
          />
        )}

        {!isLoading &&
          products.map((product, index) => (
            <div
              key={product.id}
              onMouseEnter={() => setActiveIndex(index)}
              onClick={() => handleSelect(product)}
              style={{
                padding: "10px 12px",
                borderRadius: 8,
                cursor: "pointer",
                display: "flex",
                alignItems: "center",
                justifyContent: "space-between",
                gap: 12,
                background: index === activeIndex ? "#e6f4ff" : "transparent",
              }}
            >
              <div style={{ minWidth: 0, flex: 1 }}>
                <div style={{ fontWeight: 500, fontSize: 14 }}>
                  {product.name}{" "}
                  {product.size && (
                    <Text type="secondary" style={{ fontSize: 12 }}>
                      ({product.size})
                    </Text>
                  )}
                </div>
                <Text type="secondary" style={{ fontSize: 12 }}>
                  {product.code} · Stok: {product.stock} {product.unit}
                </Text>
              </div>
              <Tag color="blue" style={{ margin: 0 }}>
                Tambah Stok
              </Tag>
            </div>
          ))}
      </div>

      <div
        style={{
          padding: "8px 16px",
          borderTop: "1px solid #f0f0f0",
          display: "flex",
          gap: 16,
          fontSize: 12,
          color: "#8c8c8c",
        }}
      >
        <span>↑↓ navigasi</span>
        <span>↵ pilih</span>
        <span>esc tutup</span>
      </div>
    </Modal>
  );
};

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../utility/supabaseClient";
import { TABLES } from "../utility/constants";
import { USE_MOCK } from "../utility/config";
import { getMockDashboardStats } from "../mocks/dashboard";
import type { DashboardStats } from "../types";

export interface UseDashboardStatsResult {
  stats: DashboardStats;
  isLoading: boolean;
  refetch: () => void;
}

const EMPTY_STATS: DashboardStats = {
  totalProducts: 0,
  totalStock: 0,
  todayInbound: 0,
  todayOutbound: 0,
};

export const useDashboardStats = (): UseDashboardStatsResult => {
  const query = useQuery<DashboardStats, Error, DashboardStats>({
    queryKey: ["dashboard-stats"],
    refetchInterval: USE_MOCK ? false : 60_000,
    queryFn: async (): Promise<DashboardStats> => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return getMockDashboardStats();
      }

      const today = dayjs().format("YYYY-MM-DD");
      const startOfDay = dayjs().startOf("day").toISOString();
      const endOfDay = dayjs().endOf("day").toISOString();

      const [productsRes, inboundRes, outboundRes] = await Promise.all([
        supabaseClient.from(TABLES.products).select("stok"),
        supabaseClient
          .from(TABLES.stockEntries)
          .select("jumlah_barang_masuk")
          .eq("tanggal_pengiriman", today),
        supabaseClient
          .from(TABLES.stockExits)
          .select("jumlah_barang_keluar")
          .gte("created_at", startOfDay)
          .lte("created_at", endOfDay),
      ]);

      if (productsRes.error) throw productsRes.error;
      if (inboundRes.error) throw inboundRes.error;
      if (outboundRes.error) throw outboundRes.error;

      const totalProducts = productsRes.data?.length ?? 0;
      const totalStock =
        productsRes.data?.reduce(
          (sum, p: { stok: number | null }) => sum + (p.stok ?? 0),
          0,
        ) ?? 0;

      const todayInbound =
        inboundRes.data?.reduce(
          (sum, e: { jumlah_barang_masuk: number | null }) =>
            sum + (e.jumlah_barang_masuk ?? 0),
          0,
        ) ?? 0;

      const todayOutbound =
        outboundRes.data?.reduce(
          (sum, exit: { jumlah_barang_keluar: number | null }) =>
            sum + (exit.jumlah_barang_keluar ?? 0),
          0,
        ) ?? 0;

      return { totalProducts, totalStock, todayInbound, todayOutbound };
    },
  });

  return {
    stats: query.data ?? EMPTY_STATS,
    isLoading: query.isLoading,
    refetch: query.refetch,
  };
};

import { useState, useEffect } from "react";

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

import { useEffect } from "react";

interface UseKeyboardShortcutOptions {
  key: string;
  ctrlKey?: boolean;
  shiftKey?: boolean;
  altKey?: boolean;
  metaKey?: boolean;
  enabled?: boolean;
  preventDefault?: boolean;
  ignoreInputs?: boolean;
  handler: (event: KeyboardEvent) => void;
}

const isEditableElement = (el: EventTarget | null): boolean => {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName.toLowerCase();
  return (
    tag === "input" ||
    tag === "textarea" ||
    tag === "select" ||
    el.isContentEditable
  );
};

export const useKeyboardShortcut = ({
  key,
  ctrlKey = false,
  shiftKey = false,
  altKey = false,
  metaKey = false,
  enabled = true,
  preventDefault = true,
  ignoreInputs = false,
  handler,
}: UseKeyboardShortcutOptions) => {
  useEffect(() => {
    if (!enabled) return;

    const onKeyDown = (event: KeyboardEvent) => {
      // Normalisasi nama key (mengabaikan case untuk huruf 'm' dan 'M')
      const isKeyMatch = event.key.toLowerCase() === key.toLowerCase() || event.key === key;
      
      if (!isKeyMatch) return;
      if (event.ctrlKey !== ctrlKey) return;
      if (event.shiftKey !== shiftKey) return;
      if (event.altKey !== altKey) return;
      if (event.metaKey !== metaKey) return;
      if (ignoreInputs && isEditableElement(event.target)) return;

      if (preventDefault) event.preventDefault();

import type { Product } from "../types";

export const useProductSearch = (keyword: string, limit = 30) => {
  const sanitize = (str: string) => str.replace(/[%_()\\'"]/g, "");
  const trimmed = sanitize(keyword).trim();

  const { data, isLoading, refetch } = useQuery<Product[]>({
    queryKey: ["product-search", trimmed, limit],
    queryFn: async () => {
      if (USE_MOCK) {
        if (!trimmed) return mockProducts.slice(0, limit);
        const lower = trimmed.toLowerCase();
        return mockProducts.filter((p) => p.name.toLowerCase().includes(lower) || p.code.toLowerCase().includes(lower)).slice(0, limit);
      }

      let query = supabaseClient.from(TABLES.products).select("*").limit(limit).order("nama", { ascending: true }); 
      if (trimmed) {
        const words = trimmed.split(/\s+/).filter(Boolean);
        const orConditions = words.map((w) => `nama.ilike.%${w}%,kode.ilike.%${w}%`).join(",");
        query = query.or(orConditions);
      }

      const { data, error } = await query;
      if (error) throw error;
      return (data ?? []).map(mapProduct);
    },
    staleTime: 10_000,
  });

  return { products: data ?? [], isLoading, refetch };

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../utility/supabaseClient";
import { TABLES } from "../utility/constants";
import { USE_MOCK } from "../utility/config";
import { mockStockInChart } from "../mocks/stockEntries";
import type { StockInChartItem } from "../types";

export const useStockInChart = (days = 30) => {
  const { data, isLoading, refetch } = useQuery<StockInChartItem[]>({
    queryKey: ["stock-in-chart", days],
    queryFn: async () => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return mockStockInChart.slice(-days);
      }

      const fromDate = dayjs().subtract(days, "day").format("YYYY-MM-DD");

      const { data, error } = await supabaseClient
        .from(TABLES.stockEntries)
        .select("tanggal_pengiriman, jumlah_barang_masuk")
        .gte("tanggal_pengiriman", fromDate)
        .order("tanggal_pengiriman", { ascending: true });

      if (error) throw error;

      const grouped = (data ?? []).reduce<Record<string, number>>(
        (
          acc,
          row: {
            tanggal_pengiriman: string | null;
            jumlah_barang_masuk: number | null;
          },
        ) => {
          if (!row.tanggal_pengiriman) return acc;
          acc[row.tanggal_pengiriman] =
            (acc[row.tanggal_pengiriman] ?? 0) + (row.jumlah_barang_masuk ?? 0);
          return acc;
        },
        {},
      );

      return Object.entries(grouped).map(([date, total]) => ({
        date,
        total,
      }));
    },
  });

  return {
    items: data ?? [],
    isLoading,
    refetch,
  };
};

import { useEffect, useRef, useState } from "react";
import { useNavigate, useSearchParams } from "react-router-dom";
import { Button, Card, Col, Empty, InputNumber, Row, Select, Space, Switch, Tag, Typography, App, Table } from "antd";
import { ArrowLeftOutlined, CheckCircleOutlined, CloseCircleOutlined, EnvironmentOutlined, SwapOutlined, UserOutlined } from "@ant-design/icons";
import { useQueryClient } from "@tanstack/react-query";
import { useProductSearch } from "../../hooks/useProductSearch";
import { usePersons } from "../../hooks/usePersons";
import { useKeyboardShortcut } from "../../hooks/useKeyboardShortcut";
import { useDebounce } from "../../hooks/useDebounce";
import { SetupWizard } from "../../components/entry/SetupWizard";
import { mockWorkplaces } from "../../mocks/entry";
import { supabaseClient } from "../../utility/supabaseClient";
import { USE_MOCK } from "../../utility/config";
import { formatNumber } from "../../utility/format";
import type { Product, Person, Workplace, OutboundLine } from "../../types"; // Menggunakan OutboundLine dari types/index

const { Title, Text } = Typography;

export const EntryPage = () => {
  const navigate = useNavigate();
  const [searchParams, setSearchParams] = useSearchParams();
  const { message } = App.useApp();
  const queryClient = useQueryClient();
  const { persons } = usePersons();

  const initialPerson = searchParams.get("person") || "";
  const initialWorkplace = searchParams.get("workplace") || "";

  const [personName, setPersonName] = useState<string>(initialPerson);
  const [workplaceName, setWorkplaceName] = useState<string>(initialWorkplace);
  const [isBreakdown, setIsBreakdown] = useState<boolean>(true);
  const [wizardOpen, setWizardOpen] = useState<boolean>(!initialPerson || !initialWorkplace);
  const [wizardStep, setWizardStep] = useState<"person" | "workplace">("person");

  const [keyword, setKeyword] = useState("");
  const debouncedKeyword = useDebounce(keyword, 250);
  const { products, isLoading: isLoadingProducts } = useProductSearch(debouncedKeyword, 30);

  const [activeItem, setActiveItem] = useState<Product | null>(null);
  const [activeQty, setActiveQty] = useState<number>(1);
  const [lines, setLines] = useState<OutboundLine[]>([]);
  const [isSubmitting, setIsSubmitting] = useState(false);

  const searchRef = useRef<any>(null);
  const qtyRef = useRef<HTMLInputElement>(null);
  const phase = activeItem ? "quantity" : "search";

  useEffect(() => {
    if (wizardOpen) return;
    if (phase === "search") setTimeout(() => searchRef.current?.focus(), 100);
    else if (phase === "quantity") setTimeout(() => qtyRef.current?.focus(), 100);
  }, [phase, wizardOpen]);

  const handleWizardSelect = (val: Person | Workplace) => {
    if (wizardStep === "person") {
      setPersonName(val.name);
      setWizardStep("workplace");
    } else {
      setWorkplaceName(val.name);
      setWizardOpen(false);
      setSearchParams({ person: personName, workplace: val.name });
    }
  };

  const handleProductSelect = (productId: number) => {
    const item = products.find((p) => p.id === productId);
    if (item) {
      if (item.stock <= 0) return message.error(`Stok ${item.name} sedang habis`);
      setActiveItem(item);
      setActiveQty(1);
      setKeyword("");
    }
  };

  const confirmQuantity = () => {
    if (!activeItem) return;
    if (activeQty <= 0) return message.warning("Jumlah minimal 1");

    const existingQty = lines.find((l) => l.itemId === activeItem.id)?.quantity ?? 0;
    // ✅ PERBAIKAN: Validasi ketat sisi client sebelum submit
    if (existingQty + activeQty > activeItem.stock) {
      return message.warning(`Jumlah melebihi sisa stok! Maksimal tambahan: ${activeItem.stock - existingQty} ${activeItem.unit}`);
    }

    setLines((prev) => {
      const idx = prev.findIndex((l) => l.itemId === activeItem.id);
      if (idx >= 0) {
        const copy = [...prev];
        copy[idx] = { ...copy[idx], quantity: copy[idx].quantity + activeQty };
        return copy;
      }
      return [{ itemId: activeItem.id, itemName: activeItem.name, size: activeItem.size, unit: activeItem.unit, maxStock: activeItem.stock, quantity: activeQty }, ...prev];
    });
    setActiveItem(null);
    setActiveQty(1);
  };

  const handleFinish = async () => {
    if (lines.length === 0) return message.warning("Tambahkan minimal 1 barang");
    if (!personName || !workplaceName) return setWizardOpen(true);

    setIsSubmitting(true);
    try {
      if (!USE_MOCK) {
        // ✅ PERBAIKAN: Gunakan RPC untuk transaksi atomik
        const { error } = await supabaseClient.rpc("issue_stock", {
          p_items: lines.map(l => ({ itemId: l.itemId, quantity: l.quantity, size: l.size, unit: l.unit, itemName: l.itemName })),
          p_person: personName,
          p_destination: workplaceName,
          p_breakdown: isBreakdown
        });
        if (error) throw error;
      }
      message.success(`Pengeluaran ${lines.length} jenis barang berhasil disimpan!`);
      await queryClient.invalidateQueries({ queryKey: ["products"] });
      await queryClient.invalidateQueries({ queryKey: ["stock-exits"] });
      await queryClient.invalidateQueries({ queryKey: ["dashboard-stats"] });
      navigate("/dashboard");
    } catch (err: any) {
      message.error(err?.message || "Gagal menyimpan pengeluaran barang (cek stok)");
    } finally {
      setIsSubmitting(false);
    }
  };

  useKeyboardShortcut({ key: "Enter", ctrlKey: true, enabled: lines.length > 0 && !isSubmitting && !wizardOpen && !activeItem, preventDefault: true, handler: handleFinish });

  return (
    <div style={{ maxWidth: 920, margin: "0 auto", paddingBottom: 32 }}>
      <SetupWizard open={wizardOpen} step={wizardStep} title={wizardStep === "person" ? "Pilih Nama Pekerja" : "Pilih Lokasi Pekerjaan"} placeholder={wizardStep === "person" ? "Ketik nama pekerja..." : "Ketik lokasi pekerjaan..."} options={wizardStep === "person" ? persons : mockWorkplaces} onSelect={handleWizardSelect} onBack={wizardStep === "workplace" ? () => setWizardStep("person") : undefined} onClose={() => { if (!personName || !workplaceName) navigate("/dashboard"); else setWizardOpen(false); }} />

      <div style={{ marginBottom: 16, display: "flex", alignItems: "center", justifyContent: "space-between", flexWrap: "wrap", gap: 12 }}>
        <Space align="center">
          <Button icon={<ArrowLeftOutlined />} onClick={() => navigate("/dashboard")} type="text" size="large" />
          <div>
            <Title level={3} style={{ margin: 0 }}>Pengeluaran Barang</Title>
            <Text type="secondary">Pilih barang yang diambil dan tentukan jumlahnya</Text>
          </div>
        </Space>
        <Card size="small" style={{ borderRadius: 10, background: isBreakdown ? "#e6f4ff" : "#fffbe6", borderColor: isBreakdown ? "#91caff" : "#ffe58f" }}>
          <Space align="center" size={10}>
            <div>
              <Text strong style={{ fontSize: 13, display: "block" }}>Status: {isBreakdown ? "Breakdown" : "Non Breakdown"}</Text>
            </div>
            <Switch checked={isBreakdown} onChange={setIsBreakdown} />
          </Space>
        </Card>
      </div>

      <Card variant="borderless" style={{ borderRadius: 12, marginBottom: 16 }} styles={{ body: { padding: 16 } }}>
        <Row justify="space-between" align="middle">
          <Col>
            <Space size={12} wrap>
              <Tag icon={<UserOutlined />} color="blue" style={{ fontSize: 14, padding: "6px 12px" }}>Pekerja: <strong>{personName || "-"}</strong></Tag>
              <Tag icon={<EnvironmentOutlined />} color="cyan" style={{ fontSize: 14, padding: "6px 12px" }}>Lokasi: <strong>{workplaceName || "-"}</strong></Tag>
            </Space>
          </Col>
          <Col><Button size="small" icon={<SwapOutlined />} onClick={() => { setWizardStep("person"); setWizardOpen(true); }}>Ubah Sesi</Button></Col>
        </Row>
      </Card>

      <Card variant="borderless" style={{ borderRadius: 12, marginBottom: 16 }} styles={{ body: { padding: 20 } }}>
        {phase === "search" && (
          <Select showSearch autoFocus ref={searchRef} size="large" style={{ width: "100%" }} placeholder="Ketik nama, ukuran, atau kode barang..." filterOption={false} value={null} loading={isLoadingProducts} onSearch={setKeyword} onChange={handleProductSelect} options={products.map((p) => ({ value: p.id, label: `${p.name} ${p.size ? `(${p.size})` : ""} — Stok: ${p.stock} ${p.unit}` }))} notFoundContent={isLoadingProducts ? "Mencari data..." : "Barang tidak ditemukan"} />
        )}
        {phase === "quantity" && activeItem && (
          <div>
            <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 12 }}>
              <Text strong style={{ fontSize: 16 }}>{activeItem.name} {activeItem.size ? `(${activeItem.size})` : ""}</Text>
              <Tag color="blue">Sisa Stok: {activeItem.stock} {activeItem.unit}</Tag>
            </div>
            <Space.Compact style={{ width: "100%" }}>
              <Button size="large" onClick={() => setActiveQty((q) => Math.max(1, q - 1))} style={{ width: 64, height: 50 }}>−</Button>
              <InputNumber ref={qtyRef as never} value={activeQty} onChange={(val) => setActiveQty(Number(val) || 1)} size="large" min={1} max={activeItem.stock} controls={false} onPressEnter={(e) => { e.preventDefault(); confirmQuantity(); }} onKeyDown={(e) => { if (e.key === "Escape") setActiveItem(null); if (e.key === "ArrowUp") setActiveQty((q) => Math.min(activeItem.stock, q + 1)); if (e.key === "ArrowDown") setActiveQty((q) => Math.max(1, q - 1)); }} style={{ flex: 1, textAlign: "center", fontSize: 22, fontWeight: 600, height: 50 }} />
              <Button size="large" onClick={() => setActiveQty((q) => Math.min(activeItem.stock, q + 1))} style={{ width: 64, height: 50 }}>+</Button>
            </Space.Compact>
            <Space style={{ marginTop: 14, width: "100%", justifyContent: "space-between" }}>
              <Space size="small"><Tag>Enter</Tag> <Text type="secondary">tambah</Text> <Tag>Esc</Tag> <Text type="secondary">batal</Text></Space>
              <Space><Button onClick={() => setActiveItem(null)}>Batal</Button><Button type="primary" onClick={confirmQuantity}>Tambah</Button></Space>
            </Space>
          </div>
        )}
      </Card>

      <Card title={`📦 Daftar Pengeluaran (${lines.length} jenis)`} variant="borderless" style={{ borderRadius: 12 }} styles={{ body: { padding: 0 } }} extra={<Button type="primary" icon={<CheckCircleOutlined />} disabled={lines.length === 0} loading={isSubmitting} onClick={handleFinish}>Simpan (Ctrl+Enter)</Button>}>
        {lines.length === 0 ? <div style={{ padding: 32 }}><Empty description="Belum ada barang di daftar" /></div> : (
          <Table dataSource={lines} rowKey="itemId" pagination={false} size="middle" columns={[
            { title: "Nama Barang", dataIndex: "itemName", render: (t, r) => <Space direction="vertical" size={0}><Text strong>{t}</Text>{r.size && <Text type="secondary">{r.size}</Text>}</Space> },

          ]} />
        )}
      </Card>

      d && d !== "-" ? new Date(d).getTime() || 0 : 0;
    return [...data]
      .sort((a, b) => time(b.date) - time(a.date))
      .slice(0, 4);
  }, [data]);

  const filteredAndSortedData = useMemo(() => {
    const filtered = data.filter((item) => {

      const bDone = COMPLETED.includes(b.status);
      if (aDone && !bDone) return 1;
      if (!aDone && bDone) return -1;

import type { Product, StockExit, StockExitItem } from "../types";

// ✅ PERBAIKAN: Tipe eksplisit menghindari unknown/any
interface DBProductRow {
  id: number; kode: string; nama: string; ukuran: string | null; spesifikasi: string | null;
  kategori: string; satuan: string; stok: number; created_at: string;
}

export const mapProduct = (row: any): Product => {
  const r = row as DBProductRow;
  return {
    id: r.id, code: r.kode, name: r.nama, size: r.ukuran, specification: r.spesifikasi,
    category: r.kategori ?? "Material", categoryId: null, categoryName: null,
    unit: r.satuan ?? "pcs", stock: r.stok ?? 0, minStock: 0, costPrice: 0,
    status: undefined, createdAt: r.created_at ?? new Date().toISOString(),
  };
};

interface StockExitRow {
  id: number; nomor_transaksi: string | null; barang_id: number; nama_pekerja: string | null;
  nama_barang: string; ukuran_barang: string | null; satuan_barang: string | null;
  jumlah_barang_keluar: number; tujuan: string | null; is_breakdown: boolean | null; created_at: string;
}

export const mapStockExits = (rows: any[]): StockExit[] => {
  const typedRows = rows as StockExitRow[];
  const grouped = new Map<string, StockExit>();

  for (const row of typedRows) {
    const key = row.nomor_transaksi ?? `temp-${row.id}`;
    if (!grouped.has(key)) {
      const created = new Date(row.created_at);
      grouped.set(key, {
        id: row.id, nomorTransaksi: key, timestamp: row.created_at, date: created.toISOString().split("T")[0],
        time: created.toISOString().slice(11, 16), person: row.nama_pekerja ?? "-",
        destination: row.tujuan, isBreakdown: row.is_breakdown ?? true, items: [],
      });
    }
    grouped.get(key)!.items.push({
      itemId: row.barang_id, itemName: row.nama_barang, size: row.ukuran_barang,
      unit: row.satuan_barang ?? "pcs", quantity: row.jumlah_barang_keluar,
    });
  }
  return Array.from(grouped.values());
};

import { createClient } from "@refinedev/supabase";
import { USE_MOCK } from "./config";

const SUPABASE_URL = import.meta.env.VITE_SUPABASE_URL;
const SUPABASE_KEY = import.meta.env.VITE_SUPABASE_KEY;

if (!USE_MOCK && (!SUPABASE_URL || !SUPABASE_KEY)) {
  console.warn("⚠️  Supabase env vars missing");
}

export const supabaseClient = createClient(
  SUPABASE_URL || "https://placeholder.supabase.co",
  SUPABASE_KEY || "placeholder-key",
  {
    db: { schema: "public" },
    auth: {
      persistSession: !USE_MOCK,
      autoRefreshToken: !USE_MOCK,
      detectSessionInUrl: !USE_MOCK,
    },
  },
);
