import type { AccessControlProvider } from "@refinedev/core";
import type { PermissionKey, RoleDefinition, UserRole } from "../../shared/types";

export const CUSTOM_ROLES_STORAGE_KEY = "custom-system-roles";

/**
 * Daftar role bawaan sistem berdasarkan urutan hierarki tertinggi ke terendah:
 * 1. superuser (Level 100) - Hak akses login tertinggi (Point 1, Point 2, & Edit)
 * 2. supervisor (Level 70) - Di bawah Super User (Hak akses Edit)
 * 3. admin (Level 50)      - Di bawah Supervisor (Hak akses Edit)
 * 4. user (Level 0)        - Pengunjung publik / belum login (Hanya baca)
 */
export const BUILT_IN_ROLES: Record<string, RoleDefinition> = {
  superuser: {
    key: "superuser",
    label: "Super User",
    badge: "SU",
    color: "purple",
    level: 100,
    isAuthenticated: true,
    permissions: ["approve_accounts", "manage_roles", "edit_data"],
  },
  supervisor: {
    key: "supervisor",
    label: "Supervisor",
    badge: "SV",
    color: "cyan",
    level: 70,
    isAuthenticated: true,
    permissions: ["edit_data"],
  },
  admin: {
    key: "admin",
    label: "Admin",
    badge: "AS",
    color: "blue",
    level: 50,
    isAuthenticated: true,
    permissions: ["edit_data"],
  },
  user: {
    key: "user",
    label: "User (Public)",
    badge: "US",
    color: "green",
    level: 0,
    isAuthenticated: false,
    permissions: [],
  },
};

export const normalizeRoleKey = (role?: UserRole | string | null): string => {
  const cleaned = String(role ?? "").trim().toLowerCase();
  return cleaned || "user";
};

export const getCustomRoles = (): RoleDefinition[] => {
  try {
    const raw = localStorage.getItem(CUSTOM_ROLES_STORAGE_KEY);
    if (!raw) return [];
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
};

export const saveCustomRoles = (roles: RoleDefinition[]) => {
  try {
    localStorage.setItem(CUSTOM_ROLES_STORAGE_KEY, JSON.stringify(roles));
  } catch {
    // ignore storage error
  }
};

export const resetCustomRoles = () => {
  try {
    localStorage.removeItem(CUSTOM_ROLES_STORAGE_KEY);
  } catch {
    // ignore
  }
};

/**
 * Menambahkan role login baru secara dinamis (hanya dapat dilakukan oleh Super User di UI,
 * atau dipanggil saat inisialisasi). Role baru otomatis memiliki izin 'edit_data'
 * dan berada di bawah Super User.
 */
export const registerCustomRole = (input: {
  key: string;
  label?: string;
  badge?: string;
  color?: string;
  level?: number;
  permissions?: PermissionKey[];
}): RoleDefinition => {
  const key = normalizeRoleKey(input.key).replace(/\s+/g, "_");
  if (!key || key === "user" || key === "superuser") {
    throw new Error("Nama role tidak valid atau sudah digunakan sistem");
  }

  const label =
    input.label?.trim() ||
    key
      .split("_")
      .map((w) => w.charAt(0).toUpperCase() + w.slice(1))
      .join(" ");

  const badge =
    input.badge?.trim().toUpperCase().slice(0, 2) ||
    label
      .split(" ")
      .map((w) => w[0])
      .join("")
      .toUpperCase()
      .slice(0, 2) ||
    "RL";

  // Pastikan hanya superuser yang memiliki approve_accounts & manage_roles
  const safePermissions: PermissionKey[] = (
    input.permissions ?? ["edit_data"]
  ).filter((p) => p === "edit_data");

  const definition: RoleDefinition = {
    key,
    label,
    badge,
    color: input.color || "geekblue",
    level: typeof input.level === "number" ? Math.min(90, Math.max(10, input.level)) : 40,
    isAuthenticated: true,
    permissions: safePermissions.length > 0 ? safePermissions : ["edit_data"],
  };

  const existing = getCustomRoles().filter((r) => normalizeRoleKey(r.key) !== key);
  saveCustomRoles([...existing, definition]);
  return definition;
};

export const getAllRoleDefinitions = (): RoleDefinition[] => {
  const custom = getCustomRoles();
  const builtInList = [
    BUILT_IN_ROLES.superuser,
    BUILT_IN_ROLES.supervisor,
    BUILT_IN_ROLES.admin,
    ...custom.filter(
      (c) => !BUILT_IN_ROLES[normalizeRoleKey(c.key)],
    ),
    BUILT_IN_ROLES.user,
  ];
  return builtInList.sort((a, b) => b.level - a.level);
};

export const getAssignableRoles = (): RoleDefinition[] => {
  return getAllRoleDefinitions().filter(
    (r) => r.isAuthenticated && r.key !== "superuser",
  );
};

export const getRoleDefinition = (role?: UserRole | string | null): RoleDefinition => {
  const key = normalizeRoleKey(role);
  if (BUILT_IN_ROLES[key]) {
    return BUILT_IN_ROLES[key];
  }
  const foundCustom = getCustomRoles().find((r) => normalizeRoleKey(r.key) === key);
  if (foundCustom) {
    return foundCustom;
  }
  // Fallback dinamis: jika ada role login baru yang belum terdaftar eksplisit,
  // anggap sebagai role login di bawah admin yang memiliki hak edit data.
  const fallbackLabel = key
    .split(/[_-\s]+/)
    .map((w) => w.charAt(0).toUpperCase() + w.slice(1))
    .join(" ");
  return {
    key,
    label: fallbackLabel || "Staff",
    badge: (fallbackLabel.slice(0, 2) || "ST").toUpperCase(),
    color: "geekblue",
    level: 30,
    isAuthenticated: true,
    permissions: ["edit_data"],
  };
};

export const getRoleLevel = (role?: UserRole | string | null): number =>
  getRoleDefinition(role).level;

export const isAuthenticatedRole = (role?: UserRole | string | null): boolean =>
  getRoleDefinition(role).isAuthenticated;

export const isSuperUser = (role?: UserRole | string | null): boolean =>
  normalizeRoleKey(role) === "superuser";

/**
 * Point 1: Hanya Super User yang dapat menyetujui / menolak pendaftaran akun baru
 */
export const canApproveAccounts = (role?: UserRole | string | null): boolean =>
  isSuperUser(role);

/**
 * Point 2: Hanya Super User yang dapat mengelola role & mengubah level akses akun
 */
export const canManageRoles = (role?: UserRole | string | null): boolean =>
  isSuperUser(role);

export const canAccessNotifications = (role?: UserRole | string | null): boolean =>
  isSuperUser(role);

/**
 * Mengedit data (CRUD barang, edit judul SPB/PO, barang masuk/keluar):
 * Super User, Supervisor, Admin, serta role login tambahan semuanya dapat mengedit.
 */
export const canEditData = (role?: UserRole | string | null): boolean => {
  const def = getRoleDefinition(role);
  return def.isAuthenticated && def.permissions.includes("edit_data");
};

export const canManageBarang = (role?: UserRole | string | null): boolean =>
  canEditData(role);

export const accessControlProvider: AccessControlProvider = {
  can: async ({ resource, action }) => {
    const user = getStoredUser();
    const role: UserRole = user?.role ?? "user";

    if (
      resource === "notifications" ||
      resource === "notifikasi" ||
      resource === "account_requests" ||
      resource === "roles"
    ) {
      return {
        can: canAccessNotifications(role),
        reason: "Hanya Super User yang memiliki akses ke halaman Notifikasi & Manajemen Role",
      };
    }

    if (
      resource === "barang" &&
      (action === "create" || action === "edit" || action === "delete")
    ) {
      return {
        can: canEditData(role),
        reason: "Hanya pengguna yang sudah login (Super User, Supervisor, Admin) yang dapat mengelola data barang",
      };
    }

    if (resource === "barang_masuk" || resource === "pengeluaran") {
      return {
        can: canEditData(role),
        reason: "Silakan masuk terlebih dahulu untuk mengakses fitur ini",
      };
    }

    return { can: true };
  },
};

import type { AuthProvider, AuthActionResponse, CheckResponse } from "@refinedev/core";
import type { User } from "@supabase/supabase-js";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import type { UserRole } from "../../shared/types";
import { registerAccountRequest } from "../../features/auth/accountRequests";
import {
  getRoleDefinition,
  isAuthenticatedRole,
  normalizeRoleKey,
} from "./accessControlProvider";

export const STORAGE_KEY = "mock-auth-user";
export const AUTH_CHANGED_EVENT = "auth-user-changed";

export interface StoredAuthUser {
  id?: string;
  email: string;
  name: string;
  role: UserRole;
}

/**
 * Cache identitas untuk render UI yang sinkron. BUKAN sumber kebenaran:
 * sumber kebenaran adalah sesi Supabase (JWT) dan RLS di database.
 * Cache dibersihkan otomatis saat sesi berakhir (lihat initAuthSync).
 */
export const getStoredUser = (): StoredAuthUser | null => {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
};

export const setStoredUser = (user: StoredAuthUser | null) => {
  try {
    if (!user) {
      localStorage.removeItem(STORAGE_KEY);
    } else {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(user));
    }
    if (typeof window !== "undefined") {
      window.dispatchEvent(new Event(AUTH_CHANGED_EVENT));
    }
  } catch {
    // ignore
  }
};

/**
 * Role HANYA dibaca dari `app_metadata` — field ini hanya bisa diubah server
 * (service role / RPC superuser), tidak bisa diubah user dari browser.
 * `user_metadata` sengaja TIDAK dipakai karena dapat dimodifikasi user sendiri.
 *
 * @returns role terautentikasi, atau null jika belum disetujui / tidak valid.
 */
export const resolveRoleFromSupabaseUser = (
  user: Pick<User, "app_metadata"> | null | undefined,
): UserRole | null => {
  const raw = user?.app_metadata?.role;
  if (typeof raw !== "string") return null;
  const key = normalizeRoleKey(raw);
  return isAuthenticatedRole(key) ? key : null;
};

const toStoredUser = (user: User, role: UserRole): StoredAuthUser => {
  const email = user.email ?? "";
  const metaName = user.user_metadata?.name;
  return {
    id: user.id,
    email,
    name:
      typeof metaName === "string" && metaName.trim()
        ? metaName.trim()
        : email.split("@")[0] || getRoleDefinition(role).label,
    role,
  };
};

const sameStoredUser = (a: StoredAuthUser | null, b: StoredAuthUser) =>
  !!a && a.id === b.id && a.email === b.email && a.name === b.name && a.role === b.role;

/**
 * Menjaga cache identitas selalu sejalan dengan sesi Supabase: dibersihkan saat
 * logout / token kedaluwarsa, dan diperbarui saat sesi/role berubah.
 */
let authSyncInitialized = false;
export const initAuthSync = () => {
  if (authSyncInitialized || typeof window === "undefined") return;
  authSyncInitialized = true;
  try {
    supabaseClient.auth.onAuthStateChange((_event, session) => {
      const role = resolveRoleFromSupabaseUser(session?.user);
      if (!session?.user || !role) {
        if (getStoredUser()) setStoredUser(null);
        return;
      }
      const next = toStoredUser(session.user, role);
      if (!sameStoredUser(getStoredUser(), next)) setStoredUser(next);
    });
  } catch {
    // klien auth tidak tersedia (mis. lingkungan test tanpa auth) -> lewati
  }
};

initAuthSync();

const loginError = (message: string): AuthActionResponse => ({
  success: false,
  error: { name: "LoginError", message },
});

export const authProvider: AuthProvider = {
  login: async ({ email, password }): Promise<AuthActionResponse> => {
    const normalizedEmail = String(email || "").trim().toLowerCase();
    if (!normalizedEmail || !password) {
      return loginError("Email dan password wajib diisi");
    }

    const { data, error } = await supabaseClient.auth.signInWithPassword({
      email: normalizedEmail,
      password,
    });

    // Pesan generik: tidak membocorkan apakah email terdaftar atau tidak.
    if (error || !data?.user) {
      return loginError("Email atau password salah");
    }

    const role = resolveRoleFromSupabaseUser(data.user);
    if (!role) {
      // Akun ada tapi belum disetujui / belum diberi role -> jangan biarkan sesi aktif.
      await supabaseClient.auth.signOut();
      setStoredUser(null);
      return loginError(
        "Akun Anda masih menunggu persetujuan Super User atau belum diberi role. Hubungi Super User.",
      );
    }

    setStoredUser(toStoredUser(data.user, role));
    return { success: true, redirectTo: "/dashboard" };
  },

  register: async ({ name, email, password, role }: any): Promise<AuthActionResponse> => {
    try {
      await registerAccountRequest({
        name: name || String(email || "").split("@")[0] || "Calon Pengguna",
        email,
        password,
        role: role || "admin",
      });
      return { success: true, redirectTo: "/login" };
    } catch (err) {
      return {
        success: false,
        error: {
          name: "RegisterError",
          message: (err as Error)?.message || "Gagal mendaftarkan akun",
        },
      };
    }
  },

  logout: async (): Promise<AuthActionResponse> => {
    setStoredUser(null);
    try {
      await supabaseClient.auth.signOut();
    } catch {
      // sesi lokal sudah dibersihkan
    }
    return { success: true, redirectTo: "/dashboard" };
  },

  check: async (): Promise<CheckResponse> => {
    const { data } = await supabaseClient.auth.getSession();
    const user = data?.session?.user;
    const role = resolveRoleFromSupabaseUser(user);

    if (!user || !role) {
      if (getStoredUser()) setStoredUser(null);
      return { authenticated: false, redirectTo: "/login" };
    }

    const next = toStoredUser(user, role);
    if (!sameStoredUser(getStoredUser(), next)) setStoredUser(next);
    return { authenticated: true };
  },

  getIdentity: async () => {
    const { data } = await supabaseClient.auth.getSession();
    const user = data?.session?.user;
    const role = resolveRoleFromSupabaseUser(user);

    if (user && role) {
      const stored = toStoredUser(user, role);
      return { id: stored.id, name: stored.name, email: stored.email, role };
    }

    return {
      id: "public-user",
      name: "User (Public)",
      email: "",
      role: "user" as UserRole,
    };
  },

  getPermissions: async () => {
    const { data } = await supabaseClient.auth.getSession();
    const role = resolveRoleFromSupabaseUser(data?.session?.user);
    return [role ?? "user"];
  },

  onError: async (error) => {
    // Sesi tidak valid / kedaluwarsa (401) -> paksa logout & kembali ke login.
    const status = (error as any)?.status ?? (error as any)?.statusCode;
    if (status === 401) {
      return { logout: true, redirectTo: "/login", error };
    }
    return { error };
  },
};

import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import type { AccountRequest, UserRole } from "../../shared/types";
import { requirePermission } from "../../shared/lib/authGuard";

/** Kebijakan password minimum (selaras dengan pengaturan Supabase Auth). */
export const MIN_PASSWORD_LENGTH = 8;

const REQUESTS_STORAGE_KEY = "account-registration-requests";

/**
 * Cache lokal daftar permintaan akun (hanya untuk badge/UX cepat).
 * TIDAK PERNAH menyimpan password. Sumber kebenaran: tabel account_requests
 * di server + Supabase Auth.
 */
export const readLocalRequests = (): AccountRequest[] => {
  try {
    const raw = localStorage.getItem(REQUESTS_STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
};

export const writeLocalRequests = (requests: AccountRequest[]) => {
  try {
    localStorage.setItem(REQUESTS_STORAGE_KEY, JSON.stringify(requests));
  } catch {
    // ignore storage error
  }
};

export const resetAccountRequests = () => {
  try {
    localStorage.removeItem(REQUESTS_STORAGE_KEY);
  } catch {
    // ignore
  }
};

export const getPendingRequestsCount = (): number => {
  return readLocalRequests().filter((r) => r.status === "pending").length;
};

const sanitizeAssignableRole = (role?: UserRole | string | null): UserRole => {
  const key = normalizeRoleKey(role);
  if (!key || key === "user" || key === "superuser") {
    return "admin";
  }
  return key;
};

const mapRow = (row: any): AccountRequest => ({
  id: String(row.id),
  name: row.name || "Calon Pengguna",
  email: String(row.email || ""),
  role: sanitizeAssignableRole(row.role ?? "admin"),
  status: row.status || "pending",
  createdAt: row.created_at || row.createdAt || new Date().toISOString(),
  decidedAt: row.decided_at || row.decidedAt || null,
});

const patchLocalRequest = (
  requestId: string,
  patch: Partial<AccountRequest>,
  fallback: AccountRequest,
): AccountRequest => {
  const list = readLocalRequests();
  const idx = list.findIndex((r) => String(r.id) === String(requestId));
  if (idx >= 0) {
    const updated = { ...list[idx], ...patch };
    list[idx] = updated;
    writeLocalRequests(list);
    return updated;
  }
  return { ...fallback, ...patch };
};

/**
 * Daftar permintaan akun. RLS membatasi baca hanya untuk Super User;
 * jika server menolak/tabel belum ada, jatuh ke cache lokal.
 */
export const getAccountRequests = async (): Promise<AccountRequest[]> => {
  try {
    const res = await supabaseClient.from(TABLES.accountRequests).select("*");
    if (!res.error && Array.isArray(res.data)) {
      const fromServer = res.data.map(mapRow).filter((r) => r.email);
      writeLocalRequests(fromServer);
      return fromServer;
    }
  } catch {
    // fallback ke cache lokal
  }
  return readLocalRequests();
};

/**
 * Pendaftaran akun: membuat user di Supabase Auth (password di-hash oleh
 * server, tidak pernah disimpan/dikirim ke tabel kita) lalu mencatat permintaan
 * persetujuan TANPA password. Akun baru tidak punya role sampai Super User
 * menyetujui (role ditulis server-side ke app_metadata).
 */
export const registerAccountRequest = async (input: {
  name: string;
  email: string;
  password: string;
  role?: UserRole;
}): Promise<AccountRequest> => {
  const name = input.name.trim();
  const email = input.email.trim().toLowerCase();
  const password = input.password;
  const requestedRole = sanitizeAssignableRole(input.role ?? "admin");

  if (!name || !email || !password) {
    throw new Error("Nama, email, dan password wajib diisi");
  }
  if (password.length < MIN_PASSWORD_LENGTH) {
    throw new Error(`Password minimal ${MIN_PASSWORD_LENGTH} karakter`);
  }

  const { data, error } = await supabaseClient.auth.signUp({
    email,
    password,
    options: { data: { name } },
  });
  if (error) {
    throw new Error(error.message || "Gagal membuat akun");
  }
  // Supabase mengembalikan user dengan identities kosong jika email sudah terdaftar.
  if (data?.user && Array.isArray(data.user.identities) && data.user.identities.length === 0) {
    throw new Error("Email ini sudah terdaftar");
  }

  // Pastikan tidak ada sesi aktif sebelum akun disetujui.
  try {
    await supabaseClient.auth.signOut();
  } catch {
    // ignore
  }

  const newRequest: AccountRequest = {
    id: `req-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
    name,
    email,
    role: requestedRole,
    status: "pending",
    createdAt: new Date().toISOString(),
    decidedAt: null,
  };

  const { error: insertError } = await supabaseClient
    .from(TABLES.accountRequests)
    .insert([
      {
        id: newRequest.id,
        name: newRequest.name,
        email: newRequest.email,
        role: newRequest.role,
        status: "pending",
        created_at: newRequest.createdAt,
      },
    ]);
  if (insertError) {
    throw new Error("Akun dibuat, tetapi permintaan persetujuan gagal dikirim. Hubungi Super User.");
  }

  writeLocalRequests([
    newRequest,
    ...readLocalRequests().filter((r) => r.email.toLowerCase() !== email),
  ]);

  return newRequest;
};

const placeholder = (id: string, status: AccountRequest["status"]): AccountRequest => ({
  id,
  name: "Pengguna",
  email: "",
  role: "admin",
  status,
  createdAt: new Date().toISOString(),
  decidedAt: null,
});

/**
 * Menyetujui akun + menetapkan role (hanya Super User).
 * Otorisasi ditegakkan DI SERVER: RPC `approve_account` memeriksa JWT
 * (app_metadata.role = 'superuser') lalu menulis role ke app_metadata user.
 */
export const approveAccountRequest = async (
  requestId: string,
  assignedRole?: UserRole,
): Promise<AccountRequest> => {
  requirePermission("approve_accounts", "Setujui pendaftaran akun");

  const existing = readLocalRequests().find((r) => String(r.id) === String(requestId));
  const targetRole = sanitizeAssignableRole(assignedRole ?? existing?.role ?? "admin");

  const { error } = await supabaseClient.rpc("approve_account", {
    p_request_id: String(requestId),
    p_role: targetRole,
  });
  if (error) throw new Error(error.message || "Gagal menyetujui akun");

  return patchLocalRequest(
    requestId,
    { status: "approved", role: targetRole, decidedAt: new Date().toISOString() },
    placeholder(requestId, "approved"),
  );
};

/** Menolak akun (hanya Super User) — role dicabut di server. */
export const rejectAccountRequest = async (
  requestId: string,
): Promise<AccountRequest> => {
  requirePermission("approve_accounts", "Tolak pendaftaran akun");

  const { error } = await supabaseClient.rpc("reject_account", {
    p_request_id: String(requestId),
  });
  if (error) throw new Error(error.message || "Gagal menolak akun");

  return patchLocalRequest(
    requestId,
    { status: "rejected", decidedAt: new Date().toISOString() },
    placeholder(requestId, "rejected"),
  );
};

/** Mengubah role akun yang sudah disetujui (hanya Super User). */
export const updateAccountRole = async (
  requestId: string,
  newRole: UserRole,
): Promise<AccountRequest> => {
  requirePermission("manage_roles", "Ubah role akun");

  const targetRole = sanitizeAssignableRole(newRole);
  const existing = readLocalRequests().find((r) => String(r.id) === String(requestId));
  if (!existing) {
    throw new Error("Data akun tidak ditemukan");
  }

  const { error } = await supabaseClient.rpc("set_account_role", {
    p_request_id: String(requestId),
    p_role: targetRole,
  });
  if (error) throw new Error(error.message || "Gagal mengubah role akun");

  return patchLocalRequest(requestId, { role: targetRole }, existing);
};

export interface ChatbotReply {
  text: string;
  suggestions: string[];
  actionLinks?: { label: string; path: string }[];
}

export const DEFAULT_CHAT_SUGGESTIONS: string[] = [
  "Ringkasan stok gudang",
  "Barang stok menipis",
  "Status monitoring SPB",
  "Info hak akses role",
];

const fetchProductsSnapshot = async (): Promise<Product[]> => {
  try {
    const res = await supabaseClient.from(TABLES.products).select("*");
    if (!res.error && Array.isArray(res.data) && res.data.length > 0) {
      return res.data.map(mapProduct);
    }
  } catch {
    // fallback ke mock
  }
  if (USE_MOCK || mockProducts.length > 0) {
    return mockProducts;
  }
  return [];
};

const fetchSpbRowsSnapshot = async (): Promise<any[]> => {
  try {
    const res = await supabaseClient.from(TABLES.spb).select("*");
    if (!res.error && Array.isArray(res.data)) {
      return res.data;
    }
  } catch {
    // ignore
  }
  return [];
};

const fetchPoRowsSnapshot = async (): Promise<any[]> => {
  try {
    const res = await supabaseClient.from(TABLES.po).select("*");
    if (!res.error && Array.isArray(res.data)) {
      return res.data;
    }
  } catch {
    // ignore
  }
  return [];
};

const fetchSjRowsSnapshot = async (): Promise<any[]> => {
  try {
    const res = await supabaseClient.from(TABLES.suratJalan).select("*");
    if (!res.error && Array.isArray(res.data)) {
      return res.data;
    }
  } catch {
    // ignore
  }
  return [];
};

export const generateChatbotResponse = async (
  rawInput: string,
  currentRole: UserRole = "user",
): Promise<ChatbotReply> => {
  const query = rawInput.trim().toLowerCase();
  const roleDef = getRoleDefinition(currentRole);

  if (!query) {
    return {
      text: "Silakan ketik pertanyaan Anda mengenai stok barang, status SPB/PO/Surat Jalan, atau hak akses sistem.",
      suggestions: DEFAULT_CHAT_SUGGESTIONS,
    };
  }

  // 1. Pertanyaan seputar Role / Hak Akses / Login
  if (
    query.includes("hak akses") ||
    query.includes("role") ||
    query.includes("login") ||
    query.includes("superuser") ||
    query.includes("super user") ||
    query.includes("supervisor") ||
    query.includes("admin")
  ) {
    const rolesList = getAllRoleDefinitions()
      .filter((r) => r.isAuthenticated)
      .map((r) => r.label)
      .join(" > ");

    const canEdit = canEditData(currentRole);
    const isSuper = canApproveAccounts(currentRole);

    return {
      text: [
        `Status Anda saat ini: **${roleDef.label}**`,
        `• Urutan hierarki login: **${rolesList}**`,
        `• Persetujuan akun baru & kelola role (Point 1 & 2): **Hanya Super User** (${isSuper ? "Anda memiliki akses" : "Khusus Super User"})`,
        `• Mengedit data barang & judul SPB/PO: **Super User, Supervisor, Admin, & role login lainnya** (${canEdit ? "Anda dapat mengedit" : "Silakan login terlebih dahulu"})`,
      ].join("\n"),
      suggestions: DEFAULT_CHAT_SUGGESTIONS,
      actionLinks: isSuper
        ? [
            { label: "Buka Notifikasi Super User", path: "/notifications" },
            { label: "Buka Inventory", path: "/inventory" },
          ]
        : [{ label: "Buka Halaman Login", path: "/login" }],
    };
  }

  // 2. Pertanyaan seputar SPB / PO / Surat Jalan
  if (
    query.includes("spb") ||
    query.includes("po") ||
    query.includes("surat jalan") ||
    query.includes("monitoring")
  ) {
    const [spbRows, poRows, sjRows] = await Promise.all([
      fetchSpbRowsSnapshot(),
      fetchPoRowsSnapshot(),
      fetchSjRowsSnapshot(),
    ]);

    const realItems = spbRows.filter((r) => !isKeteranganBarang(r));
    if (realItems.length === 0 && poRows.length === 0 && sjRows.length === 0) {
      return {
        text: "Saat ini belum ada data SPB, PO, atau Surat Jalan yang tercatat di database. Anda dapat memantau dokumen melalui halaman Monitoring SPB.",
        suggestions: DEFAULT_CHAT_SUGGESTIONS,
        actionLinks: [{ label: "Buka Monitoring SPB", path: "/monitoring-spb" }],
      };
    }

    // Cek apakah pengguna menanyakan nomor SPB spesifik (misal: "spb 10" atau "spb 12")
    const numberMatch = query.match(/(?:spb|po|sj|nomor|no)\s*[-:#]?\s*([0-9a-zA-Z/-]+)/i);
    if (numberMatch && numberMatch[1] && !["status", "monitoring", "belum", "sudah", "semua"].includes(numberMatch[1].toLowerCase())) {
      const targetNo = numberMatch[1].toLowerCase();
      const matchedSpb = realItems.filter(
        (r) =>
          String(r.no_spb ?? "").toLowerCase().includes(targetNo) ||
          String(r.no_po ?? "").toLowerCase().includes(targetNo) ||
          cleanSpbTitle(r.judul_spb).toLowerCase().includes(targetNo),
      );

      if (matchedSpb.length > 0) {
        const sample = matchedSpb[0];
        const titleText = cleanSpbTitle(sample.judul_spb) || "Tanpa Judul";
        const itemPreview = matchedSpb
          .slice(0, 5)
          .map(
            (item) =>
              `• ${item.nama_barang || "-"} (${item.jumlah_permintaan ?? 0} ${item.satuan || "unit"}) — Status: ${resolveSpbItemStatus(item)}`,
          )
          .join("\n");

        return {
          text: [
            `Ditemukan **${matchedSpb.length} item barang** untuk dokumen **SPB ${sample.no_spb || "-"}** (${titleText}):`,
            itemPreview,
          ].join("\n"),
          suggestions: ["Status monitoring SPB", "Ringkasan stok gudang"],
          actionLinks: [
            {
              label: `Lihat Detail SPB ${sample.no_spb || ""}`,
              path: `/monitoring-spb/detail?spb=${encodeURIComponent(String(sample.no_spb || ""))}`,
            },
          ],
        };
      }
    }

    // Ringkasan status seluruh item SPB
    const statusCounts: Record<string, number> = {
      "Belum Diproses": 0,
      "Belum PO": 0,
      "Sudah PO": 0,
      "Datang Sesuai": 0,
      "Datang Kurang": 0,
      "Datang Lebih": 0,
    };

    realItems.forEach((row) => {
      const st = resolveSpbItemStatus(row);
      statusCounts[st] = (statusCounts[st] ?? 0) + 1;
    });

    const uniqueSpb = new Set(
      realItems.map((r) => `${r.no_spb ?? ""}__${cleanSpbTitle(r.judul_spb)}`),
    ).size;

    return {
      text: [
        `Ringkasan **Monitoring SPB & PO** saat ini:`,
        `• Total Kartu SPB: **${uniqueSpb} kartu** (${realItems.length} item barang)`,
        `• Belum Diproses: **${statusCounts["Belum Diproses"] || 0} item**`,
        `• Belum PO: **${statusCounts["Belum PO"] || 0} item**`,
        `• Sudah PO: **${statusCounts["Sudah PO"] || 0} item**`,
        `• Datang (Sesuai/Kurang/Lebih): **${(statusCounts["Datang Sesuai"] || 0) + (statusCounts["Datang Kurang"] || 0) + (statusCounts["Datang Lebih"] || 0)} item**`,
        `• Total Data PO: **${poRows.length} baris** | Surat Jalan: **${sjRows.length} baris**`,
      ].join("\n"),
      suggestions: ["Ringkasan stok gudang", "Barang stok menipis"],
      actionLinks: [{ label: "Buka Monitoring SPB", path: "/monitoring-spb" }],
    };
  }

  // 3. Pertanyaan seputar Stok Menipis / Habis
  if (
    query.includes("menipis") ||
    query.includes("habis") ||
    query.includes("low stock") ||
    query.includes("kurang")
  ) {
    const products = await fetchProductsSnapshot();
    const lowOrOut = products.filter(
      (p) => p.stock <= 0 || p.stock <= (p.minStock || 5) || p.status === "low" || p.status === "out",
    );

    if (lowOrOut.length === 0) {
      return {
        text: `Seluruh **${products.length} item barang** di gudang saat ini memiliki stok dalam batas aman.`,
        suggestions: DEFAULT_CHAT_SUGGESTIONS,
        actionLinks: [{ label: "Buka Inventory", path: "/inventory" }],
      };
    }

    const listText = lowOrOut
      .slice(0, 6)
      .map(
        (p) =>
          `• **${p.name}** (${p.code}) — Sisa stok: **${formatNumber(p.stock)} ${p.unit}**`,
      )
      .join("\n");

    return {
      text: [
        `Terdapat **${lowOrOut.length} barang** dengan stok menipis atau habis:`,
        listText,
      ].join("\n"),
      suggestions: ["Ringkasan stok gudang", "Status monitoring SPB"],
      actionLinks: [{ label: "Kelola Stok di Inventory", path: "/inventory" }],
    };
  }

  // 4. Pertanyaan seputar Ringkasan Stok / Gudang / Inventory
  if (
    query.includes("ringkasan") ||
    query.includes("total") ||
    query.includes("dashboard") ||
    query.includes("inventory") ||
    query.includes("gudang")
  ) {
    const products = await fetchProductsSnapshot();
    const totalItems = products.length;
    const totalUnits = products.reduce((acc, p) => acc + (Number(p.stock) || 0), 0);
    const materialCount = products.filter((p) => p.category === "Material").length;
    const toolsCount = products.filter((p) => p.category === "Alat Bantu").length;

    return {
      text: [
        `Ringkasan **Inventaris Gudang** saat ini:`,
        `• Total Jenis Barang: **${formatNumber(totalItems)} item**`,
        `• Total Akumulasi Stok: **${formatNumber(totalUnits)} unit**`,
        `• Kategori Material: **${formatNumber(materialCount)} item**`,
        `• Kategori Alat Bantu: **${formatNumber(toolsCount)} item**`,
      ].join("\n"),
      suggestions: ["Barang stok menipis", "Status monitoring SPB", "Info hak akses role"],
      actionLinks: [
        { label: "Buka Dashboard", path: "/dashboard" },
        { label: "Buka Inventory", path: "/inventory" },
      ],
    };
  }

  // 5. Pencarian spesifik nama/kode barang (misal: "stok kabel", "cari semen", atau nama barang langsung)
  const products = await fetchProductsSnapshot();
  const cleanKeyword = query
    .replace(/^(cek|cari|berapa|info|lihat|stok|barang)\s+/gi, "")
    .replace(/^stok\s+/gi, "")
    .trim();

  if (cleanKeyword) {
    const matchedProducts = products.filter(
      (p) =>
        p.name.toLowerCase().includes(cleanKeyword) ||
        p.code.toLowerCase().includes(cleanKeyword) ||
        (p.size && p.size.toLowerCase().includes(cleanKeyword)) ||
        (p.category && p.category.toLowerCase().includes(cleanKeyword)),
    );

    if (matchedProducts.length > 0) {
      const topMatches = matchedProducts
        .slice(0, 6)
        .map(
          (p) =>
            `• **${p.name}** ${p.size ? `(${p.size})` : ""} [${p.code}] — Stok: **${formatNumber(p.stock)} ${p.unit}** (${p.category})`,
        )
        .join("\n");

      return {
        text: [
          `Ditemukan **${matchedProducts.length} barang** yang cocok dengan pencarian **"${cleanKeyword}"**:`,
          topMatches,
        ].join("\n"),
        suggestions: DEFAULT_CHAT_SUGGESTIONS,
        actionLinks: [{ label: "Lihat di Inventory", path: "/inventory" }],
      };
    }
  }

  // 6. Fallback umum yang ramah dan mudah dipahami
  return {
    text: [
      `Saya adalah **Asisten Chatbot Gudang**. Berikut hal yang dapat saya bantu:`,
      `• **Cek Stok Barang**: ketik nama atau kode barang (contoh: *"cek stok kabel"*)`,
      `• **Cek Stok Menipis/Habis**: ketik *"barang stok menipis"*`,
      `• **Cek Monitoring SPB & PO**: ketik *"status monitoring SPB"* atau nomor SPB (contoh: *"SPB 10"*)`,
      `• **Info Hak Akses & Role**: ketik *"info hak akses role"*`,
    ].join("\n"),
    suggestions: DEFAULT_CHAT_SUGGESTIONS,
    actionLinks: [
      { label: "Dashboard", path: "/dashboard" },
      { label: "Inventory", path: "/inventory" },
      { label: "Monitoring SPB", path: "/monitoring-spb" },
    ],
  };
};

import { useState, useEffect, useRef } from "react";
import { useNavigate } from "react-router-dom";
import { Button, Input, Space, Tag, Typography } from "antd";
import {
  CloseOutlined,
  RobotOutlined,
  SendOutlined,
  ReloadOutlined,
  ArrowRightOutlined,
} from "@ant-design/icons";
import type { ChatMessage, UserRole } from "../../shared/types";
import {
  AUTH_CHANGED_EVENT,
  getStoredUser,
} from "../../app/providers/authProvider";
import { getRoleDefinition } from "../../app/providers/accessControlProvider";
import {
  DEFAULT_CHAT_SUGGESTIONS,
  generateChatbotResponse,
} from "./ChatbotEngine"; // 🔧 PERBAIKAN: Huruf kapital C dan E

const { Title, Text } = Typography;

export interface ChatbotWidgetProps {
  defaultOpen?: boolean;
  role?: UserRole;
}

const createWelcomeMessage = (roleLabel: string): ChatMessage => ({
  id: "welcome-msg",
  sender: "bot",
  text: `Halo! Saya Asisten Chatbot Gudang (Mode akses Anda: ${roleLabel}). Silakan tanyakan stok barang, stok menipis, status Monitoring SPB/PO, atau info hak akses.`,
  timestamp: new Date().toLocaleTimeString("id-ID", {
    hour: "2-digit",
    minute: "2-digit",
  }),
  suggestions: DEFAULT_CHAT_SUGGESTIONS,
});

export const ChatbotWidget = ({
  defaultOpen = false,
  role,
}: ChatbotWidgetProps) => {
  const navigate = useNavigate();
  const [open, setOpen] = useState<boolean>(defaultOpen);
  const [inputValue, setInputValue] = useState<string>("");
  const [isSending, setIsSending] = useState<boolean>(false);
  const [storedUser, setStoredUserState] = useState(() => getStoredUser());
  const messagesEndRef = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    const syncAuth = () => setStoredUserState(getStoredUser());
    window.addEventListener(AUTH_CHANGED_EVENT, syncAuth);
    window.addEventListener("storage", syncAuth);
    return () => {
      window.removeEventListener(AUTH_CHANGED_EVENT, syncAuth);
      window.removeEventListener("storage", syncAuth);
    };
  }, []);

  const effectiveRole: UserRole = role ?? storedUser?.role ?? "user";
  const roleDef = getRoleDefinition(effectiveRole);

  const [messages, setMessages] = useState<ChatMessage[]>(() => [
    createWelcomeMessage(roleDef.label),
  ]);

  useEffect(() => {
    if (open && typeof messagesEndRef.current?.scrollIntoView === "function") {
      messagesEndRef.current.scrollIntoView({ behavior: "smooth" });
    }
  }, [messages, open]);

  const handleSendMessage = async (customText?: string) => {
    const textToSend = (customText ?? inputValue).trim();
    if (!textToSend || isSending) return;

    const nowTime = new Date().toLocaleTimeString("id-ID", {
      hour: "2-digit",
      minute: "2-digit",
    });

    const userMsg: ChatMessage = {
      id: `user-${Date.now()}`,
      sender: "user",
      text: textToSend,
      timestamp: nowTime,
    };

    setMessages((prev) => [...prev, userMsg]);
    if (!customText) {
      setInputValue("");
    }
    setIsSending(true);

    try {
      const reply = await generateChatbotResponse(textToSend, effectiveRole);
      const botMsg: ChatMessage = {
        id: `bot-${Date.now()}-${Math.random().toString(36).slice(2, 5)}`,
        sender: "bot",
        text: reply.text,
        timestamp: new Date().toLocaleTimeString("id-ID", {
          hour: "2-digit",
          minute: "2-digit",
        }),
        suggestions: reply.suggestions,
        actionLinks: reply.actionLinks,
      };
      setMessages((prev) => [...prev, botMsg]);
    } finally {
      setIsSending(false);
    }
  };

  const handleResetChat = () => {
    setMessages([createWelcomeMessage(roleDef.label)]);
  };

  if (!open) {
    return (
      <div
        style={{
          position: "fixed",
          right: 24,
          bottom: 24,
          zIndex: 1500,
        }}
      >
        <Button

          size="large"
          shape="round"

          data-testid="chatbot-open-button"
          style={{
            height: 48,
            paddingInline: 20,
            fontWeight: 600,
            boxShadow: "0 8px 24px rgba(22, 119, 255, 0.35)",
          }}
        >
          Chatbot Gudang

  }

  return (

      aria-label="Jendela Chatbot Gudang"
      data-testid="chatbot-fullscreen-window"

        position: "fixed",
        top: 0,
        left: 0,
        width: "100vw",
        height: "100vh",
        zIndex: 2000,
        background: "#f5f7fa",
        display: "flex",
        flexDirection: "column",
      }}
    >
      {/* Header Layar Penuh dengan Tombol Silang (X) di Sebelah Kiri Atas */}
      <div
        style={{
          height: 64,
          padding: "0 20px",
          background: "#ffffff",
          borderBottom: "1px solid #e6ebf1",
          display: "flex",
          alignItems: "center",
          justifyContent: "space-between",
          flexShrink: 0,

            icon={<CloseOutlined style={{ fontSize: 18 }} />}

              display: "flex",
              alignItems: "center",
              justifyContent: "center",
              width: 40,
              height: 40,
              borderRadius: 8,
              border: "1px solid #d9d9d9",
            }}
          />

          <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
            <div
              style={{
                width: 36,
                height: 36,
                borderRadius: 10,
                background: "linear-gradient(135deg, #1677ff, #0e4fb8)",
                color: "#fff",
                display: "flex",
                alignItems: "center",
                justifyContent: "center",
                fontSize: 18,

                  display: "flex",
                  flexDirection: "column",
                  alignItems: isUser ? "flex-end" : "flex-start",
                }}

                <div
                  style={{
                    maxWidth: "85%",

                    fontSize: 14,
                    lineHeight: 1.6,

                >
                  {msg.text}

                <Text
                  type="secondary"
                  style={{ fontSize: 11, marginTop: 4, paddingInline: 4 }}
                >
                  {isUser ? "Anda" : "Bot"} · {msg.timestamp}
                </Text>

                {/* Tombol Navigasi Cepat jika tersedia */}
                {!isUser && msg.actionLinks && msg.actionLinks.length > 0 && (

                        }}
                      >
                        {link.label}
                      </Button>
                    ))}
                  </Space>
                )}

                {/* Saran Pertanyaan Cepat */}
                {!isUser && msg.suggestions && msg.suggestions.length > 0 && (
                  <Space wrap size={6} style={{ marginTop: 8 }}>
                    {msg.suggestions.map((sug) => (
                      <Button
                        key={sug}
                        size="small"
                        onClick={() => handleSendMessage(sug)}
                        style={{ borderRadius: 16, fontSize: 12 }}
                      >
                        {sug}
                      </Button>
                    ))}
                  </Space>
                )}
              </div>
            );
          })}
          <div ref={messagesEndRef} />
        </div>
      </div>

      {/* Input Chat di Bagian Bawah */}
      <div
        style={{
          padding: "16px 20px",
          background: "#ffffff",
          borderTop: "1px solid #e6ebf1",
          flexShrink: 0,
        }}
      >
        <div
          style={{
            maxWidth: 880,
            margin: "0 auto",
            display: "flex",
            gap: 10,
          }}
        >
          <Input
            size="large"
            placeholder="Ketik pertanyaan (contoh: Cek stok kabel, Status monitoring SPB, Barang stok menipis)..."
            value={inputValue}
            onChange={(e) => setInputValue(e.target.value)}
            onPressEnter={() => handleSendMessage()}
            aria-label="Input Pesan Chatbot"
            autoFocus
          />
          <Button
            type="primary"
            size="large"
            icon={<SendOutlined />}
            loading={isSending}
            onClick={() => handleSendMessage()}
            aria-label="Kirim Pesan Chatbot"
          >
            Kirim
          </Button>
        </div>
      </div>
    </div>
  );
};

export default ChatbotWidget;

// ✅ Path import disesuaikan dengan arsitektur baru

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { getMockDashboardStats } from "../../mocks/dashboard";
import type { DashboardStats } from "../../shared/types";

export interface UseDashboardStatsResult {
  stats: DashboardStats;
  isLoading: boolean;
  refetch: () => void;
}

const EMPTY_STATS: DashboardStats = {
  totalProducts: 0,
  totalStock: 0,
  todayInbound: 0,
  todayOutbound: 0,
};

export const useDashboardStats = (): UseDashboardStatsResult => {
  const query = useQuery<DashboardStats, Error, DashboardStats>({
    queryKey: ["dashboard-stats"],
    refetchInterval: USE_MOCK ? false : 60_000,
    queryFn: async (): Promise<DashboardStats> => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return getMockDashboardStats();
      }

      const today = dayjs().format("YYYY-MM-DD");
      const startOfDay = dayjs().startOf("day").toISOString();
      const endOfDay = dayjs().endOf("day").toISOString();

      const [productsRes, inboundRes, outboundRes] = await Promise.all([
        supabaseClient.from(TABLES.products).select("stok"),
        supabaseClient
          .from(TABLES.stockEntries)
          .select("jumlah_barang_masuk")
          .eq("tanggal_pengiriman", today),
        supabaseClient
          .from(TABLES.stockExits)
          .select("jumlah_barang_keluar")
          .gte("created_at", startOfDay)
          .lte("created_at", endOfDay),
      ]);

      if (productsRes.error) throw productsRes.error;
      if (inboundRes.error) throw inboundRes.error;
      if (outboundRes.error) throw outboundRes.error;

      const totalProducts = productsRes.data?.length ?? 0;
      const totalStock =
        productsRes.data?.reduce(
          (sum, p: { stok: number | null }) => sum + (p.stok ?? 0),
          0,
        ) ?? 0;

      const todayInbound =
        inboundRes.data?.reduce(
          (sum, e: { jumlah_barang_masuk: number | null }) =>
            sum + (e.jumlah_barang_masuk ?? 0),
          0,
        ) ?? 0;

      const todayOutbound =
        outboundRes.data?.reduce(
          (sum, exit: { jumlah_barang_keluar: number | null }) =>
            sum + (exit.jumlah_barang_keluar ?? 0),
          0,
        ) ?? 0;

      return { totalProducts, totalStock, todayInbound, todayOutbound };
    },
  });

  return {
    stats: query.data ?? EMPTY_STATS,
    isLoading: query.isLoading,
    refetch: query.refetch,
  };
};

import { useQuery } from "@tanstack/react-query";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockLowStockProducts } from "../../mocks/products";
import { mapProduct } from "../../shared/lib/mappers";
import type { Product } from "../../shared/types";

const LOW_STOCK_THRESHOLD = 5;

export const useLowStock = (limit = 5) => {
  const { data, isLoading, refetch } = useQuery<Product[]>({
    queryKey: ["low-stock", limit],
    queryFn: async () => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return mockLowStockProducts.slice(0, limit);
      }

      const { data, error } = await supabaseClient
        .from(TABLES.products)
        .select("*")
        .lte("stok", LOW_STOCK_THRESHOLD)
        .order("stok", { ascending: true })
        .limit(limit);

      if (error) throw error;

      return (data ?? []).map((row) => {
        const product = mapProduct(row as Record<string, unknown>);
        return {
          ...product,
          minStock: LOW_STOCK_THRESHOLD,
          status: product.stock <= 0 ? "out" : "low",
        };
      });
    },
  });

  return {
    items: data ?? [],
    total: data?.length ?? 0,
    isLoading,
    refetch,
  };
};

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockStockInChart } from "../../mocks/stockEntries";
import type { StockInChartItem } from "../../shared/types";

export const useStockInChart = (days = 30) => {
  const { data, isLoading, refetch } = useQuery<StockInChartItem[]>({
    queryKey: ["stock-in-chart", days],
    queryFn: async () => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return mockStockInChart.slice(-days);
      }

      const fromDate = dayjs().subtract(days, "day").format("YYYY-MM-DD");

      const { data, error } = await supabaseClient
        .from(TABLES.stockEntries)
        .select("tanggal_pengiriman, jumlah_barang_masuk")
        .gte("tanggal_pengiriman", fromDate)
        .order("tanggal_pengiriman", { ascending: true });

      if (error) throw error;

      const grouped = (data ?? []).reduce<Record<string, number>>(
        (
          acc,
          row: {
            tanggal_pengiriman: string | null;
            jumlah_barang_masuk: number | null;
          },
        ) => {
          if (!row.tanggal_pengiriman) return acc;
          acc[row.tanggal_pengiriman] =
            (acc[row.tanggal_pengiriman] ?? 0) + (row.jumlah_barang_masuk ?? 0);
          return acc;
        },
        {},
      );

      return Object.entries(grouped).map(([date, total]) => ({
        date,
        total,
      }));
    },
  });

  return {
    items: data ?? [],
    isLoading,
    refetch,
  };
};

export const createBarang = async (input: BarangInput): Promise<Product> => {
  // 🔒 Verifikasi: user harus login dan punya izin edit_data
  requirePermission("edit_data", "Tambah barang baru");

  const rowPayload = {
    kode:
      input.kode?.trim() ||

    stok: Number(input.stok ?? 0),
  };

  const { data, error } = await supabaseClient
    .from(TABLES.products)
    .insert([rowPayload])
    .select()
    .single();

  if (error) throw error;

  return mapProduct(
    data ?? {
      id: Date.now(),
      ...rowPayload,
      created_at: new Date().toISOString(),
    },
  );
};

export const updateBarang = async (
  id: number,
  input: Partial<BarangInput>,
): Promise<Product> => {

  const patch: Record<string, unknown> = {};
  if (input.kode !== undefined) patch.kode = input.kode.trim();
  if (input.nama !== undefined) patch.nama = input.nama.trim();
  if (input.ukuran !== undefined) patch.ukuran = input.ukuran?.trim() || null;
  if (input.spesifikasi !== undefined)
    patch.spesifikasi = input.spesifikasi?.trim() || null;
  if (input.kategori !== undefined) patch.kategori = input.kategori;
  if (input.satuan !== undefined) patch.satuan = input.satuan.trim();
  if (input.stok !== undefined) patch.stok = Number(input.stok);

  const { data, error } = await supabaseClient
    .from(TABLES.products)
    .update(patch)
    .eq("id", id)
    .select()
    .single();

  if (error) throw error;

  return mapProduct(
    data ?? {
      id,
      kode: String(patch.kode ?? `BRG-${id}`),
      nama: String(patch.nama ?? ""),
      ukuran: (patch.ukuran as string | null) ?? null,
      spesifikasi: (patch.spesifikasi as string | null) ?? null,
      kategori: String(patch.kategori ?? "Material"),
      satuan: String(patch.satuan ?? "pcs"),
      stok: Number(patch.stok ?? 0),
      created_at: new Date().toISOString(),
    },
  );
};

export const deleteBarang = async (id: number): Promise<void> => {
  // 🔒 Verifikasi: user harus login dan punya izin edit_data
  requirePermission("edit_data", "Hapus barang");

  const { error } = await supabaseClient
    .from(TABLES.products)
    .delete()
    .eq("id", id);

  if (error) throw error;
};

        await new Promise((r) => setTimeout(r, 300));

          : mockProducts;

        .from(TABLES.products)

        // ✅ Urutkan berdasarkan stok terbanyak terlebih dahulu

        query = query.eq("kategori", category);
      }

      const { data, error } = await query;
      if (error) throw error;
      return (data ?? []).map(mapProduct);
    },
  });

  return {
    products: data ?? [],
    isLoading,
    refetch,
  };
};

import { useQuery } from "@tanstack/react-query";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockProducts } from "../../mocks/products";
import { mapProduct } from "../../shared/lib/mappers";
import type { Product } from "../../shared/types";

export const useProductSearch = (keyword: string, limit = 30) => {
  const sanitize = (str: string) => str.replace(/[%_()\\'"]/g, "");
  const trimmed = sanitize(keyword).trim();

  const { data, isLoading, refetch } = useQuery<Product[]>({
    queryKey: ["product-search", trimmed, limit],
    queryFn: async () => {
      if (USE_MOCK) {
        if (!trimmed) return mockProducts.slice(0, limit);
        const lower = trimmed.toLowerCase();
        return mockProducts.filter((p) => p.name.toLowerCase().includes(lower) || p.code.toLowerCase().includes(lower)).slice(0, limit);
      }

      let query = supabaseClient.from(TABLES.products).select("*").limit(limit).order("nama", { ascending: true }); 
      if (trimmed) {
        const words = trimmed.split(/\s+/).filter(Boolean);
        const orConditions = words.map((w) => `nama.ilike.%${w}%,kode.ilike.%${w}%`).join(",");
        query = query.or(orConditions);
      }

      const { data, error } = await query;
      if (error) throw error;
      return (data ?? []).map(mapProduct);
    },
    staleTime: 10_000,
  });

  return { products: data ?? [], isLoading, refetch };
};

/**
 * Aturan akses data SPB (sisi klien).

 * CATATAN KEAMANAN: aturan ini hanya lapisan UX/defense-in-depth. Penegakan
 * sebenarnya ada di database (RLS pada tabel `spb`, lihat
 * supabase/migrations/20261103_enforce_auth_rls.sql). Jangan mengandalkan

 */
import { getVerifiedUser } from "../../shared/lib/authGuard";

/** Nilai kolom `status` tabel spb yang hanya boleh dilihat user yang login. */
export const RESTRICTED_SPB_STATUS = "belum di proses";

/** Label filter di UI yang hanya tersedia untuk user yang login. */
export const RESTRICTED_FILTER_LABEL = "Belum Diproses";

export const isRestrictedSpbStatus = (status: unknown): boolean =>
  String(status ?? "").trim().toLowerCase() === RESTRICTED_SPB_STATUS;

/** True jika user saat ini sudah login (role terautentikasi). */
export const canViewRestrictedSpb = (): boolean => getVerifiedUser() !== null;

/**
 * Menyaring baris SPB mentah: user yang belum login tidak menerima baris
 * berstatus "belum di proses".
 */
export const filterVisibleSpbRows = <T extends { status?: unknown }>(
  rows: T[],
  canViewRestricted: boolean,
): T[] =>
  canViewRestricted
    ? rows
    : rows.filter((row) => !isRestrictedSpbStatus(row.status));

/**
 * Validasi URL file sumber (mis. Google Drive) sebelum dijadikan link.
 * Hanya `https:` yang diterima — mencegah `javascript:`/`data:` URL dari data DB.
 * @returns URL ternormalisasi, atau null jika kosong/tidak aman.
 */
export const toSafeExternalUrl = (value: unknown): string | null => {
  if (typeof value !== "string") return null;
  const raw = value.trim();
  if (!raw) return null;
  try {
    const url = new URL(raw);
    return url.protocol === "https:" ? url.toString() : null;
  } catch {
    return null;
  }
};

/**
 * Mengambil seluruh baris dari sebuah tabel dengan pagination `.range()`
 * untuk menghindari batas default 1000 baris PostgREST.
 */
export const fetchAllRows = async (
  table: string,
  select: string,
): Promise<any[]> => {
  const all: any[] = [];
  let from = 0;

  for (let guard = 0; guard < 100; guard += 1) {
    const { data, error } = await supabaseClient
      .from(table)
      .select(select)
      .range(from, from + PAGE_SIZE - 1);
    if (error) throw error;

    const rows = (data as any[]) ?? [];
    all.push(...rows);
    if (rows.length < PAGE_SIZE) break;

    from += PAGE_SIZE;
  }

  return all;
};

/**
 * Mengambil daftar barang detail berdasarkan nomor SPB.
 */
export const fetchDetailBySpb = async (
  spbNo: string,
  judulSpb: string | null = null,
  canViewRestricted = false,
): Promise<MonitoringDetailItem[]> => {
  let spbQuery = supabaseClient
    .from("spb")
    .select(SPB_SELECT)
    .eq("no_spb", spbNo);
  if (judulSpb) spbQuery = spbQuery.eq("judul_spb", judulSpb);

  const spbRows = filterVisibleSpbRows(
    sortById(unwrap(await spbQuery)),
    canViewRestricted,
  );
  if (spbRows.length === 0) return [];

  const spbIds = spbRows.map((r) => r.id);

  const [poRes, sjRes] = await Promise.all([
    supabaseClient
      .from("po")
      .select("id, no_po, tanggal_po, id_spb, source_file")
      .in("id_spb", spbIds),
    supabaseClient.from("surat_jalan").select(SJ_SELECT).in("id_spb", spbIds),
  ]);

  const poBySpbId = new Map<string, Row>();
  unwrap(poRes).forEach((p) => {
    const key = String(p.id_spb);
    if (cleanDocNo(p.no_po) && !poBySpbId.has(key)) poBySpbId.set(key, p);
  });

  const spbItems = spbRows.map((r) =>
    mapSpbRow(r, poBySpbId.get(String(r.id)) ?? null),
  );
  const sjItems = sortById(unwrap(sjRes))
    .filter((sj) => !isIgnoredSuratJalan(sj.no_surat_jalan))
    .map(mapSjRow);

  return [...spbItems, ...sjItems];
};

/**
 * Mengambil daftar barang detail berdasarkan nomor PO.
 */
export const fetchDetailByPo = async (
  poNo: string,
): Promise<MonitoringDetailItem[]> => {
  const poRows = sortById(
    unwrap(await supabaseClient.from("po").select("*").eq("no_po", poNo)),
  );
  if (poRows.length === 0) return [];

  const poIds = poRows.map((p) => p.id);
  const parentSpbIds = unique(poRows.map((p) => p.id_spb).filter(hasValue));

  const [parentSpbRes, spbPointingRes, sjRes] = await Promise.all([
    parentSpbIds.length > 0
      ? supabaseClient.from("spb").select("id").in("id", parentSpbIds)
      : Promise.resolve({ data: [], error: null }),
    supabaseClient.from("spb").select("id, id_po").in("id_po", poIds),
    supabaseClient
      .from("surat_jalan")
      .select(SJ_SELECT)
      .in("id_po", poIds)
      .is("id_spb", null),
  ]);

  const existingParentSpb = new Set(
    unwrap(parentSpbRes).map((s) => String(s.id)),
  );
  const poIdsUsedBySpb = new Set(
    unwrap(spbPointingRes).map((s) => String(s.id_po)),
  );

  const isLinkedToSpb = (p: Row) =>
    (hasValue(p.id_spb) && existingParentSpb.has(String(p.id_spb))) ||
    poIdsUsedBySpb.has(String(p.id));

  const poItems = poRows.filter((p) => !isLinkedToSpb(p)).map(mapPoRow);
  const sjItems = sortById(unwrap(sjRes))
    .filter((sj) => !isIgnoredSuratJalan(sj.no_surat_jalan))
    .map(mapSjRow);

  return [...poItems, ...sjItems];
};

};

/**
 * Mengambil rincian satu barang berdasarkan ID dan tipe sumber ("spb" | "po" | "sj").
 */
export const fetchSingleItemById = async (
  id: string,
  type: string,
): Promise<SPBItemDetail | null> => {
  if (type === "po") {
    const { data, error } = await supabaseClient
      .from("po")
      .select("*")
      .eq("id", id)
      .single();
    if (error) throw error;
    if (!data) return null;
    return mapSinglePoItem(data);
  }

  if (type === "sj") {
    const { data, error } = await supabaseClient
      .from("surat_jalan")
      .select(
        `*, spb:spb!id_spb (no_spb, tgl_spb, source_file), po:po!id_po (no_po, tanggal_po, nama_supplier, source_file)`,
      )
      .eq("id", id)
      .single();
    if (error) throw error;
    if (!data) return null;
    return mapSingleSjItem(data);
  }

  const { data, error } = await supabaseClient
    .from(TABLES.spb)
    .select(`*, po (no_po, tanggal_po, nama_supplier, source_file)`)
    .eq("id", id)
    .single();

  if (error) throw error;
  if (!data) return null;

  if (isRestrictedSpbStatus(data.status) && !canViewRestrictedSpb()) {
    return null;
  }

  let dataPo = Array.isArray(data.po) ? data.po[0] : data.po;

  if (!dataPo && data.id) {
    try {
      const fallbackRes = await supabaseClient
        .from("po")
        .select("no_po, tanggal_po, nama_supplier, source_file")
        .eq("id_spb", data.id)
        .single();
      if (fallbackRes?.data) {
        dataPo = fallbackRes.data;
      }
    } catch {
      // ignore fallback error
    }
  }

  return mapSingleSpbItem(data, dataPo);
};

// src/features/spb/hooks/useEditDocumentTitle.ts
import { useState, type MouseEvent } from "react";
import { message } from "antd";
import type { EditingTitleState } from "../types";
import { updateDocumentTitle } from "../useMonitoringSPB";
import { buildLocalTitleKey } from "../utils/titleOverrides";

export const useEditDocumentTitle = (
  allowEdit: boolean,
  onAfterSave?: () => void,
) => {
  const [editingTitle, setEditingTitle] = useState<EditingTitleState | null>(
    null,
  );
  const [titleInput, setTitleInput] = useState("");
  const [savingTitle, setSavingTitle] = useState(false);
  const [localTitles, setLocalTitles] = useState<Record<string, string>>({});

  const openEditModal = (
    e: MouseEvent,
    type: "spb" | "po",
    docNo: string,
    currentTitle?: string | null,
  ) => {
    if (!allowEdit) return;
    e.stopPropagation();
    const initial = currentTitle || "";
    setEditingTitle({ type, docNo, currentTitle: initial });
    setTitleInput(initial);
  };

  const closeEditModal = () => setEditingTitle(null);

  const handleSaveTitle = async () => {
    if (!editingTitle) return;
    setSavingTitle(true);
    const { type, docNo, currentTitle } = editingTitle;
    const trimmed = titleInput.trim();
    const localKey = buildLocalTitleKey(type, docNo, currentTitle);

    try {
      await updateDocumentTitle(type, docNo, trimmed, currentTitle);
      setLocalTitles((prev) => ({ ...prev, [localKey]: trimmed }));
      setEditingTitle(null);
      message.success(
        `Judul ${type.toUpperCase()} ${docNo} berhasil diperbarui`,
      );
      onAfterSave?.();
    } catch {
      setLocalTitles((prev) => ({ ...prev, [localKey]: trimmed }));
      setEditingTitle(null);
    } finally {
      setSavingTitle(false);
    }
  };

  return {
    editingTitle,
    titleInput,
    setTitleInput,
    savingTitle,
    localTitles,
    openEditModal,
    closeEditModal,
    handleSaveTitle,
  };
};

import { useEffect, useState } from "react";
import { RESTRICTED_FILTER_LABEL } from "../accessRules";
import type { FilterType, SearchMode, SortBy } from "../types";

export const useMonitoringSPBFilter = (canViewRestricted: boolean) => {
  const [storedFilter, setFilter] = useState<FilterType>(() => {
    return (localStorage.getItem("spb_filter") as FilterType) || "Semua";
  });

  // Filter tersimpan "Belum Diproses" diabaikan jika user belum login
  const filter: FilterType =
    !canViewRestricted && storedFilter === RESTRICTED_FILTER_LABEL
      ? "Semua"
      : storedFilter;

  const [searchTerm, setSearchTerm] = useState(() => {
    return localStorage.getItem("spb_searchTerm") || "";
  });

  const [searchMode, setSearchMode] = useState<SearchMode>(() => {
    return (localStorage.getItem("spb_searchMode") as SearchMode) || "no";
  });

  const [sortBy, setSortBy] = useState<SortBy>(() => {
    return (localStorage.getItem("spb_sortBy") as SortBy) || "status";
  });

  useEffect(() => {
    localStorage.setItem("spb_filter", filter);
    localStorage.setItem("spb_searchTerm", searchTerm);
    localStorage.setItem("spb_searchMode", searchMode);
    localStorage.setItem("spb_sortBy", sortBy);
  }, [filter, searchTerm, searchMode, sortBy]);

  return {
    filter,
    setFilter,
    searchTerm,
    setSearchTerm,
    searchMode,
    setSearchMode,
    sortBy,
    setSortBy,
  };
};

// src/features/spb/suratJalanRules.ts
//
// Aturan bisnis Monitoring SPB yang dipakai BERSAMA oleh:
//   - useMonitoringSPB        (daftar card)
//   - useMonitoringSPBDetail  (isi tabel saat card dibuka)
// Dengan satu sumber aturan, jumlah barang di card selalu sama dengan jumlah
// baris di halaman detail.

/** Surat Jalan yang diabaikan sepenuhnya (data lama). Yang diproses: no 3 dan seterusnya. */
export const IGNORED_SURAT_JALAN: readonly string[] = ["1", "2"];

export const isIgnoredSuratJalan = (no: unknown): boolean =>
  IGNORED_SURAT_JALAN.includes(String(no ?? "").trim());

/** true jika nilai kolom (mis. no_spb / no_po / judul) benar-benar terisi. */
export const hasValue = (val: unknown): boolean =>
  val !== null && val !== undefined && String(val).trim() !== "";

/**
 * Normalisasi id FK (id_spb / id_po / id) menjadi number.
 *
 * Data id di DB sekarang bertipe number (bigint / integer). Namun untuk
 * keamanan terhadap sisa data lama yang mungkin masih string, helper ini
 * menyamakan keduanya menjadi number agar semua perbandingan konsisten.
 *
 * Nilai yang tidak valid, 0, atau negatif dianggap "tidak ada id".
 * (id 0 tidak dipakai di database, sehingga aman dijadikan sentinel null.)
 */
export const toIdNumber = (val: unknown): number | null => {
  if (val === null || val === undefined) return null;
  if (typeof val === "string" && val.trim() === "") return null;
  const n = Number(val);
  if (!Number.isFinite(n) || n <= 0) return null;
  return n;
};

/**
 * Normalisasi nomor dokumen (no_spb / no_po) dari DB.
 * String kosong, "-", "null", "undefined" dianggap TIDAK ADA nomor -> null.
 */
export const cleanDocNo = (val: unknown): string | null => {
  if (val === null || val === undefined) return null;
  const s = String(val).trim();
  const lower = s.toLowerCase();
  if (s === "" || s === "-" || lower === "null" || lower === "undefined") {
    return null;
  }
  return s;
};

export type SjTarget = "spb" | "po" | "sj";

/**
 * Menentukan di card mana sebuah baris Surat Jalan ditampilkan:
 *  - punya id_spb                -> card SPB
 *  - tanpa id_spb, punya id_po   -> card PO
 *  - tanpa id_spb dan id_po      -> card Surat Jalan
 *
 * Menggunakan `toIdNumber` agar "0", "0.0", " ", null semua dianggap kosong.
 */
export const getSjTarget = (sj: {
  id_spb?: unknown;
  id_po?: unknown;
}): SjTarget => {
  if (toIdNumber(sj.id_spb) !== null) return "spb";
  if (toIdNumber(sj.id_po) !== null) return "po";
  return "sj";
};

// ============================================================================
// FUNGSI TAMBAHAN UNTUK CHATBOT & UI MAPPING
// ============================================================================

/**
 * Membersihkan judul SPB dari teks "null", "undefined", atau tanda strip.
 */
export const cleanSpbTitle = (title: unknown): string => {
  if (!title) return "";
  const s = String(title).trim();
  const lower = s.toLowerCase();
  if (lower === "null" || lower === "undefined" || s === "-") return "";
  return s;
};

/**
 * Memeriksa apakah sebuah baris data SPB hanyalah catatan/keterangan kosong
 * (tidak dihitung sebagai barang riil).
 */
export const isKeteranganBarang = (row: any): boolean => {
  if (!row) return true;
  const name = String(row.nama_barang_spb || row.nama_barang || "").toLowerCase();
  if (!name.trim()) return true;
  return name.includes("keterangan") || name.includes("catatan");
};

/**
 * Menentukan status dari satu baris item SPB (untuk konsumsi Chatbot).
 * Menggunakan `toIdNumber` agar deteksi "sudah PO" tidak lagi bergantung pada
 * tipe string/number dari kolom id_po.
 */
export const resolveSpbItemStatus = (row: any): string => {
  if (!row) return "Belum PO";

  const tgl = row.material_datang_tgl;
  const req = Number(row.qty_permintaan || 0);
  const arr = Number(row.material_datang_qty || 0);
  const hasPo = !!cleanDocNo(row.no_po) || toIdNumber(row.id_po) !== null;

  if (tgl) {
    if (req === arr) return "Datang Sesuai";
    if (req < arr) return "Datang Lebih";
    return "Datang Kurang";
  }
  if (hasPo) return "Sudah PO";

  const raw = String(row.status || "").toLowerCase().trim();
  if (raw === "belum di proses") return "Belum Diproses";

  return "Belum PO";
};

// src/features/spb/useMonitoringSPB.ts
import { useQuery } from "@tanstack/react-query";
import { USE_MOCK } from "../../shared/lib/config";
import type { SPBDocumentGroup } from "../../shared/types";
import { canViewRestrictedSpb } from "./accessRules";
import { fetchMonitoringRawData, updateDocumentTitle } from "./api";
import { groupMonitoringData } from "./utils/groupMonitoringData";
import { getTitleOverrides, saveTitleOverride } from "./utils/titleOverrides";

export { getTitleOverrides, saveTitleOverride, updateDocumentTitle };

export const useMonitoringSPB = () => {
  const canViewRestricted = canViewRestrictedSpb();

  const { data, isLoading, error, refetch } = useQuery<SPBDocumentGroup[]>({
    queryKey: ["monitoring-spb", canViewRestricted],
    queryFn: async () => {
      if (USE_MOCK) return [];

      const overrides = getTitleOverrides();
      const { allSpbData, poData, sjData } = await fetchMonitoringRawData();

      return groupMonitoringData({
        allSpbData,
        poData,
        sjData,
        canViewRestricted,
        overrides,
      });
    },
  });

  return {
    data: data ?? [],
    isLoading,
    error: error as Error | null,
    refetch,
  };
};

// src/features/spb/useMonitoringSPBDetail.ts
import { useQuery } from "@tanstack/react-query";
import { canViewRestrictedSpb } from "./accessRules";
import {
  fetchDetailByPo,
  fetchDetailBySpb,
  fetchDetailBySuratJalan,
} from "./api";
import type { ItemSource, MonitoringDetailItem } from "./types";

export type { ItemSource, MonitoringDetailItem };

export const useMonitoringSPBDetail = (
  spbNo: string | null,
  poNo: string | null,
  sjNo: string | null = null,
  judulSpb: string | null = null,
) => {
  const enabled = !!(spbNo || poNo || sjNo);
  const canViewRestricted = canViewRestrictedSpb();

  const { data, isLoading, error } = useQuery<MonitoringDetailItem[]>({
    queryKey: [
      "monitoring-spb-detail",
      spbNo,
      poNo,
      sjNo,
      judulSpb,
      canViewRestricted,
    ],
    queryFn: async () => {
      if (spbNo) return fetchDetailBySpb(spbNo, judulSpb, canViewRestricted);
      if (poNo) return fetchDetailByPo(poNo);
      if (sjNo) return fetchDetailBySuratJalan(sjNo);
      return [];
    },
    enabled,
    retry: false,
    refetchOnWindowFocus: false,
  });

  return {
    items: data ?? [],
    isLoading: enabled ? isLoading : false,
    error: error as Error | null,
  };
};

import { useQuery } from "@tanstack/react-query";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import type { SPBItemDetail } from "../../shared/types";
import { canViewRestrictedSpb, isRestrictedSpbStatus } from "./accessRules";

export const useSPBItemDetail = (itemId: string | null) => {
  const enabled = !!itemId;
  const { data, isLoading, error } = useQuery<SPBItemDetail | null>({
    queryKey: ["spb-item-detail", itemId],
    queryFn: async () => {
      if (!itemId) return null;

      const { data, error } = await supabaseClient
        .from(TABLES.spb)
        .select(`
          *,
          po (
            no_po,
            tanggal_po,
            nama_supplier,
            supplier
          )
        `)
        .eq("id", itemId)
        .single();

      if (error) throw error;
      if (!data) return null;

      // 🔒 SPB "belum di proses" hanya untuk user yang login
      if (isRestrictedSpbStatus(data.status) && !canViewRestrictedSpb()) return null;

      const dataPo = Array.isArray(data.po) ? data.po[0] : data.po;

      return {
        ...data,
        no_po: dataPo?.no_po || "-",
        tanggal_po: dataPo?.tanggal_po || null,
        supplier: dataPo?.nama_supplier || dataPo?.supplier || "-",
        nama_barang: data.nama_barang_spb || "-",
        tanggal_spb: data.tgl_spb || data.tanggal_pembuatan_spb || null,
        jumlah_permintaan: data.qty_permintaan || 0,
        satuan: data.satuan || "-",
        tanggal_datang: data.material_datang_tgl || null,
        ukuran: data.ukuran || "-",
        merk: data.merk || "-",
        spesifikasi_teknis: data.spesifikasi_teknis || "-",
        lokasi_pemasangan: data.lokasi_pemasangan || "-",
        rencana_pemasangan: data.rencana_pemasangan || "-",
        qty_datang: data.material_datang_qty ?? data.qty_datang ?? 0,
        satuan_datang: data.material_datang_satuan || data.satuan_datang || data.satuan || "-",
        lama_proses: data.lama_proses || "-",
        kurang_lebih: data.kurang_lebih || "-",
        keterangan: data.keterangan || "-",
        status: data.status || "-",
      } as SPBItemDetail;
    },
    enabled,
  });

  return {
    item: data ?? null,
    isLoading: enabled ? isLoading : false,
    error: error as Error | null,
  };
};

// src/features/spb/utils/filterAndSortSpb.ts
import type { SPBDocumentGroup } from "../../../shared/types";
import type { FilterType, SearchMode, SortBy } from "../types";
import { getNumberFromStr } from "./statusHelpers";
import { buildLocalTitleKey } from "./titleOverrides";

const getFilterStatusWeight = (status?: string): number => {
  if (status === "Belum Diproses") return 0.5;
  if (status === "Sudah PO" || status === "Belum PO") return 1;
  if (status === "Datang") return 2;
  if (status === "Selesai") return 3;
  return 4;
};

export interface FilterAndSortParams {
  data: SPBDocumentGroup[];
  filter: FilterType;
  searchTerm: string;
  searchMode: SearchMode;
  sortBy: SortBy;
  localTitles: Record<string, string>;
}

/**
 * Menyaring dan mengurutkan daftar dokumen SPB/PO/SJ pada halaman Monitoring SPB.
 */
export const filterAndSortSpbDocuments = ({
  data,
  filter,
  searchTerm,
  searchMode,
  sortBy,
  localTitles,
}: FilterAndSortParams): SPBDocumentGroup[] => {
  const searchLower = searchTerm.toLowerCase();

  const filtered = data.filter((item) => {
    let matchesSearch = false;

    const spbKey = item.spbNo
      ? buildLocalTitleKey("spb", item.spbNo, item.spbTitle)
      : null;
    const poKey = item.poNo
      ? buildLocalTitleKey("po", item.poNo, item.poTitle)
      : null;

    const spbTitle =
      (spbKey && localTitles[spbKey] !== undefined
        ? localTitles[spbKey]
        : item.spbTitle) || "";
    const poTitle =
      (poKey && localTitles[poKey] !== undefined
        ? localTitles[poKey]
        : item.poTitle) || "";

    if (searchMode === "no") {
      const matchSpb = String(item.spbNo || "")
        .toLowerCase()
        .includes(searchLower);
      const matchTitle = String(spbTitle).toLowerCase().includes(searchLower);
      const matchPo = String(item.poNo || "")
        .toLowerCase()
        .includes(searchLower);
      const matchPoTitle = String(poTitle).toLowerCase().includes(searchLower);
      const matchSj = String(item.suratJalanNo || "")
        .toLowerCase()
        .includes(searchLower);

      matchesSearch =
        !searchTerm ||
        matchSpb ||
        matchTitle ||
        matchPo ||
        matchPoTitle ||
        matchSj;
    } else {
      matchesSearch =
        !searchTerm ||
        !!item.itemNames?.some((name) =>
          String(name).toLowerCase().includes(searchLower),
        );
    }

    if (!matchesSearch) return false;

    switch (filter) {
      case "Belum Diproses":
        return item.status === "Belum Diproses";
      case "Belum di PO":
        return (!item.poNo || item.poNo === "-") && item.status === "Belum PO";
      case "Sudah di PO":
        return !!item.poNo && item.poNo !== "-" && item.status !== "Selesai";
      case "Datang":
        return item.status === "Datang" || item.status === "Selesai";
      case "Selesai":
        return item.status === "Selesai";
      default:
        return true;
    }
  });

  return filtered.sort((a, b) => {
    if (sortBy === "status") {
      const wA = getFilterStatusWeight(a.status);
      const wB = getFilterStatusWeight(b.status);
      if (wA !== wB) return wA - wB;
      return (
        getNumberFromStr(a.spbNo, a.poNo) - getNumberFromStr(b.spbNo, b.poNo)
      );
    }

    if (sortBy === "number") {
      const numA = getNumberFromStr(a.spbNo, a.poNo);
      const numB = getNumberFromStr(b.spbNo, b.poNo);
      if (numA !== numB) return numA - numB;
      return getFilterStatusWeight(a.status) - getFilterStatusWeight(b.status);
    }

    if (sortBy === "oldest") {
      const dateA = a.date !== "-" ? new Date(a.date).getTime() : 0;
      const dateB = b.date !== "-" ? new Date(b.date).getTime() : 0;
      return dateA - dateB;
    }

    const dateA = a.date !== "-" ? new Date(a.date).getTime() : 0;
    const dateB = b.date !== "-" ? new Date(b.date).getTime() : 0;
    return dateB - dateA;
  });
};

/**
 * Mengambil maksimal 10 dokumen terbaru berdasarkan tanggal.
 */
export const getRecentSpbDocuments = (
  data: SPBDocumentGroup[],
  limit = 10,
): SPBDocumentGroup[] => {
  return [...data]
    .sort((a, b) => {
      const dateA = a.date !== "-" ? new Date(a.date).getTime() : 0;
      const dateB = b.date !== "-" ? new Date(b.date).getTime() : 0;
      return dateB - dateA;
    })
    .slice(0, limit);
};

// src/features/spb/utils/groupMonitoringData.ts
import type { SPBDocumentGroup } from "../../../shared/types";
import {
  filterVisibleSpbRows,
  isRestrictedSpbStatus,
  toSafeExternalUrl,
} from "../accessRules";
import {
  cleanDocNo,
  getSjTarget,
  isIgnoredSuratJalan,
  toIdNumber,
} from "../suratJalanRules";
import { getSortNumber, toStrOrNull } from "./statusHelpers";
import { buildPoOverrideKey, buildSpbOverrideKey } from "./titleOverrides";

interface InternalGroup {
  data: SPBDocumentGroup;
  hasAnyPO: boolean;
  hasAnyArrival: boolean;
  hasAnyBelumDiproses: boolean;
  hasAnySudahDiproses: boolean;
  latestTimestamp: number;
  poSet: Set<string>;
  completedItems: number;
  itemNames: Set<string>;
  firstItemName: string | null;
}

const buildSpbGroupKey = (
  spbNo: string | null,
  spbTitle: string | null,
  fallbackId: unknown,
): string =>
  spbNo
    ? `SPB_${spbNo}::${spbTitle ?? ""}`
    : `SPB_ID_${fallbackId ?? "UNKNOWN"}`;

/**
 * Judul fallback untuk PO agar card PO tidak pernah kosong walau judul_po NULL di DB.
 *
 * Urutan prioritas:
 *   1. Override manual dari user (localStorage)
 *   2. Kolom judul_po dari DB
 *   3. Judul SPB pasangannya (jika PO ini menempel ke SPB)
 *   4. Nama barang pertama di PO ini
 *   5. "PO {no_po}" sebagai penutup
 */
export const resolvePoTitle = (
  poNo: string | null,
  rawPoTitle: string | null,
  overrideValue: string | undefined,
  spbTitle: string | null,
  firstItemName: string | null,
): string | null => {
  if (overrideValue !== undefined) return overrideValue;
  if (rawPoTitle) return rawPoTitle;
  if (spbTitle) return spbTitle;
  if (firstItemName) return firstItemName;
  if (poNo) return `PO ${poNo}`;
  return null;
};

const updateLatestDate = (group: InternalGroup, dateStr: string | null) => {
  if (!dateStr) return;
  const time = new Date(dateStr).getTime();
  if (!isNaN(time) && time > group.latestTimestamp) {
    group.latestTimestamp = time;
    group.data.date = dateStr;
  }
};

const getGroupStatusWeight = (status: string): number => {
  if (status === "Belum Diproses") return 0;
  if (status === "Sudah PO" || status === "Belum PO") return 1;
  if (status === "Datang") return 2;
  if (status === "Selesai") return 3;
  return 4;
};

export interface GroupMonitoringInput {
  allSpbData: any[];
  poData: any[];
  sjData: any[];
  canViewRestricted: boolean;
  overrides: Record<string, string>;
}

/**
 * Mengelompokkan baris mentah SPB, PO, dan Surat Jalan menjadi daftar card `SPBDocumentGroup[]`.
 */
export const groupMonitoringData = ({
  allSpbData,
  poData,
  sjData,
  canViewRestricted,
  overrides,
}: GroupMonitoringInput): SPBDocumentGroup[] => {
  // 1. Terapkan aturan akses: sembunyikan SPB "belum di proses" untuk user belum login
  const spbData = filterVisibleSpbRows(allSpbData, canViewRestricted);
  const hiddenSpbIds = new Set<number>();
  if (!canViewRestricted) {
    allSpbData.forEach((s: any) => {
      if (isRestrictedSpbStatus(s.status)) {
        const id = toIdNumber(s.id);
        if (id !== null) hiddenSpbIds.add(id);
      }
    });
  }

  // 2. Normalisasi judul SPB per no_spb
  const titleBySpbNo = new Map<string, string>();
  spbData.forEach((row: any) => {
    const no = toStrOrNull(row.no_spb);
    const t = toStrOrNull(row.judul_spb);
    if (no && t && !titleBySpbNo.has(no)) titleBySpbNo.set(no, t);
  });

  // 3. Index id SPB & PO (number-keyed)
  const spbById = new Map<number, any>();
  spbData.forEach((s: any) => {
    const id = toIdNumber(s.id);
    if (id !== null) spbById.set(id, s);
  });

  const poBySpbId = new Map<number, any[]>();
  const poById = new Map<number, any>();
  poData.forEach((p: any) => {
    const poIdNum = toIdNumber(p.id);
    if (poIdNum !== null) poById.set(poIdNum, p);
    const idSpb = toIdNumber(p.id_spb);
    if (idSpb !== null) {
      if (!poBySpbId.has(idSpb)) poBySpbId.set(idSpb, []);
      poBySpbId.get(idSpb)!.push(p);
    }
  });

  const groups = new Map<string, InternalGroup>();

  // 4. Proses baris SPB
  spbData.forEach((row: any) => {
    const spbNo = toStrOrNull(row.no_spb);
    const rawTitle = toStrOrNull(row.judul_spb);
    const spbTitleRaw =
      rawTitle ?? (spbNo ? titleBySpbNo.get(spbNo) ?? null : null);
    const key = buildSpbGroupKey(spbNo, spbTitleRaw, row.id);

    const overrideKey = spbNo ? buildSpbOverrideKey(spbNo, spbTitleRaw) : null;
    const spbOverride = overrideKey ? overrides[overrideKey] : undefined;

    if (!groups.has(key)) {
      groups.set(key, {
        data: {
          id: key,
          spbNo,
          spbTitle:
            spbOverride !== undefined ? spbOverride : spbTitleRaw || null,
          poNo: null,
          poTitle: null,
          date: "-",
          itemsCount: 0,
          sortNumber: getSortNumber(spbNo),
          status: "Belum PO",
        },
        hasAnyPO: false,
        hasAnyArrival: false,
        hasAnyBelumDiproses: false,
        hasAnySudahDiproses: false,
        latestTimestamp: 0,
        poSet: new Set<string>(),
        completedItems: 0,
        itemNames: new Set<string>(),
        firstItemName: null,
      });
    }

    const group = groups.get(key)!;
    if (!group.data.spbTitle && spbTitleRaw) {
      group.data.spbTitle =
        spbOverride !== undefined ? spbOverride : spbTitleRaw;
    }
    if (!group.data.spbSourceFile) {
      group.data.spbSourceFile = toSafeExternalUrl(row.source_file);
    }
    const parentPoId = toIdNumber(row.id_po);
    if (!group.data.poSourceFile && parentPoId !== null) {
      group.data.poSourceFile = toSafeExternalUrl(
        poById.get(parentPoId)?.source_file,
      );
    }
    group.data.itemsCount += 1;

    const itemName = toStrOrNull(row.nama_barang_spb);
    if (itemName) {
      group.itemNames.add(itemName);
      if (!group.firstItemName) group.firstItemName = itemName;
    }

    updateLatestDate(group, row.tgl_spb);

    const rowStatus = toStrOrNull(row.status)?.toLowerCase() || "";
    if (rowStatus === "belum di proses") {
      group.hasAnyBelumDiproses = true;
    } else if (rowStatus === "sudah di proses") {
      group.hasAnySudahDiproses = true;
    }

    const req = row.qty_permintaan || 0;
    const arr = row.material_datang_qty || 0;
    const isArrived = !!row.material_datang_tgl;

    if (isArrived) {
      group.hasAnyArrival = true;
      updateLatestDate(group, row.material_datang_tgl);
      if (arr >= req) group.completedItems += 1;
    }

    const joinedPo = Array.isArray(row.po) ? row.po[0] : row.po;
    if (joinedPo && cleanDocNo(joinedPo.no_po)) {
      group.poSet.add(String(cleanDocNo(joinedPo.no_po)));
      group.hasAnyPO = true;
      updateLatestDate(group, joinedPo.tanggal_po);
    }

    const spbIdNum = toIdNumber(row.id);
    const linkedPos =
      spbIdNum !== null ? poBySpbId.get(spbIdNum) || [] : [];
    linkedPos.forEach((p: any) => {
      if (cleanDocNo(p.no_po)) {
        if (!group.data.poSourceFile) {
          group.data.poSourceFile = toSafeExternalUrl(p.source_file);
        }
        group.poSet.add(String(cleanDocNo(p.no_po)));
        group.hasAnyPO = true;
        updateLatestDate(group, p.tanggal_po);
      }
    });
  });

  // 5. Proses PO (murni atau orphan)
  poData.forEach((p: any) => {
    const idSpb = toIdNumber(p.id_spb);
    const poId = toIdNumber(p.id);

    if (idSpb !== null && hiddenSpbIds.has(idSpb)) return;

    const linkedViaIdSpb = idSpb !== null && spbById.has(idSpb);
    const linkedViaSpbIdPo =
      poId !== null &&
      spbData.some((s: any) => toIdNumber(s.id_po) === poId);

    if (linkedViaIdSpb || linkedViaSpbIdPo) return;

    const poNo = cleanDocNo(p.no_po);
    const key = poNo ? `PO_${poNo}` : `PO_ID_${p.id}`;

    if (!groups.has(key)) {
      groups.set(key, {
        data: {
          id: key,
          spbNo: null,
          spbTitle: null,
          poNo,
          poTitle: null,
          date: p.tanggal_po || "-",
          itemsCount: 0,
          sortNumber: getSortNumber(poNo),
          status: "Sudah PO",
        },
        hasAnyPO: true,
        hasAnyArrival: false,
        hasAnyBelumDiproses: false,
        hasAnySudahDiproses: true,
        latestTimestamp: 0,
        poSet: new Set([poNo || ""]),
        completedItems: 0,
        itemNames: new Set<string>(),
        firstItemName: null,
      });
    }

    const group = groups.get(key)!;
    if (!group.data.poSourceFile) {
      group.data.poSourceFile = toSafeExternalUrl(p.source_file);
    }
    if (!group.data.poTitle) {
      const rawPoTitle = toStrOrNull(p.judul_po);
      if (rawPoTitle) group.data.poTitle = rawPoTitle;
    }
    group.data.itemsCount += 1;

    const itemName = toStrOrNull(p.nama_barang_po);
    if (itemName) {
      group.itemNames.add(itemName);
      if (!group.firstItemName) group.firstItemName = itemName;
    }

    updateLatestDate(group, p.tanggal_po);

    if (p.material_datang_tgl) {
      group.hasAnyArrival = true;
      updateLatestDate(group, p.material_datang_tgl);
      if ((p.material_datang_qty || 0) > 0) group.completedItems += 1;
    }
  });

  // 6. Proses Surat Jalan (no 3+)
  sjData.forEach((sj: any) => {
    if (isIgnoredSuratJalan(sj.no_surat_jalan)) return;

    const spbRel = Array.isArray(sj.spb) ? sj.spb[0] : sj.spb;
    const poRel = Array.isArray(sj.po) ? sj.po[0] : sj.po;
    const sjNo = toStrOrNull(sj.no_surat_jalan);
    const target = getSjTarget(sj);
    const idSpb = toIdNumber(sj.id_spb);

    if (idSpb !== null && hiddenSpbIds.has(idSpb)) return;

    let targetKey: string;
    let spbNo: string | null = null;
    let spbTitle: string | null = null;
    let poNo: string | null = null;

    if (target === "spb") {
      const parentSpb = idSpb !== null ? spbById.get(idSpb) : null;
      spbNo = cleanDocNo(parentSpb?.no_spb) ?? cleanDocNo(spbRel?.no_spb);
      const rawTitle =
        toStrOrNull(parentSpb?.judul_spb) ?? toStrOrNull(spbRel?.judul_spb);
      spbTitle =
        rawTitle ?? (spbNo ? titleBySpbNo.get(spbNo) ?? null : null);
      targetKey = buildSpbGroupKey(spbNo, spbTitle, sj.id_spb);
    } else if (target === "po") {
      poNo = cleanDocNo(poRel?.no_po);
      targetKey = poNo ? `PO_${poNo}` : `PO_ID_${sj.id_po}`;
    } else {
      targetKey = `SJ_${sjNo}`;
    }

    if (!groups.has(targetKey)) {
      groups.set(targetKey, {
        data: {
          id: targetKey,
          spbNo,
          spbTitle,
          poNo,
          poTitle: poNo
            ? resolvePoTitle(
                poNo,
                null,
                overrides[buildPoOverrideKey(poNo)],
                null,
                null,
              )
            : null,
          suratJalanNo: target === "sj" ? sjNo : null,
          date: "-",
          itemsCount: 0,
          sortNumber: getSortNumber(spbNo ?? poNo ?? sjNo),
          status: "Sudah PO",
        },
        hasAnyPO: true,
        hasAnyArrival: false,
        hasAnyBelumDiproses: false,
        hasAnySudahDiproses: true,
        latestTimestamp: 0,
        poSet: new Set<string>(poNo ? [poNo] : []),
        completedItems: 0,
        itemNames: new Set<string>(),
        firstItemName: null,
      });
    }

    const group = groups.get(targetKey)!;
    group.data.itemsCount += 1;

    if (sj.nama_barang) {
      const sjName = String(sj.nama_barang);
      group.itemNames.add(sjName);
      if (!group.firstItemName) group.firstItemName = sjName;
    }

    if (sj.material_datang_tgl) {
      group.hasAnyArrival = true;
      updateLatestDate(group, sj.material_datang_tgl);
      if ((sj.material_datang_qty || 0) > 0) group.completedItems += 1;
    }
  });

  // 7. Finalisasi status + judul PO fallback + sorting
  const result = Array.from(groups.values()).map((g) => {
    const isSelesai =
      g.data.itemsCount > 0 && g.completedItems >= g.data.itemsCount;

    let status = "Belum PO";
    if (isSelesai) status = "Selesai";
    else if (g.hasAnyArrival) status = "Datang";
    else if (g.hasAnyPO) status = "Sudah PO";
    else if (g.hasAnyBelumDiproses && !g.hasAnySudahDiproses) {
      status = "Belum Diproses";
    }

    const combinedPo = Array.from(g.poSet).filter(Boolean).join(", ");
    const firstPo = Array.from(g.poSet).filter(Boolean)[0];

    const poTitle = g.data.poNo
      ? resolvePoTitle(
          firstPo ?? null,
          g.data.poTitle ?? null,
          firstPo && overrides[buildPoOverrideKey(firstPo)] !== undefined
            ? overrides[buildPoOverrideKey(firstPo)]
            : undefined,
          g.data.spbTitle ?? null,
          g.firstItemName,
        )
      : null;

    return {
      ...g.data,
      poNo: combinedPo || null,
      poTitle,
      status,
      sortNumber:
        getSortNumber(g.data.spbNo) + getSortNumber(combinedPo) * 0.01,
      itemNames: Array.from(g.itemNames),
    };
  });

  return result.sort((a, b) => {
    const weightA = getGroupStatusWeight(a.status || "");
    const weightB = getGroupStatusWeight(b.status || "");
    if (weightA !== weightB) return weightA - weightB;

    const dateA = a.date !== "-" ? new Date(a.date).getTime() : 0;
    const dateB = b.date !== "-" ? new Date(b.date).getTime() : 0;
    if (dateA === dateB) return a.sortNumber - b.sortNumber;
    return dateB - dateA;
  });
};

import dayjs from "dayjs";

export const getSortNumber = (val: unknown): number => {
  if (val === null || val === undefined || val === "") return 999999;
  const s = String(val);
  const match = s.match(/\d+/);
  return match ? parseInt(match[0], 10) : 999999;
};

export const getNumberFromStr = (
  spb: string | null,
  po: string | null,
): number => {
  const str = spb || po;
  if (!str) return 999999;
  const match = String(str).match(/\d+/);
  return match ? parseInt(match[0], 10) : 999999;
};

export const toStrOrNull = (val: unknown): string | null => {
  if (val === null || val === undefined || val === "") return null;
  const s = String(val).trim();
  if (!s || s.toLowerCase() === "null" || s.toLowerCase() === "undefined") {
    return null;
  }
  return s;
};

};

  if (status === "Datang Sesuai" || status === "Datang") return "green";
  if (status === "Datang Kurang") return "volcano";
  if (status === "Datang Lebih") return "purple";
  return "default";
};

/**
 * Warna Tag status untuk halaman Rincian Barang (Item Detail).
 */
export const getItemDetailStatusColor = (status?: string | null): string => {
  if (status === "Belum Diproses") return "purple";
  if (status === "Belum PO" || status === "Belum Datang") return "orange";
  if (status === "Sudah PO") return "blue";
  if (
    status === "Datang Sesuai" ||
    status === "Datang" ||
    status === "Selesai"
  ) {
    return "green";
  }
  if (status === "Datang Kurang") return "volcano";
  if (status === "Datang Lebih") return "purple";
  return "blue";
};

/**
 * Menghitung selisih hari (lama proses) antara tanggal mulai dan tanggal datang.
 */
export const calcLamaProses = (
  tglMulai: string | null | undefined,
  tglDatang: string | null | undefined,
  fallback?: string | number | null,
): string => {
  if (fallback && fallback !== "-") return String(fallback);
  if (tglMulai && tglDatang) {
    const start = dayjs(tglMulai);
    const end = dayjs(tglDatang);
    if (start.isValid() && end.isValid()) {
      const diff = end.diff(start, "day");
      return `${diff >= 0 ? diff : 0} Hari`;
    }
  }
  return "-";
};

    setStep("person");

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockStockExits } from "../../mocks/stockExitList";
import { mapStockExits } from "../../shared/lib/mappers";
import type { StockExit } from "../../shared/types";

interface Options { onlyToday?: boolean; breakdown?: boolean; limit?: number; }

export const useStockExits = ({ onlyToday = false, breakdown, limit = 50 }: Options = {}) => {
  const { data, isLoading, refetch } = useQuery<StockExit[]>({
    queryKey: ["stock-exits", onlyToday, breakdown, limit],
    queryFn: async () => {
      if (USE_MOCK) return mockStockExits;
      let query = supabaseClient.from(TABLES.stockExits).select("*").order("created_at", { ascending: false });

      if (onlyToday) {
        const start = dayjs().startOf("day").toISOString();
        const end = dayjs().endOf("day").toISOString();
        query = query.gte("created_at", start).lte("created_at", end);
      }
      if (breakdown !== undefined) query = query.eq("is_breakdown", breakdown);
      
      // ✅ PERBAIKAN: Batasi data history agar tabel tidak hang
      query = query.limit(limit);

      const { data, error } = await query;
      if (error) throw error;
      return mapStockExits(data ?? []);
    },
    refetchInterval: USE_MOCK ? false : 30_000,
  });
  return { exits: data ?? [], isLoading, refetch };
};

import { Result, Button, Typography } from "antd";

const { Paragraph, Text } = Typography;

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  public state: State = { hasError: false, error: null };

  public static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  public componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error("Uncaught error:", error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div style={{ padding: 40, display: "flex", justifyContent: "center" }}>
          <Result
            status="500"
            title="Terjadi Kesalahan (500)"
            subTitle="Maaf, aplikasi mengalami masalah internal."

              <Button type="primary" onClick={() => window.location.reload()}>

                {this.state.error?.message}
              </Paragraph>

          </Result>
        </div>
      );
    }
    return this.props.children;
  }
}

/**
 * @file useAuthGuard.ts
 * @description React hook untuk memverifikasi autentikasi sebelum operasi mutasi.
 *
 * Hook ini menyediakan:
 * 1. Status auth real-time (reactive via event listener)
 * 2. Fungsi guard untuk membungkus mutation callbacks
 * 3. Integrasi dengan Ant Design message untuk feedback UX
 *
 * @example
 * function MyComponent() {
 *   const { isAuthenticated, canEdit, guardMutation } = useAuthGuard();
 *
 *   const handleSave = guardMutation(async (data) => {
 *     await supabaseClient.from('barang').insert(data);
 *   }, { action: 'Simpan barang', permission: 'edit_data' });
 *
 *   return (
 *     <Button onClick={() => handleSave(data)} disabled={!canEdit}>
 *       Simpan
 *     </Button>
 *   );
 * }
 */

import { useState, useEffect, useCallback, useRef } from "react";
import { useNavigate } from "react-router-dom";
import { App } from "antd";
import {
  getVerifiedUser,
  AuthenticationError,
  AuthorizationError,
  SessionExpiredError,
  type AuthStatus,
} from "../lib/authGuard";
import { AUTH_CHANGED_EVENT, type StoredAuthUser } from "../../app/providers/authProvider";
import { canEditData } from "../../app/providers/accessControlProvider";
import type { PermissionKey } from "../types";

// ─────────────────────────────────────────────────────────────────────────────
// Types
// ─────────────────────────────────────────────────────────────────────────────

interface GuardMutationOptions {
  /** Deskripsi aksi untuk audit log dan pesan error */
  action?: string;
  /** Permission yang diperlukan (default: cukup login saja) */
  permission?: PermissionKey;
  /** Pesan error kustom jika auth gagal */
  errorMessage?: string;
  /** Redirect ke login jika belum auth (default: true) */
  redirectOnFail?: boolean;
  /** Callback tambahan ketika auth gagal */
  onAuthFailure?: (error: Error) => void;
}

interface UseAuthGuardReturn {
  /** Apakah user sudah login dengan role terautentikasi */
  isAuthenticated: boolean;
  /** Data user saat ini (null jika belum login) */
  user: StoredAuthUser | null;
  /** Apakah user punya izin edit data */
  canEdit: boolean;
  /** Apakah user adalah Super User */
  isSuperUser: boolean;
  /** Status auth lengkap */
  authStatus: AuthStatus;
  /**
   * Membungkus fungsi mutation dengan verifikasi auth.
   * Jika auth gagal, menampilkan pesan error dan tidak menjalankan mutation.
   */
  guardMutation: <TArgs extends unknown[], TReturn>(
    fn: (...args: TArgs) => Promise<TReturn>,
    options?: GuardMutationOptions,
  ) => (...args: TArgs) => Promise<TReturn | undefined>;
  /**
   * Mengecek auth secara manual dan mengembalikan true/false.
   * Menampilkan pesan error jika gagal.
   */
  checkAuth: (options?: GuardMutationOptions) => boolean;
}

// ─────────────────────────────────────────────────────────────────────────────
// Hook Implementation
// ─────────────────────────────────────────────────────────────────────────────

export const useAuthGuard = (): UseAuthGuardReturn => {
  const { message: antdMessage } = App.useApp();
  const navigate = useNavigate();

  // State yang reactive terhadap perubahan auth
  const [authStatus, setAuthStatus] = useState<AuthStatus>(() => {
    const user = getVerifiedUser();
    return {
      isAuthenticated: !!user,
      user,
      canEdit: user ? canEditData(user.role) : false,
      isSuperUser: user?.role === "superuser",
    };
  });

  // Ref untuk menghindari stale closure di dalam callbacks
  const authStatusRef = useRef(authStatus);
  authStatusRef.current = authStatus;

  // Listen ke perubahan auth (login/logout)
  useEffect(() => {
    const handleAuthChange = () => {
      const user = getVerifiedUser();
      const newStatus: AuthStatus = {
        isAuthenticated: !!user,
        user,
        canEdit: user ? canEditData(user.role) : false,
        isSuperUser: user?.role === "superuser",
      };
      setAuthStatus(newStatus);
    };

    window.addEventListener(AUTH_CHANGED_EVENT, handleAuthChange);
    window.addEventListener("storage", handleAuthChange);

    return () => {
      window.removeEventListener(AUTH_CHANGED_EVENT, handleAuthChange);
      window.removeEventListener("storage", handleAuthChange);
    };
  }, []);

  /**
   * Handle error auth dengan UX yang baik
   */
  const handleAuthError = useCallback(
    (error: Error, options: GuardMutationOptions = {}) => {
      const { redirectOnFail = true, onAuthFailure } = options;

      if (error instanceof AuthenticationError || error instanceof SessionExpiredError) {
        antdMessage.error(
          options.errorMessage ??
            error.message ??
            "Anda harus login terlebih dahulu",
        );

        if (redirectOnFail) {
          // Delay sedikit agar user bisa membaca pesan error
          setTimeout(() => navigate("/login", { replace: true }), 1500);
        }
      } else if (error instanceof AuthorizationError) {
        antdMessage.warning(
          options.errorMessage ??
            error.message ??
            "Anda tidak memiliki izin untuk aksi ini",
        );
      }

      onAuthFailure?.(error);
    },
    [antdMessage, navigate],
  );

  /**
   * Manual auth check dengan feedback UI
   */
  const checkAuth = useCallback(
    (options: GuardMutationOptions = {}): boolean => {
      const user = getVerifiedUser();

      if (!user) {
        handleAuthError(new AuthenticationError(), options);
        return false;
      }

      if (options.permission) {
        const permissionCheckers: Record<string, (role: string) => boolean> = {
          edit_data: canEditData,
        };

        const checker = permissionCheckers[options.permission];
        if (checker && !checker(user.role)) {
          handleAuthError(
            new AuthorizationError(options.permission),
            options,
          );
          return false;
        }
      }

      return true;
    },
    [handleAuthError],
  );

  /**
   * Guard wrapper untuk mutation functions
   */
  const guardMutation = useCallback(
    <TArgs extends unknown[], TReturn>(
      fn: (...args: TArgs) => Promise<TReturn>,
      options: GuardMutationOptions = {},
    ) => {
      return async (...args: TArgs): Promise<TReturn | undefined> => {
        // Verifikasi auth sebelum menjalankan mutation
        if (!checkAuth(options)) {
          return undefined;
        }

        try {
          return await fn(...args);
        } catch (error) {
          // Tangkap error auth yang mungkin terjadi selama operasi
          if (
            error instanceof AuthenticationError ||
            error instanceof AuthorizationError ||
            error instanceof SessionExpiredError
          ) {
            handleAuthError(error, options);
            return undefined;
          }
          // Re-throw error non-auth agar bisa di-handle oleh caller
          throw error;
        }
      };
    },
    [checkAuth, handleAuthError],
  );

  return {
    isAuthenticated: authStatus.isAuthenticated,
    user: authStatus.user,
    canEdit: authStatus.canEdit,
    isSuperUser: authStatus.isSuperUser,
    authStatus,
    guardMutation,
    checkAuth,
  };
};

export default useAuthGuard;

import { useState, useEffect } from "react";

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

import { useEffect } from "react";

interface UseKeyboardShortcutOptions {
  key: string;
  ctrlKey?: boolean;
  shiftKey?: boolean;
  altKey?: boolean;
  metaKey?: boolean;
  enabled?: boolean;
  preventDefault?: boolean;
  ignoreInputs?: boolean;
  handler: (event: KeyboardEvent) => void;
}

const isEditableElement = (el: EventTarget | null): boolean => {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName.toLowerCase();
  return (
    tag === "input" ||
    tag === "textarea" ||
    tag === "select" ||
    el.isContentEditable
  );
};

export const useKeyboardShortcut = ({
  key,
  ctrlKey = false,
  shiftKey = false,
  altKey = false,
  metaKey = false,
  enabled = true,
  preventDefault = true,
  ignoreInputs = false,
  handler,
}: UseKeyboardShortcutOptions) => {
  useEffect(() => {
    if (!enabled) return;

    const onKeyDown = (event: KeyboardEvent) => {
      // Normalisasi nama key (mengabaikan case untuk huruf 'm' dan 'M')
      const isKeyMatch = event.key.toLowerCase() === key.toLowerCase() || event.key === key;
      
      if (!isKeyMatch) return;
      if (event.ctrlKey !== ctrlKey) return;
      if (event.shiftKey !== shiftKey) return;
      if (event.altKey !== altKey) return;
      if (event.metaKey !== metaKey) return;
      if (ignoreInputs && isEditableElement(event.target)) return;

      if (preventDefault) event.preventDefault();
      handler(event);
    };

    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  }, [
    key,
    ctrlKey,
    shiftKey,
    altKey,
    metaKey,
    enabled,
    preventDefault,
    ignoreInputs,
    handler,
  ]);
};

/**
 * @file authGuard.ts
 * @description Lapisan keamanan terpusat untuk memverifikasi autentikasi
 * sebelum operasi mutasi (POST/PUT/DELETE) ke server.
 *
 * PRINSIP KEAMANAN:
 * 1. Defense-in-depth: Verifikasi di UI + API layer + Supabase RLS
 * 2. Fail-closed: Jika status auth tidak jelas, tolak akses
 * 3. Least privilege: Hanya user dengan role terautentikasi yang bisa mutasi
 * 4. Audit trail: Log setiap percobaan akses yang ditolak
 *
 * @example
 * // Di dalam fungsi API / mutation:
 * import { requireAuth, requirePermission } from '@/shared/lib/authGuard';
 *
 * const createBarang = async (input: BarangInput) => {
 *   requireAuth();              // Memastikan user sudah login
 *   requirePermission('edit_data'); // Memastikan punya izin edit
 *   // ... lanjut operasi
 * };
 */

interface AuditLogEntry {
  timestamp: string;
  action: string;
  userId: string | null;
  userEmail: string | null;
  userRole: string | null;
  result: "allowed" | "denied";
  reason?: string;
  metadata?: Record<string, unknown>;
}

const MAX_AUDIT_LOG_SIZE = 500;
const AUDIT_LOG_KEY = "security-audit-log";

/**
 * Mencatat aktivitas keamanan ke localStorage untuk audit trail.
 * Pada production, ini sebaiknya dikirim ke server logging.
 */
const logAudit = (entry: Omit<AuditLogEntry, "timestamp">): void => {
  try {
    const fullEntry: AuditLogEntry = {
      ...entry,
      timestamp: new Date().toISOString(),
    };

    // Console warning untuk akses yang ditolak (dev mode)
    if (entry.result === "denied") {
      console.warn(
        `🔒 [Security] Akses ditolak: ${entry.action}`,
        {
          user: entry.userEmail ?? "anonymous",
          role: entry.userRole ?? "none",
          reason: entry.reason,
        },
      );
    }

    // Simpan ke localStorage sebagai audit trail sederhana
    const raw = localStorage.getItem(AUDIT_LOG_KEY);
    const logs: AuditLogEntry[] = raw ? JSON.parse(raw) : [];
    logs.unshift(fullEntry);

    // Batas ukuran agar tidak membebani storage
    if (logs.length > MAX_AUDIT_LOG_SIZE) {
      logs.length = MAX_AUDIT_LOG_SIZE;
    }

    localStorage.setItem(AUDIT_LOG_KEY, JSON.stringify(logs));
  } catch {
    // Jangan sampai audit logging mengganggu operasi utama
  }
};

/**
 * Membaca log audit. Berguna untuk dashboard admin / debugging.
 */
export const getAuditLogs = (): AuditLogEntry[] => {
  try {
    const raw = localStorage.getItem(AUDIT_LOG_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
};

/**
 * Membersihkan log audit.
 */
export const clearAuditLogs = (): void => {
  try {
    localStorage.removeItem(AUDIT_LOG_KEY);
  } catch {
    // ignore
  }
};

/**
 * Mendapatkan user yang sudah terverifikasi dari stored session.
 * Melakukan validasi menyeluruh terhadap data user.
 *
 * @returns StoredAuthUser jika valid, null jika tidak
 */
export const getVerifiedUser = (): StoredAuthUser | null => {
  const user = getStoredUser();

  if (!user) return null;

  // Validasi integritas data user
  if (!user.email || typeof user.email !== "string") return null;
  if (!user.role || typeof user.role !== "string") return null;
  if (!isAuthenticatedRole(user.role)) return null;

  return user;
};

/**
 * Memverifikasi sesi Supabase masih valid (untuk mode non-mock).
 * Ini memastikan token JWT belum kedaluwarsa.
 *
 * @returns true jika sesi masih valid
 */
export const verifySupabaseSession = async (): Promise<boolean> => {
  if (USE_MOCK) return true;

  try {
    const { data, error } = await supabaseClient.auth.getSession();
    if (error) return false;
    return !!data?.session;
  } catch {
    return false;
  }
};

// ─────────────────────────────────────────────────────────────────────────────
// Guard Functions (Synchronous — untuk penggunaan di awal setiap mutation)
// ─────────────────────────────────────────────────────────────────────────────

/**
 * GUARD UTAMA: Memastikan user sudah login sebelum operasi mutasi.
 *
 * Melempar `AuthenticationError` jika user belum login.
 *
 * @param actionDescription - Deskripsi aksi untuk audit log (opsional)
 * @returns User yang sudah terverifikasi
 *
 * @throws {AuthenticationError} Jika user belum login
 *
 * @example
 * const handleSave = async (data: any) => {
 *   const user = requireAuth('Simpan data barang');
 *   console.log(`${user.name} sedang menyimpan data...`);
 *   await supabaseClient.from('barang').insert(data);
 * };
 */
export const requireAuth = (actionDescription?: string): StoredAuthUser => {
  const user = getVerifiedUser();
  const action = actionDescription ?? "operasi mutasi";

  if (!user) {
    logAudit({
      action,
      userId: null,
      userEmail: null,
      userRole: null,
      result: "denied",
      reason: "User belum login atau sesi tidak valid",
    });

    throw new AuthenticationError();
  }

  logAudit({
    action,
    userId: user.id ?? null,
    userEmail: user.email,
    userRole: user.role,
    result: "allowed",
  });

  return user;
};

/**
 * GUARD ASYNC: Versi async dari requireAuth yang juga memvalidasi sesi Supabase.
 * Gunakan ini untuk operasi kritikal yang membutuhkan validasi ekstra.
 *
 * @param actionDescription - Deskripsi aksi untuk audit log
 * @returns User yang sudah terverifikasi
 *
 * @throws {AuthenticationError} Jika user belum login
 * @throws {SessionExpiredError} Jika sesi Supabase sudah kedaluwarsa
 *
 * @example
 * const handleDelete = async (id: number) => {
 *   const user = await requireAuthStrict('Hapus barang');
 *   await supabaseClient.from('barang').delete().eq('id', id);
 * };
 */
export const requireAuthStrict = async (
  actionDescription?: string,
): Promise<StoredAuthUser> => {
  // Step 1: Verifikasi user di local state
  const user = requireAuth(actionDescription);

  // Step 2: Verifikasi sesi Supabase masih valid (non-mock only)
  if (!USE_MOCK) {
    const isSessionValid = await verifySupabaseSession();
    if (!isSessionValid) {
      logAudit({
        action: actionDescription ?? "operasi mutasi (strict)",
        userId: user.id ?? null,
        userEmail: user.email,
        userRole: user.role,
        result: "denied",
        reason: "Sesi Supabase sudah kedaluwarsa",
      });

      throw new SessionExpiredError();
    }
  }

  return user;
};

/**
 * GUARD PERMISSION: Memastikan user memiliki permission tertentu.
 *
 * @param permission - Permission yang dibutuhkan
 * @param actionDescription - Deskripsi aksi untuk audit log
 * @returns User yang sudah terverifikasi
 *
 * @throws {AuthenticationError} Jika user belum login
 * @throws {AuthorizationError} Jika user tidak punya permission
 *
 * @example
 * const handleEdit = async (id: number, data: any) => {
 *   requirePermission('edit_data', 'Edit data barang');
 *   await supabaseClient.from('barang').update(data).eq('id', id);
 * };
 */
export const requirePermission = (
  permission: PermissionKey,
  actionDescription?: string,
): StoredAuthUser => {
  const user = requireAuth(actionDescription);
  const action = actionDescription ?? `operasi dengan izin ${permission}`;

  const permissionCheckers: Record<PermissionKey, (role: string) => boolean> = {
    edit_data: canEditData,
    approve_accounts: canApproveAccounts,
    manage_roles: canManageRoles,
  };

  const checker = permissionCheckers[permission];
  if (!checker || !checker(user.role)) {
    logAudit({
      action,
      userId: user.id ?? null,
      userEmail: user.email,
      userRole: user.role,
      result: "denied",
      reason: `Tidak memiliki izin: ${permission}`,
    });

    throw new AuthorizationError(permission);
  }

  return user;
};

// Utility: Guarded Supabase Operations
// ─────────────────────────────────────────────────────────────────────────────

/**
 * Helper type untuk operasi Supabase yang membutuhkan auth.
 */
type SupabaseOperation<T> = () => Promise<T>;

/**
 * Menjalankan operasi Supabase dengan verifikasi auth di depan.
 * Wrapper yang menggabungkan auth check + operasi dalam satu panggilan.
 *
 * @param operation - Fungsi async yang menjalankan operasi Supabase
 * @param options - Konfigurasi guard
 * @returns Hasil dari operasi
 *
 * @example
 * const result = await guardedMutation(
 *   () => supabaseClient.from('barang').insert(data).select().single(),
 *   { action: 'Tambah barang', permission: 'edit_data' }
 * );
 */
export const guardedMutation = async <T>(
  operation: SupabaseOperation<T>,
  options: {
    action?: string;
    permission?: PermissionKey;
    strict?: boolean;
  } = {},
): Promise<T> => {
  const { action, permission, strict = false } = options;

  // Step 1: Auth check
  if (strict) {
    await requireAuthStrict(action);
  } else if (permission) {
    requirePermission(permission, action);
  } else {
    requireAuth(action);
  }

  // Step 2: Execute operation
  return operation();
};

  };

import type { Product, StockExit } from "../types";

  return {
    id: r.id, code: r.kode, name: r.nama, size: r.ukuran ?? null, specification: r.spesifikasi ?? null,
    category: r.kategori ?? "Material", categoryId: null, categoryName: null,
    unit: r.satuan ?? "pcs", stock: r.stok ?? 0, minStock: 0, costPrice: 0,
    status: undefined, createdAt: r.created_at ?? new Date().toISOString(),
  };

interface StockExitRow {
  id: number; nomor_transaksi: string | null; barang_id: number; nama_pekerja: string | null;
  nama_barang: string; ukuran_barang: string | null; satuan_barang: string | null;
  jumlah_barang_keluar: number; tujuan: string | null; is_breakdown: boolean | null; created_at: string;
}

export const mapStockExits = (rows: any[]): StockExit[] => {
  const typedRows = rows as StockExitRow[];
  const grouped = new Map<string, StockExit>();

  for (const row of typedRows) {
    const key = row.nomor_transaksi ?? `temp-${row.id}`;
    if (!grouped.has(key)) {
      const created = new Date(row.created_at);
      grouped.set(key, {
        id: row.id, nomorTransaksi: key, timestamp: row.created_at, date: created.toISOString().split("T")[0],
        time: created.toISOString().slice(11, 16), person: row.nama_pekerja ?? "-",
        destination: row.tujuan, isBreakdown: row.is_breakdown ?? true, items: [],
      });
    }
    grouped.get(key)!.items.push({
      itemId: row.barang_id, itemName: row.nama_barang, size: row.ukuran_barang,
      unit: row.satuan_barang ?? "pcs", quantity: row.jumlah_barang_keluar,
    });
  }
  return Array.from(grouped.values());
};
