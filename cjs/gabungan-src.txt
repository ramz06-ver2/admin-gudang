import type { AccessControlProvider } from "@refinedev/core";
import type { UserRole } from "../../shared/types";
import { getStoredUser } from "./authProvider";

export const canManageBarang = (role?: UserRole | string | null): boolean =>
  role === "admin" || role === "superuser";

export const canAccessNotifications = (role?: UserRole | string | null): boolean =>
  role === "superuser";

export const accessControlProvider: AccessControlProvider = {
  can: async ({ resource, action }) => {
    const user = getStoredUser();
    const role: UserRole = user?.role ?? "user";

    if (resource === "notifications" || resource === "notifikasi") {
      return {
        can: canAccessNotifications(role),
        reason: "Hanya Super User yang memiliki akses ke halaman Notifikasi",
      };
    }

    if (
      resource === "barang" &&
      (action === "create" || action === "edit" || action === "delete")
    ) {
      return {
        can: canManageBarang(role),
        reason: "Hanya Admin dan Super User yang dapat mengelola data barang",
      };
    }

    if (resource === "barang_masuk" || resource === "pengeluaran") {
      return {
        can: canManageBarang(role),
        reason: "Silakan masuk sebagai Admin atau Super User",
      };
    }

    return { can: true };
  },
};

import type { AuthProvider, AuthActionResponse, CheckResponse } from "@refinedev/core";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { USE_MOCK } from "../../shared/lib/config";
import type { UserRole } from "../../shared/types";
import {
  findAccountRequestByEmail,
  registerAccountRequest,
} from "../../features/auth/accountRequests";

export const STORAGE_KEY = "mock-auth-user";

export interface StoredAuthUser {
  id?: string;
  email: string;
  name: string;
  role: UserRole;
}

export const getStoredUser = (): StoredAuthUser | null => {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
};

export const setStoredUser = (user: StoredAuthUser | null) => {
  try {
    if (!user) {
      localStorage.removeItem(STORAGE_KEY);
    } else {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(user));
    }
  } catch {
    // ignore
  }
};

export const authProvider: AuthProvider = {
  login: async ({ email, password }): Promise<AuthActionResponse> => {
    const normalizedEmail = String(email || "").trim().toLowerCase();

    // 1. Cek apakah akun berasal dari alur pendaftaran user -> notifikasi super user
    const accountReq = findAccountRequestByEmail(normalizedEmail);
    if (accountReq) {
      if (accountReq.status === "pending") {
        return {
          success: false,
          error: {
            name: "LoginError",
            message: "Akun Anda masih menunggu persetujuan Super User",
          },
        };
      }
      if (accountReq.status === "rejected") {
        return {
          success: false,
          error: {
            name: "LoginError",
            message: "Pendaftaran akun Anda tidak diizinkan oleh Super User",
          },
        };
      }
      if (accountReq.status === "approved") {
        if (!accountReq.password || accountReq.password === password) {
          setStoredUser({
            id: accountReq.id,
            email: accountReq.email,
            name: accountReq.name,
            role: "admin",
          });
          return { success: true, redirectTo: "/dashboard" };
        }
        return {
          success: false,
          error: { name: "LoginError", message: "Email/password salah" },
        };
      }
    }

    // 2. Akun bawaan Super User & Admin
    if (normalizedEmail === "superuser@admin.com" && password === "super123") {
      setStoredUser({
        id: "superuser-1",
        email: normalizedEmail,
        name: "Super User",
        role: "superuser",
      });
      return { success: true, redirectTo: "/dashboard" };
    }

    if (normalizedEmail === "admin@admin.com" && password === "admin123") {
      setStoredUser({
        id: "admin-1",
        email: normalizedEmail,
        name: "Administrator",
        role: "admin",
      });
      return { success: true, redirectTo: "/dashboard" };
    }

    if (USE_MOCK) {
      return {
        success: false,
        error: { name: "LoginError", message: "Email/password salah" },
      };
    }

    const { data, error } = await supabaseClient.auth.signInWithPassword({
      email,
      password,
    });
    if (error || !data?.user) {
      return {
        success: false,
        error: {
          name: "LoginError",
          message: error?.message || "User tidak ditemukan",
        },
      };
    }

    const metaRole = (data.user as any)?.user_metadata?.role as UserRole | undefined;
    const resolvedRole: UserRole =
      metaRole === "superuser" || normalizedEmail.startsWith("superuser@")
        ? "superuser"
        : "admin";

    setStoredUser({
      id: data.user.id,
      email: data.user.email || normalizedEmail,
      name: data.user.email?.split("@")[0] ?? "Admin",
      role: resolvedRole,
    });

    return { success: true, redirectTo: "/dashboard" };
  },

  register: async ({ name, email, password }): Promise<AuthActionResponse> => {
    try {
      await registerAccountRequest({
        name: name || String(email || "").split("@")[0] || "Calon Admin",
        email,
        password,
      });
      return {
        success: true,
        redirectTo: "/login",
      };
    } catch (err) {
      return {
        success: false,
        error: {
          name: "RegisterError",
          message: (err as Error)?.message || "Gagal mendaftarkan akun",
        },
      };
    }
  },

  logout: async (): Promise<AuthActionResponse> => {
    setStoredUser(null);
    if (!USE_MOCK) {
      try {
        await supabaseClient.auth.signOut();
      } catch {
        // ignore
      }
    }
    return { success: true, redirectTo: "/dashboard" };
  },

  check: async (): Promise<CheckResponse> => {
    const stored = getStoredUser();
    if (stored && (stored.role === "admin" || stored.role === "superuser")) {
      return { authenticated: true };
    }
    if (USE_MOCK) {
      return { authenticated: false, redirectTo: "/login" };
    }
    const { data } = await supabaseClient.auth.getSession();
    return {
      authenticated: !!data?.session,
      redirectTo: "/login",
    };
  },

  getIdentity: async () => {
    const stored = getStoredUser();
    if (stored) {
      return {
        id: stored.id || (stored.role === "superuser" ? "superuser-1" : "admin-1"),
        name: stored.name,
        email: stored.email,
        role: stored.role,
      };
    }
    if (!USE_MOCK) {
      const { data } = await supabaseClient.auth.getUser();
      if (data?.user) {
        const email = data.user.email || "";
        const metaRole = (data.user as any)?.user_metadata?.role as UserRole | undefined;
        const role: UserRole =
          metaRole === "superuser" || email.toLowerCase().startsWith("superuser@")
            ? "superuser"
            : "admin";
        return {
          id: data.user.id,
          name: email.split("@")[0] || (role === "superuser" ? "Super User" : "Admin"),
          email,
          role,
        };
      }
    }
    return {
      id: "public-user",
      name: "User (Public)",
      email: "",
      role: "user" as UserRole,
    };
  },

  getPermissions: async () => {
    const stored = getStoredUser();
    if (stored) return [stored.role];
    if (!USE_MOCK) {
      const { data } = await supabaseClient.auth.getSession();
      if (data?.session) return ["admin"];
    }
    return ["user"];
  },

  onError: async (error) => ({ error }),
};

import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import type { AccountRequest } from "../../shared/types";

const REQUESTS_STORAGE_KEY = "account-registration-requests";

export const readLocalRequests = (): AccountRequest[] => {
  try {
    const raw = localStorage.getItem(REQUESTS_STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
};

export const writeLocalRequests = (requests: AccountRequest[]) => {
  try {
    localStorage.setItem(REQUESTS_STORAGE_KEY, JSON.stringify(requests));
  } catch {
    // ignore storage error
  }
};

export const resetAccountRequests = () => {
  try {
    localStorage.removeItem(REQUESTS_STORAGE_KEY);
  } catch {
    // ignore
  }
};

export const getPendingRequestsCount = (): number => {
  return readLocalRequests().filter((r) => r.status === "pending").length;
};

export const getAccountRequests = async (): Promise<AccountRequest[]> => {
  const local = readLocalRequests();
  try {
    const res = await supabaseClient
      .from(TABLES.accountRequests)
      .select("*");
    if (!res.error && Array.isArray(res.data) && res.data.length > 0) {
      const mergedMap = new Map<string, AccountRequest>();
      local.forEach((item) => mergedMap.set(item.email.toLowerCase(), item));
      res.data.forEach((row: any) => {
        const emailKey = String(row.email || "").toLowerCase();
        if (!emailKey) return;
        const existing = mergedMap.get(emailKey);
        mergedMap.set(emailKey, {
          id: String(row.id ?? existing?.id ?? Date.now()),
          name: row.name || existing?.name || "Calon Admin",
          email: row.email,
          password: row.password ?? existing?.password,
          role: "admin",
          status: row.status || existing?.status || "pending",
          createdAt: row.created_at || row.createdAt || existing?.createdAt || new Date().toISOString(),
          decidedAt: row.decided_at || row.decidedAt || existing?.decidedAt || null,
        });
      });
      const merged = Array.from(mergedMap.values());
      writeLocalRequests(merged);
      return merged;
    }
  } catch {
    // fallback ke localStorage jika tabel belum tersedia
  }
  return local;
};

export const registerAccountRequest = async (input: {
  name: string;
  email: string;
  password: string;
}): Promise<AccountRequest> => {
  const name = input.name.trim();
  const email = input.email.trim().toLowerCase();
  const password = input.password;

  if (!name || !email || !password) {
    throw new Error("Nama, email, dan password wajib diisi");
  }

  const existingList = readLocalRequests();
  const duplicate = existingList.find(
    (r) => r.email.toLowerCase() === email && r.status !== "rejected",
  );
  if (duplicate) {
    throw new Error(
      duplicate.status === "pending"
        ? "Email ini sudah terdaftar dan masih menunggu persetujuan Super User"
        : "Email ini sudah terdaftar sebagai Admin",
    );
  }

  const newRequest: AccountRequest = {
    id: `req-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
    name,
    email,
    password,
    role: "admin",
    status: "pending",
    createdAt: new Date().toISOString(),
    decidedAt: null,
  };

  const updated = [
    newRequest,
    ...existingList.filter((r) => r.email.toLowerCase() !== email),
  ];
  writeLocalRequests(updated);

  try {
    await supabaseClient.from(TABLES.accountRequests).insert([
      {
        id: newRequest.id,
        name: newRequest.name,
        email: newRequest.email,
        password: newRequest.password,
        role: "admin",
        status: "pending",
        created_at: newRequest.createdAt,
      },
    ]);
  } catch {
    // fallback ke localStorage
  }

  return newRequest;
};

export const approveAccountRequest = async (
  requestId: string,
): Promise<AccountRequest> => {
  const list = readLocalRequests();
  const idx = list.findIndex((r) => String(r.id) === String(requestId));
  const decidedAt = new Date().toISOString();

  let updatedItem: AccountRequest;
  if (idx >= 0) {
    updatedItem = {
      ...list[idx],
      status: "approved",
      role: "admin",
      decidedAt,
    };
    list[idx] = updatedItem;
    writeLocalRequests(list);
  } else {
    updatedItem = {
      id: requestId,
      name: "Admin Baru",
      email: "",
      role: "admin",
      status: "approved",
      createdAt: decidedAt,
      decidedAt,
    };
  }

  try {
    await supabaseClient
      .from(TABLES.accountRequests)
      .update({ status: "approved", decided_at: decidedAt })
      .eq("id", requestId);
  } catch {
    // fallback ke localStorage
  }

  return updatedItem;
};

export const rejectAccountRequest = async (
  requestId: string,
): Promise<AccountRequest> => {
  const list = readLocalRequests();
  const idx = list.findIndex((r) => String(r.id) === String(requestId));
  const decidedAt = new Date().toISOString();

  let updatedItem: AccountRequest;
  if (idx >= 0) {
    updatedItem = {
      ...list[idx],
      status: "rejected",
      decidedAt,
    };
    list[idx] = updatedItem;
    writeLocalRequests(list);
  } else {
    updatedItem = {
      id: requestId,
      name: "Calon Admin",
      email: "",
      role: "admin",
      status: "rejected",
      createdAt: decidedAt,
      decidedAt,
    };
  }

  try {
    await supabaseClient
      .from(TABLES.accountRequests)
      .update({ status: "rejected", decided_at: decidedAt })
      .eq("id", requestId);
  } catch {
    // fallback ke localStorage
  }

  return updatedItem;
};

export const findAccountRequestByEmail = (
  email: string,
): AccountRequest | undefined => {
  const normalized = email.trim().toLowerCase();
  return readLocalRequests().find((r) => r.email.toLowerCase() === normalized);
};

import { useEffect, useState } from "react";
import { useGetIdentity } from "@refinedev/core";
import {
  Card,
  Typography,
  Table,
  Button,
  Tag,
  Space,
  Empty,
  Result,
  message,
} from "antd";
import type { ColumnsType } from "antd/es/table";
import {
  BellOutlined,
  CheckCircleOutlined,
  CloseCircleOutlined,
} from "@ant-design/icons";
import type { AccountRequest, UserRole } from "../../../shared/types";
import {
  getAccountRequests,
  approveAccountRequest,
  rejectAccountRequest,
} from "../accountRequests";
import { getStoredUser } from "../../../app/providers/authProvider";
import { formatDate } from "../../../shared/lib/format";

const { Title, Text } = Typography;

interface IdentityUser {
  id: string;
  name: string;
  email?: string;
  role: UserRole;
}

export const NotificationsPage = () => {
  const { data: identity } = useGetIdentity<IdentityUser>();
  const storedUser = getStoredUser();
  const effectiveRole: UserRole =
    identity?.role ?? storedUser?.role ?? "user";

  const [requests, setRequests] = useState<AccountRequest[]>([]);
  const [loading, setLoading] = useState<boolean>(true);

  const loadRequests = async () => {
    setLoading(true);
    try {
      const data = await getAccountRequests();
      setRequests(data);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadRequests();
  }, []);

  if (effectiveRole !== "superuser") {
    return (
      <Result
        status="403"
        title="Akses Ditolak"
        subTitle="Hanya Super User yang dapat mengakses halaman Notifikasi persetujuan akun."
      />
    );
  }

  const handleApprove = async (req: AccountRequest) => {
    const updated = await approveAccountRequest(req.id);
    setRequests((prev) =>
      prev.map((item) => (item.id === req.id ? updated : item)),
    );
    message.success(`Akun ${req.email} diizinkan dan aktif sebagai Admin.`);
  };

  const handleReject = async (req: AccountRequest) => {
    const updated = await rejectAccountRequest(req.id);
    setRequests((prev) =>
      prev.map((item) => (item.id === req.id ? updated : item)),
    );
    message.info(`Permintaan akun ${req.email} tidak diizinkan (dibatalkan).`);
  };

  const getStatusTag = (status: AccountRequest["status"]) => {
    if (status === "pending") {
      return <Tag color="orange">Menunggu Persetujuan</Tag>;
    }
    if (status === "approved") {
      return <Tag color="green">Diizinkan (Admin Aktif)</Tag>;
    }
    return <Tag color="red">Tidak Diizinkan (Batal)</Tag>;
  };

  const columns: ColumnsType<AccountRequest> = [
    {
      title: "Nama Calon Admin",
      dataIndex: "name",
      key: "name",
      render: (v: string) => <Text strong>{v}</Text>,
    },
    {
      title: "Email",
      dataIndex: "email",
      key: "email",
    },
    {
      title: "Tanggal Pengajuan",
      dataIndex: "createdAt",
      key: "createdAt",
      render: (v: string) => (v ? formatDate(v) : "-"),
    },
    {
      title: "Status",
      dataIndex: "status",
      key: "status",
      render: (status: AccountRequest["status"]) => getStatusTag(status),
    },
    {
      title: "Aksi Persetujuan",
      key: "actions",
      render: (_: unknown, record: AccountRequest) => {
        if (record.status !== "pending") {
          return (
            <Text type="secondary" style={{ fontSize: 12 }}>
              {record.status === "approved"
                ? "Akun Admin telah aktif"
                : "Pembuatan akun dibatalkan"}
            </Text>
          );
        }
        return (
          <Space>
            <Button
              type="primary"
              size="small"
              icon={<CheckCircleOutlined />}
              onClick={() => handleApprove(record)}
              aria-label={`Izinkan ${record.email}`}
            >
              Izinkan
            </Button>
            <Button
              danger
              size="small"
              icon={<CloseCircleOutlined />}
              onClick={() => handleReject(record)}
              aria-label={`Tidak Izinkan ${record.email}`}
            >
              Tidak Izinkan
            </Button>
          </Space>
        );
      },
    },
  ];

  return (
    <Space direction="vertical" size="large" style={{ width: "100%" }}>
      <Space align="center">
        <div
          style={{
            width: 42,
            height: 42,
            borderRadius: 10,
            background: "#722ed115",
            color: "#722ed1",
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
            fontSize: 20,
          }}
        >
          <BellOutlined />
        </div>
        <div>
          <Title level={3} style={{ margin: 0 }}>
            Notifikasi Super User
          </Title>
          <Text type="secondary">
            Kelola permintaan pembuatan akun Admin baru
          </Text>
        </div>
      </Space>

      <Card variant="borderless" style={{ borderRadius: 12 }}>
        {requests.length === 0 ? (
          <Empty description="Belum ada permintaan pendaftaran akun baru" />
        ) : (
          <Table<AccountRequest>
            dataSource={requests}
            columns={columns}
            rowKey="id"
            loading={loading}
            pagination={{ pageSize: 10 }}
          />
        )}
      </Card>
    </Space>
  );
};

export default NotificationsPage;

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { getMockDashboardStats } from "../../mocks/dashboard";
import type { DashboardStats } from "../../shared/types";

export interface UseDashboardStatsResult {
  stats: DashboardStats;
  isLoading: boolean;
  refetch: () => void;
}

const EMPTY_STATS: DashboardStats = {
  totalProducts: 0,
  totalStock: 0,
  todayInbound: 0,
  todayOutbound: 0,
};

export const useDashboardStats = (): UseDashboardStatsResult => {
  const query = useQuery<DashboardStats, Error, DashboardStats>({
    queryKey: ["dashboard-stats"],
    refetchInterval: USE_MOCK ? false : 60_000,
    queryFn: async (): Promise<DashboardStats> => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return getMockDashboardStats();
      }

      const today = dayjs().format("YYYY-MM-DD");
      const startOfDay = dayjs().startOf("day").toISOString();
      const endOfDay = dayjs().endOf("day").toISOString();

      const [productsRes, inboundRes, outboundRes] = await Promise.all([
        supabaseClient.from(TABLES.products).select("stok"),
        supabaseClient
          .from(TABLES.stockEntries)
          .select("jumlah_barang_masuk")
          .eq("tanggal_pengiriman", today),
        supabaseClient
          .from(TABLES.stockExits)
          .select("jumlah_barang_keluar")
          .gte("created_at", startOfDay)
          .lte("created_at", endOfDay),
      ]);

      if (productsRes.error) throw productsRes.error;
      if (inboundRes.error) throw inboundRes.error;
      if (outboundRes.error) throw outboundRes.error;

      const totalProducts = productsRes.data?.length ?? 0;
      const totalStock =
        productsRes.data?.reduce(
          (sum, p: { stok: number | null }) => sum + (p.stok ?? 0),
          0,
        ) ?? 0;

      const todayInbound =
        inboundRes.data?.reduce(
          (sum, e: { jumlah_barang_masuk: number | null }) =>
            sum + (e.jumlah_barang_masuk ?? 0),
          0,
        ) ?? 0;

      const todayOutbound =
        outboundRes.data?.reduce(
          (sum, exit: { jumlah_barang_keluar: number | null }) =>
            sum + (exit.jumlah_barang_keluar ?? 0),
          0,
        ) ?? 0;

      return { totalProducts, totalStock, todayInbound, todayOutbound };
    },
  });

  return {
    stats: query.data ?? EMPTY_STATS,
    isLoading: query.isLoading,
    refetch: query.refetch,
  };
};

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockStockInChart } from "../../mocks/stockEntries";
import type { StockInChartItem } from "../../shared/types";

export const useStockInChart = (days = 30) => {
  const { data, isLoading, refetch } = useQuery<StockInChartItem[]>({
    queryKey: ["stock-in-chart", days],
    queryFn: async () => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return mockStockInChart.slice(-days);
      }

      const fromDate = dayjs().subtract(days, "day").format("YYYY-MM-DD");

      const { data, error } = await supabaseClient
        .from(TABLES.stockEntries)
        .select("tanggal_pengiriman, jumlah_barang_masuk")
        .gte("tanggal_pengiriman", fromDate)
        .order("tanggal_pengiriman", { ascending: true });

      if (error) throw error;

      const grouped = (data ?? []).reduce<Record<string, number>>(
        (
          acc,
          row: {
            tanggal_pengiriman: string | null;
            jumlah_barang_masuk: number | null;
          },
        ) => {
          if (!row.tanggal_pengiriman) return acc;
          acc[row.tanggal_pengiriman] =
            (acc[row.tanggal_pengiriman] ?? 0) + (row.jumlah_barang_masuk ?? 0);
          return acc;
        },
        {},
      );

      return Object.entries(grouped).map(([date, total]) => ({
        date,
        total,
      }));
    },
  });

  return {
    items: data ?? [],
    isLoading,
    refetch,
  };
};

import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { mapProduct } from "../../shared/lib/mappers";
import type { Product } from "../../shared/types";

export interface BarangInput {
  kode?: string;
  nama: string;
  ukuran?: string | null;
  spesifikasi?: string | null;
  kategori?: string;
  satuan?: string;
  stok?: number;
}

export const createBarang = async (input: BarangInput): Promise<Product> => {
  const rowPayload = {
    kode:
      input.kode?.trim() ||
      `BRG-${String(Math.floor(100 + Math.random() * 900))}`,
    nama: input.nama.trim(),
    ukuran: input.ukuran?.trim() || null,
    spesifikasi: input.spesifikasi?.trim() || null,
    kategori: input.kategori || "Material",
    satuan: input.satuan?.trim() || "pcs",
    stok: Number(input.stok ?? 0),
  };

  const { data, error } = await supabaseClient
    .from(TABLES.products)
    .insert([rowPayload])
    .select()
    .single();

  if (error) throw error;

  return mapProduct(
    data ?? {
      id: Date.now(),
      ...rowPayload,
      created_at: new Date().toISOString(),
    },
  );
};

export const updateBarang = async (
  id: number,
  input: Partial<BarangInput>,
): Promise<Product> => {
  const patch: Record<string, unknown> = {};
  if (input.kode !== undefined) patch.kode = input.kode.trim();
  if (input.nama !== undefined) patch.nama = input.nama.trim();
  if (input.ukuran !== undefined) patch.ukuran = input.ukuran?.trim() || null;
  if (input.spesifikasi !== undefined)
    patch.spesifikasi = input.spesifikasi?.trim() || null;
  if (input.kategori !== undefined) patch.kategori = input.kategori;
  if (input.satuan !== undefined) patch.satuan = input.satuan.trim();
  if (input.stok !== undefined) patch.stok = Number(input.stok);

  const { data, error } = await supabaseClient
    .from(TABLES.products)
    .update(patch)
    .eq("id", id)
    .select()
    .single();

  if (error) throw error;

  return mapProduct(
    data ?? {
      id,
      kode: String(patch.kode ?? `BRG-${id}`),
      nama: String(patch.nama ?? ""),
      ukuran: (patch.ukuran as string | null) ?? null,
      spesifikasi: (patch.spesifikasi as string | null) ?? null,
      kategori: String(patch.kategori ?? "Material"),
      satuan: String(patch.satuan ?? "pcs"),
      stok: Number(patch.stok ?? 0),
      created_at: new Date().toISOString(),
    },
  );
};

export const deleteBarang = async (id: number): Promise<void> => {
  const { error } = await supabaseClient
    .from(TABLES.products)
    .delete()
    .eq("id", id);

  if (error) throw error;
};

import { useQuery } from "@tanstack/react-query";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockProducts } from "../../mocks/products";
import { mapProduct } from "../../shared/lib/mappers";
import type { Product } from "../../shared/types";

export const useProducts = (category?: string) => {
  const { data, isLoading, refetch } = useQuery<Product[]>({
    queryKey: ["products", category ?? "all"],
    queryFn: async () => {
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        const filtered = category
          ? mockProducts.filter((p) => p.category === category)
          : mockProducts;
        return [...filtered].sort((a, b) => b.stock - a.stock);
      }

      let query = supabaseClient
        .from(TABLES.products)
        .select("*")
        // ✅ Urutkan berdasarkan stok terbanyak terlebih dahulu
        .order("stok", { ascending: false })
        .order("nama", { ascending: true });

      if (category) {
        query = query.eq("kategori", category);
      }

      const { data, error } = await query;
      if (error) throw error;
      return (data ?? []).map(mapProduct);
    },
  });

  return {
    products: data ?? [],
    isLoading,
    refetch,
  };
};

import { useQuery } from "@tanstack/react-query";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockProducts } from "../../mocks/products";
import { mapProduct } from "../../shared/lib/mappers";
import type { Product } from "../../shared/types";

export const useProductSearch = (keyword: string, limit = 30) => {
  const sanitize = (str: string) => str.replace(/[%_()\\'"]/g, "");
  const trimmed = sanitize(keyword).trim();

  const { data, isLoading, refetch } = useQuery<Product[]>({
    queryKey: ["product-search", trimmed, limit],
    queryFn: async () => {
      if (USE_MOCK) {
        if (!trimmed) return mockProducts.slice(0, limit);
        const lower = trimmed.toLowerCase();
        return mockProducts.filter((p) => p.name.toLowerCase().includes(lower) || p.code.toLowerCase().includes(lower)).slice(0, limit);
      }

      let query = supabaseClient.from(TABLES.products).select("*").limit(limit).order("nama", { ascending: true }); 
      if (trimmed) {
        const words = trimmed.split(/\s+/).filter(Boolean);
        const orConditions = words.map((w) => `nama.ilike.%${w}%,kode.ilike.%${w}%`).join(",");
        query = query.or(orConditions);
      }

      const { data, error } = await query;
      if (error) throw error;
      return (data ?? []).map(mapProduct);
    },
    staleTime: 10_000,
  });

  return { products: data ?? [], isLoading, refetch };
};

// src/features/spb/suratJalanRules.ts
//
// Aturan bisnis Monitoring SPB yang dipakai BERSAMA oleh:
//   - useMonitoringSPB        (daftar card)
//   - useMonitoringSPBDetail  (isi tabel saat card dibuka)
// Dengan satu sumber aturan, jumlah barang di card selalu sama dengan jumlah
// baris di halaman detail.

/** Surat Jalan yang diabaikan sepenuhnya (data lama). Yang diproses: no 3 dan seterusnya. */
export const IGNORED_SURAT_JALAN: readonly string[] = ["1", "2"];

export const isIgnoredSuratJalan = (no: unknown): boolean =>
  IGNORED_SURAT_JALAN.includes(String(no ?? "").trim());

/** true jika nilai kolom (mis. id_spb / id_po) benar-benar terisi. */
export const hasValue = (val: unknown): boolean =>
  val !== null && val !== undefined && String(val).trim() !== "";

/**
 * Normalisasi nomor dokumen (no_spb / no_po) dari DB.
 * String kosong, "-", "null", "undefined" dianggap TIDAK ADA nomor -> null.
 */
export const cleanDocNo = (val: unknown): string | null => {
  if (val === null || val === undefined) return null;
  const s = String(val).trim();
  const lower = s.toLowerCase();
  if (s === "" || s === "-" || lower === "null" || lower === "undefined") return null;
  return s;
};

export type SjTarget = "spb" | "po" | "sj";

/**
 * Menentukan di card mana sebuah baris Surat Jalan ditampilkan:
 *  - punya id_spb                -> card SPB
 *  - tanpa id_spb, punya id_po   -> card PO
 *  - tanpa id_spb dan id_po      -> card Surat Jalan
 */
export const getSjTarget = (sj: { id_spb?: unknown; id_po?: unknown }): SjTarget => {
  if (hasValue(sj.id_spb)) return "spb";
  if (hasValue(sj.id_po)) return "po";
  return "sj";
};

export const updateDocumentTitle = async (
  type: "spb" | "po",
  docNo: string,
  newTitle: string,
) => {
  const trimmed = newTitle.trim();
  const overrideKey = `${type.toUpperCase()}_${docNo}`;
  saveTitleOverride(overrideKey, trimmed);

  if (type === "spb") {
    const { error } = await supabaseClient
      .from("spb")
      .update({ judul_spb: trimmed })
      .eq("no_spb", docNo);
    if (error) throw error;
  } else {
    const { error } = await supabaseClient
      .from("po")
      .update({ judul_po: trimmed })
      .eq("no_po", docNo);
    if (error) throw error;
  }
};

};

      });

      // Finalisasi Status & Sorting
      const result = Array.from(groups.values()).map((g) => {
        const isSelesai = g.data.itemsCount > 0 && g.completedItems >= g.data.itemsCount;
        let status = "Belum PO";
        if (isSelesai) status = "Selesai";
        else if (g.hasAnyArrival) status = "Datang";
        else if (g.hasAnyPO) status = "Sudah PO";

        const combinedPo = Array.from(g.poSet).filter(Boolean).join(", ");
        const firstPo = Array.from(g.poSet).filter(Boolean)[0];
        const poTitle =
          g.data.poTitle ??
          (firstPo && overrides[`PO_${firstPo}`] !== undefined
            ? overrides[`PO_${firstPo}`]
            : null);

        return {
          ...g.data,
          poNo: combinedPo || null,
          poTitle,
          status,
          sortNumber: getSortNumber(g.data.spbNo) + getSortNumber(combinedPo) * 0.01,
          itemNames: Array.from(g.itemNames),
        };
      });

      // ✅ URUTAN: Sudah PO & Belum PO (1) -> Datang (2) -> Selesai (3)
      const getStatusWeight = (status: string) => {
        if (status === "Sudah PO") return 1;
        if (status === "Belum PO") return 1;
        if (status === "Datang") return 2;
        if (status === "Selesai") return 3;
        return 4;
      };

      return result.sort((a, b) => {
        const weightA = getStatusWeight(a.status || "");
        const weightB = getStatusWeight(b.status || "");
        if (weightA !== weightB) return weightA - weightB;

        const dateA = a.date !== "-" ? new Date(a.date).getTime() : 0;
        const dateB = b.date !== "-" ? new Date(b.date).getTime() : 0;
        if (dateA === dateB) return a.sortNumber - b.sortNumber;
        return dateB - dateA;
      });
    },
  });

  return { data: data ?? [], isLoading, error: error as Error | null, refetch };
};

// src/features/spb/useMonitoringSPBDetail.ts
import { useQuery } from "@tanstack/react-query";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import type { SPBItemDetail } from "../../shared/types";
import { cleanDocNo, hasValue, isIgnoredSuratJalan } from "./suratJalanRules";

/**
 * Isi tabel detail sebuah card:
 *
 *  Card SPB          -> baris tabel `spb` (no_spb = ?)
 *                       + baris `surat_jalan` (no >= 3) yang id_spb-nya milik SPB itu
 *  Card PO           -> baris tabel `po` (no_po = ?) yang TIDAK terhubung ke SPB
 *                       + baris `surat_jalan` (no >= 3) yang id_po-nya milik PO itu, id_spb kosong
 *  Card Surat Jalan  -> baris `surat_jalan` (no_surat_jalan = ?) yang id_spb DAN id_po kosong
 *
 * Surat Jalan no 1 dan 2 selalu diabaikan.
 * Aturannya sama dengan pengelompokan di useMonitoringSPB, jadi jumlah barang
 * di card = jumlah baris di tabel ini.
 */

export type ItemSource = "spb" | "po" | "sj";

export type MonitoringDetailItem = SPBItemDetail & {
  /** Unik lintas tabel (id SPB / PO / SJ bisa sama), dipakai sebagai rowKey. */
  key: string;
  /** Tabel asal baris ini; dipakai untuk navigasi ke halaman rincian barang. */
  sumber: ItemSource;
  no_surat_jalan?: string | null;
};

type Row = Record<string, any>;

const SPB_SELECT = `*, po:po!id_po (no_po, tanggal_po)`;
const SJ_SELECT = `*, spb:spb!id_spb (no_spb, judul_spb), po:po!id_po (no_po)`;

// ───────────────────────── helper kecil ─────────────────────────

const first = (v: unknown): Row | null => {
  const x = Array.isArray(v) ? v[0] : v;
  return x && typeof x === "object" ? (x as Row) : null;
};

const numOrZero = (v: unknown): number => {
  const n = Number(v);
  return Number.isFinite(n) ? n : 0;
};

const sortById = (rows: Row[]): Row[] =>
  [...rows].sort((a, b) => numOrZero(a.id) - numOrZero(b.id));

const unique = <T,>(arr: T[]): T[] => Array.from(new Set(arr));

const unwrap = (res: { data: unknown; error: unknown }): Row[] => {
  if (res.error) throw res.error;
  return (res.data as Row[] | null) ?? [];
};

const spbStatus = (req: number, arr: number, tglDatang: string | null, hasPo: boolean) => {
  if (tglDatang) {
    if (req === arr) return "Datang Sesuai";
    if (req < arr) return "Datang Lebih";
    return "Datang Kurang";
  }
  return hasPo ? "Sudah PO" : "Belum PO";
};

// ───────────────────────── mapper baris -> item ─────────────────────────

const mapSpbRow = (row: Row, fallbackPo: Row | null): MonitoringDetailItem => {
  const joinedPo = first(row.po);
  const poNo = cleanDocNo(joinedPo?.no_po) ?? cleanDocNo(fallbackPo?.no_po);
  const req = numOrZero(row.qty_permintaan);
  const arr = numOrZero(row.material_datang_qty);
  const tgl: string | null = row.material_datang_tgl || null;

  return {
    key: `spb-${row.id}`,
    sumber: "spb",
    id: row.id,
    no_spb: cleanDocNo(row.no_spb) ?? "-",
    judul_spb: row.judul_spb ?? null,
    tanggal_spb: row.tgl_spb ?? null,
    nama_barang: row.nama_barang_spb || "-",
    ukuran: row.ukuran ?? null,
    spesifikasi_teknis: row.spesifikasi_teknis ?? null,
    jumlah_permintaan: req,
    satuan: row.satuan ?? null,
    no_po: poNo ?? "-",
    tanggal_po: joinedPo?.tanggal_po ?? fallbackPo?.tanggal_po ?? null,
    tanggal_datang: tgl,
    qty_datang: arr,
    status: spbStatus(req, arr, tgl, !!poNo),
  };
};

const mapPoRow = (row: Row): MonitoringDetailItem => {
  const tgl: string | null = row.material_datang_tgl || null;
  return {
    key: `po-${row.id}`,
    sumber: "po",
    id: row.id,
    no_spb: "-",
    nama_barang: row.nama_barang_po || "-",
    jumlah_permintaan: null,
    satuan: row.material_datang_satuan ?? row.satuan ?? null,
    no_po: cleanDocNo(row.no_po) ?? "-",
    tanggal_po: row.tanggal_po ?? null,
    tanggal_datang: tgl,
    qty_datang: numOrZero(row.material_datang_qty),
    status: tgl ? "Datang" : "Sudah PO",
  };
};

const mapSjRow = (row: Row): MonitoringDetailItem => {
  const spb = first(row.spb);
  const po = first(row.po);
  const tgl: string | null = row.material_datang_tgl || null;
  return {
    key: `sj-${row.id}`,
    sumber: "sj",
    id: row.id,
    no_surat_jalan: row.no_surat_jalan != null ? String(row.no_surat_jalan) : null,
    no_spb: cleanDocNo(spb?.no_spb) ?? "-",
    judul_spb: spb?.judul_spb ?? null,
    nama_barang: row.nama_barang || "-",
    jumlah_permintaan: null,
    satuan: row.material_datang_satuan ?? row.satuan ?? null,
    no_po: cleanDocNo(po?.no_po) ?? "-",
    tanggal_po: null,
    tanggal_datang: tgl,
    qty_datang: numOrZero(row.material_datang_qty),
    status: tgl ? "Datang" : "Belum Datang",
  };
};

// ───────────────────────── skenario fetch ─────────────────────────

/** Card SPB */
const fetchBySpb = async (spbNo: string): Promise<MonitoringDetailItem[]> => {
  const spbRows = sortById(
    unwrap(await supabaseClient.from("spb").select(SPB_SELECT).eq("no_spb", spbNo)),
  );
  if (spbRows.length === 0) return [];

  const spbIds = spbRows.map((r) => r.id);

  const [poRes, sjRes] = await Promise.all([
    // PO yang menunjuk SPB lewat po.id_spb (cadangan jika spb.id_po kosong)
    supabaseClient.from("po").select("id, no_po, tanggal_po, id_spb").in("id_spb", spbIds),
    supabaseClient.from("surat_jalan").select(SJ_SELECT).in("id_spb", spbIds),
  ]);

  const poBySpbId = new Map<string, Row>();
  unwrap(poRes).forEach((p) => {
    const key = String(p.id_spb);
    if (cleanDocNo(p.no_po) && !poBySpbId.has(key)) poBySpbId.set(key, p);
  });

  const spbItems = spbRows.map((r) => mapSpbRow(r, poBySpbId.get(String(r.id)) ?? null));
  const sjItems = sortById(unwrap(sjRes))
    .filter((sj) => !isIgnoredSuratJalan(sj.no_surat_jalan))
    .map(mapSjRow);

  return [...spbItems, ...sjItems];
};

/** Card PO (tanpa SPB) */
const fetchByPo = async (poNo: string): Promise<MonitoringDetailItem[]> => {
  const poRows = sortById(
    unwrap(await supabaseClient.from("po").select("*").eq("no_po", poNo)),
  );
  if (poRows.length === 0) return [];

  const poIds = poRows.map((p) => p.id);
  const parentSpbIds = unique(poRows.map((p) => p.id_spb).filter(hasValue));

  const [parentSpbRes, spbPointingRes, sjRes] = await Promise.all([
    parentSpbIds.length > 0
      ? supabaseClient.from("spb").select("id").in("id", parentSpbIds)
      : Promise.resolve({ data: [], error: null }),
    supabaseClient.from("spb").select("id, id_po").in("id_po", poIds),
    // Surat Jalan yang punya id_po tapi TIDAK punya id_spb
    supabaseClient.from("surat_jalan").select(SJ_SELECT).in("id_po", poIds).is("id_spb", null),
  ]);

  const existingParentSpb = new Set(unwrap(parentSpbRes).map((s) => String(s.id)));
  const poIdsUsedBySpb = new Set(unwrap(spbPointingRes).map((s) => String(s.id_po)));

  // PO yang sudah terhubung ke SPB tampil di card SPB, bukan di sini.
  const isLinkedToSpb = (p: Row) =>
    (hasValue(p.id_spb) && existingParentSpb.has(String(p.id_spb))) ||
    poIdsUsedBySpb.has(String(p.id));

  const poItems = poRows.filter((p) => !isLinkedToSpb(p)).map(mapPoRow);
  const sjItems = sortById(unwrap(sjRes))
    .filter((sj) => !isIgnoredSuratJalan(sj.no_surat_jalan))
    .map(mapSjRow);

  return [...poItems, ...sjItems];
};

/** Card Surat Jalan murni (tanpa id_spb dan id_po) */
const fetchBySuratJalan = async (sjNo: string): Promise<MonitoringDetailItem[]> => {
  if (isIgnoredSuratJalan(sjNo)) return [];

  const rows = unwrap(
    await supabaseClient
      .from("surat_jalan")
      .select(SJ_SELECT)
      .eq("no_surat_jalan", sjNo)
      .is("id_spb", null)
      .is("id_po", null),
  );

  return sortById(rows).map(mapSjRow);
};

// ───────────────────────── hook ─────────────────────────

export const useMonitoringSPBDetail = (
  spbNo: string | null,
  poNo: string | null,
  sjNo: string | null = null,
) => {
  const enabled = !!(spbNo || poNo || sjNo);

  const { data, isLoading, error } = useQuery<MonitoringDetailItem[]>({
    queryKey: ["monitoring-spb-detail", spbNo, poNo, sjNo],
    queryFn: async () => {
      // Urutan prioritas sama dengan navigasi di MonitoringSPBPage: SPB -> PO -> SJ
      if (spbNo) return fetchBySpb(spbNo);
      if (poNo) return fetchByPo(poNo);
      if (sjNo) return fetchBySuratJalan(sjNo);
      return [];
    },
    enabled,
    retry: false,
    refetchOnWindowFocus: false,
  });

  return {
    items: data ?? [],
    isLoading: enabled ? isLoading : false,
    error: error as Error | null,
  };
};

    if (step === "workplace") {
      setWorkplace(null);
      setStep("person");
    }
  }, [step]);

  const selectItem = useCallback((item: Product) => {
    setActiveItem(item);
    setActiveQty(1);
    setPhase("quantity");
  }, []);

  const confirmQuantity = useCallback(() => {
    if (!activeItem) return;
    setLines((prev) => {
      const idx = prev.findIndex((l) => l.itemId === activeItem.id);
      if (idx >= 0) {
        const copy = [...prev];
        copy[idx] = { ...copy[idx], quantity: copy[idx].quantity + activeQty };
        return copy;
      }
      return [
        ...prev,
        {
          itemId: activeItem.id,
          itemName: activeItem.name,
          unit: activeItem.unit,
          quantity: activeQty,
        },
      ];
    });
    setActiveItem(null);
    setActiveQty(1);
    setPhase("search");
  }, [activeItem, activeQty]);

  const cancelQuantity = useCallback(() => {
    setActiveItem(null);
    setActiveQty(1);
    setPhase("search");
  }, []);

  const removeLine = useCallback((itemId: number) => {
    setLines((prev) => prev.filter((l) => l.itemId !== itemId));
  }, []);

  const increaseQty = useCallback(() => setActiveQty((q) => q + 1), []);
  const decreaseQty = useCallback(
    () => setActiveQty((q) => Math.max(1, q - 1)),
    [],
  );

    step,
    person,
    workplace,
    currentWizardData,
    selectWizardValue,
    backWizard,
    wizardOpen,
    openWizard,
    closeWizard,
    phase,

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../../shared/lib/supabaseClient";
import { TABLES } from "../../shared/lib/constants";
import { USE_MOCK } from "../../shared/lib/config";
import { mockStockExits } from "../../mocks/stockExitList";
import { mapStockExits } from "../../shared/lib/mappers";
import type { StockExit } from "../../shared/types";

interface Options { onlyToday?: boolean; breakdown?: boolean; limit?: number; }

export const useStockExits = ({ onlyToday = false, breakdown, limit = 50 }: Options = {}) => {
  const { data, isLoading, refetch } = useQuery<StockExit[]>({
    queryKey: ["stock-exits", onlyToday, breakdown, limit],
    queryFn: async () => {
      if (USE_MOCK) return mockStockExits;
      let query = supabaseClient.from(TABLES.stockExits).select("*").order("created_at", { ascending: false });

      if (onlyToday) {
        const start = dayjs().startOf("day").toISOString();
        const end = dayjs().endOf("day").toISOString();
        query = query.gte("created_at", start).lte("created_at", end);
      }
      if (breakdown !== undefined) query = query.eq("is_breakdown", breakdown);
      
      // ✅ PERBAIKAN: Batasi data history agar tabel tidak hang
      query = query.limit(limit);

      const { data, error } = await query;
      if (error) throw error;
      return mapStockExits(data ?? []);
    },
    refetchInterval: USE_MOCK ? false : 30_000,
  });
  return { exits: data ?? [], isLoading, refetch };
};

}

export class ErrorBoundary extends Component<Props, State> {
  public state: State = { hasError: false, error: null };

  public static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };

import { useState, useEffect } from "react";

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

import { useEffect } from "react";

interface UseKeyboardShortcutOptions {
  key: string;
  ctrlKey?: boolean;
  shiftKey?: boolean;
  altKey?: boolean;
  metaKey?: boolean;
  enabled?: boolean;
  preventDefault?: boolean;
  ignoreInputs?: boolean;
  handler: (event: KeyboardEvent) => void;
}

const isEditableElement = (el: EventTarget | null): boolean => {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName.toLowerCase();
  return (
    tag === "input" ||
    tag === "textarea" ||
    tag === "select" ||
    el.isContentEditable
  );
};

export const useKeyboardShortcut = ({
  key,
  ctrlKey = false,
  shiftKey = false,
  altKey = false,
  metaKey = false,
  enabled = true,
  preventDefault = true,
  ignoreInputs = false,
  handler,
}: UseKeyboardShortcutOptions) => {
  useEffect(() => {
    if (!enabled) return;

    const onKeyDown = (event: KeyboardEvent) => {
      // Normalisasi nama key (mengabaikan case untuk huruf 'm' dan 'M')
      const isKeyMatch = event.key.toLowerCase() === key.toLowerCase() || event.key === key;
      
      if (!isKeyMatch) return;
      if (event.ctrlKey !== ctrlKey) return;
      if (event.shiftKey !== shiftKey) return;
      if (event.altKey !== altKey) return;
      if (event.metaKey !== metaKey) return;
      if (ignoreInputs && isEditableElement(event.target)) return;

      if (preventDefault) event.preventDefault();
      handler(event);
    };

    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  }, [
    key,
    ctrlKey,
    shiftKey,
    altKey,
    metaKey,
    enabled,
    preventDefault,
    ignoreInputs,
    handler,
  ]);
};

  id: number; kode: string; nama: string; ukuran: string | null; spesifikasi: string | null;
  kategori: string; satuan: string; stok: number; created_at: string;

  return {
    id: r.id, code: r.kode, name: r.nama, size: r.ukuran ?? null, specification: r.spesifikasi ?? null,
    category: r.kategori ?? "Material", categoryId: null, categoryName: null,
    unit: r.satuan ?? "pcs", stock: r.stok ?? 0, minStock: 0, costPrice: 0,
    status: undefined, createdAt: r.created_at ?? new Date().toISOString(),
  };

interface StockExitRow {
  id: number; nomor_transaksi: string | null; barang_id: number; nama_pekerja: string | null;
  nama_barang: string; ukuran_barang: string | null; satuan_barang: string | null;
  jumlah_barang_keluar: number; tujuan: string | null; is_breakdown: boolean | null; created_at: string;
}

export const mapStockExits = (rows: any[]): StockExit[] => {
  const typedRows = rows as StockExitRow[];
  const grouped = new Map<string, StockExit>();

  for (const row of typedRows) {
    const key = row.nomor_transaksi ?? `temp-${row.id}`;
    if (!grouped.has(key)) {
      const created = new Date(row.created_at);
      grouped.set(key, {
        id: row.id, nomorTransaksi: key, timestamp: row.created_at, date: created.toISOString().split("T")[0],
        time: created.toISOString().slice(11, 16), person: row.nama_pekerja ?? "-",
        destination: row.tujuan, isBreakdown: row.is_breakdown ?? true, items: [],
      });
    }
    grouped.get(key)!.items.push({
      itemId: row.barang_id, itemName: row.nama_barang, size: row.ukuran_barang,
      unit: row.satuan_barang ?? "pcs", quantity: row.jumlah_barang_keluar,
    });
  }
  return Array.from(grouped.values());
};
