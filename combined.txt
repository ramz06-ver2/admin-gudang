import type {
  AuthProvider,
  AuthActionResponse,
  CheckResponse,
} from "@refinedev/core";
import { supabaseClient } from "./utility/supabaseClient";
import { USE_MOCK } from "./utility/config";

const MOCK_USER = {
  email: "admin@admin.com",
  password: "admin123",
  name: "Administrator",
};

const STORAGE_KEY = "mock-auth-user";

const getMockUser = () => {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
};

export const authProvider: AuthProvider = {
  login: async ({ email, password }): Promise<AuthActionResponse> => {
    if (USE_MOCK) {
      await new Promise((r) => setTimeout(r, 400));
      if (email === MOCK_USER.email && password === MOCK_USER.password) {
        localStorage.setItem(
          STORAGE_KEY,
          JSON.stringify({ email, name: MOCK_USER.name, role: "admin" }),
        );
        return { success: true, redirectTo: "/dashboard" };
      }
      return {
        success: false,
        error: {
          name: "LoginError",
          message: "Email atau password salah",
        },
      };
    }

    const { data, error } = await supabaseClient.auth.signInWithPassword({
      email,
      password,
    });

    if (error) {
      return {
        success: false,
        error: { name: "LoginError", message: error.message },
      };
    }
    if (!data?.user) {
      return {
        success: false,
        error: { name: "LoginError", message: "User tidak ditemukan" },
      };
    }
    return { success: true, redirectTo: "/dashboard" };
  },

  logout: async (): Promise<AuthActionResponse> => {
    if (USE_MOCK) {
      localStorage.removeItem(STORAGE_KEY);
      return { success: true, redirectTo: "/dashboard" };
    }
    await supabaseClient.auth.signOut();
    return { success: true, redirectTo: "/dashboard" };
  },

  // ✅ Selalu izinkan akses agar User (Public) bisa melihat stok tanpa harus login
  check: async (): Promise<CheckResponse> => {
    return { authenticated: true };
  },

  getIdentity: async () => {
    if (USE_MOCK) {
      const user = getMockUser();
      if (user) {
        return {
          id: "admin-1",
          name: user.name,
          email: user.email,
          role: "admin",
        };
      }
      return {
        id: "public-user",
        name: "User (Public)",
        email: "",
        role: "user",
      };
    }

    const { data } = await supabaseClient.auth.getUser();
    if (data?.user) {
      return {
        id: data.user.id,
        name: data.user.email?.split("@")[0] ?? "Admin",
        email: data.user.email,
        role: "admin",
      };
    }

    return {
      id: "public-user",
      name: "User (Public)",
      email: "",
      role: "user",
    };
  },

  getPermissions: async () => {
    if (USE_MOCK) {
      const user = getMockUser();
      return user ? ["admin"] : ["user"];
    }
    const { data } = await supabaseClient.auth.getSession();
    return data.session ? ["admin"] : ["user"];
  },

  onError: async (error) => {
    return { error };
  },
};

import { useState, useRef, useEffect, useCallback } from "react";
import { useNavigate } from "react-router-dom";
import { Input, Modal, Spin, Empty, Tag, Typography } from "antd";
import { SearchOutlined } from "@ant-design/icons";
import { useKeyboardShortcut } from "../../hooks/useKeyboardShortcut";
import { useProductSearch } from "../../hooks/useProductSearch";
import type { Product } from "../../types";

const { Text } = Typography;

export const GlobalSearch = () => {
  const navigate = useNavigate();
  const [open, setOpen] = useState(false);
  const [keyword, setKeyword] = useState("");
  const [activeIndex, setActiveIndex] = useState(0);
  const inputRef = useRef<HTMLInputElement>(null);

  const { products, isLoading } = useProductSearch(keyword);

  // Reset state saat modal terbuka
  useEffect(() => {
    if (open) {
      setKeyword("");
      setActiveIndex(0);
      setTimeout(() => inputRef.current?.focus(), 50);
    }
  }, [open]);

  // Clamp active index saat hasil berubah
  useEffect(() => {
    if (activeIndex >= products.length) setActiveIndex(0);
  }, [products.length, activeIndex]);

  const handleSelect = useCallback(
    (product: Product) => {
      setOpen(false);
      navigate(`/stock-entries/create?productId=${product.id}`);
    },
    [navigate],
  );

  // ⌨️ Space → buka palette (kecuali sedang mengetik di input)
  useKeyboardShortcut({
    key: " ",
    ignoreInputs: true,
    enabled: !open,
    handler: () => setOpen(true),
  });

  // ⌨️ Ctrl+K → buka palette juga (opsional, umum dipakai)
  useKeyboardShortcut({
    key: "k",
    ctrlKey: true,
    enabled: !open,
    handler: () => setOpen(true),
  });

  const handleKeyDown = (event: React.KeyboardEvent) => {
    if (event.key === "ArrowDown") {
      event.preventDefault();
      setActiveIndex((i) => Math.min(i + 1, products.length - 1));
    } else if (event.key === "ArrowUp") {
      event.preventDefault();
      setActiveIndex((i) => Math.max(i - 1, 0));
    } else if (event.key === "Enter") {
      event.preventDefault();
      const selected = products[activeIndex];
      if (selected) handleSelect(selected);
    } else if (event.key === "Escape") {
      event.preventDefault();
      setOpen(false);
    }
  };

  return (
    <Modal
      open={open}
      onCancel={() => setOpen(false)}
      footer={null}
      closable={false}
      width={560}
      destroyOnClose
      styles={{ body: { padding: 0 } }}
      style={{ top: 80 }}
    >
      <div style={{ padding: 12, borderBottom: "1px solid #f0f0f0" }}>
        <Input
          ref={inputRef as never}
          size="large"
          prefix={<SearchOutlined />}
          placeholder="Cari produk berdasarkan nama atau kode..."
          value={keyword}
          onChange={(e) => setKeyword(e.target.value)}
          onKeyDown={handleKeyDown}
          variant="borderless"
          autoFocus
        />
      </div>

      <div style={{ maxHeight: 360, overflowY: "auto", padding: 8 }}>
        {isLoading && (
          <div style={{ textAlign: "center", padding: 24 }}>
            <Spin />
          </div>
        )}

        {!isLoading && products.length === 0 && (
          <Empty
            description="Produk tidak ditemukan"
            image={Empty.PRESENTED_IMAGE_SIMPLE}
            style={{ padding: 24 }}
          />
        )}

        {!isLoading &&
          products.map((product, index) => (
            <div
              key={product.id}
              onMouseEnter={() => setActiveIndex(index)}
              onClick={() => handleSelect(product)}
              style={{
                padding: "10px 12px",
                borderRadius: 8,
                cursor: "pointer",
                display: "flex",
                alignItems: "center",
                justifyContent: "space-between",
                gap: 12,
                background:
                  index === activeIndex ? "#e6f4ff" : "transparent",
              }}
            >
              <div style={{ minWidth: 0, flex: 1 }}>
                <div style={{ fontWeight: 500, fontSize: 14 }}>
                  {product.name}
                </div>
                <Text type="secondary" style={{ fontSize: 12 }}>
                  {product.code} · Stok: {product.stock} {product.unit}
                </Text>
              </div>
              <Tag color="blue" style={{ margin: 0 }}>
                Input
              </Tag>
            </div>
          ))}
      </div>

      <div
        style={{
          padding: "8px 16px",
          borderTop: "1px solid #f0f0f0",
          display: "flex",
          gap: 16,
          fontSize: 12,
          color: "#8c8c8c",
        }}
      >
        <span>↑↓ navigasi</span>
        <span>↵ pilih</span>
        <span>esc tutup</span>
      </div>
    </Modal>
  );
};

import { useEffect, useRef, useState } from "react";
import {
  AutoComplete,
  Button,
  Card,
  Empty,
  Input,
  InputNumber,
  List,
  Space,
  Tag,
  Typography,
} from "antd";
import {
  CheckCircleOutlined,
  CloseCircleOutlined,
  EnvironmentOutlined,
  SearchOutlined,
  TeamOutlined,
  UserOutlined,
} from "@ant-design/icons";
import { useProductSearch } from "../../hooks/useProductSearch";
import { formatNumber } from "../../utility/format";
import type { Person, Team, Workplace, EntryLine } from "../../types/entry";
import type { Product } from "../../types";

const { Title, Text } = Typography;

interface Props {
  team: Team;
  person: Person;
  workplace: Workplace;
  phase: "search" | "quantity";
  activeItem: Product | null;
  activeQty: number;
  lines: EntryLine[];
  onSelectItem: (item: Product) => void;
  onConfirmQuantity: () => void;
  onCancelQuantity: () => void;
  onIncreaseQty: () => void;
  onDecreaseQty: () => void;
  onRemoveLine: (itemId: number) => void;
  onFinish: () => void;
}

export const EntryBoard = ({
  team,
  person,
  workplace,
  phase,
  activeItem,
  activeQty,
  lines,
  onSelectItem,
  onConfirmQuantity,
  onCancelQuantity,
  onIncreaseQty,
  onDecreaseQty,
  onRemoveLine,
  onFinish,
}: Props) => {
  // ==========================================
  // Local state
  // ==========================================
  const [keyword, setKeyword] = useState("");
  const [highlightedIdx, setHighlightedIdx] = useState(0);
  const searchRef = useRef<HTMLInputElement>(null);
  const qtyRef = useRef<HTMLInputElement>(null);

  const { products } = useProductSearch(keyword, 8);

  // ==========================================
  // Auto-focus sesuai phase
  // ==========================================
  useEffect(() => {
    if (phase === "search") {
      setTimeout(() => searchRef.current?.focus(), 100);
    } else if (phase === "quantity") {
      setTimeout(() => qtyRef.current?.focus(), 100);
    }
  }, [phase]);

  // ==========================================
  // Reset highlight saat keyword berubah
  // ==========================================
  useEffect(() => {
    setHighlightedIdx(0);
  }, [keyword]);

  // ==========================================
  // Ctrl+Enter → selesai
  // ==========================================
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (e.ctrlKey && e.key === "Enter") {
        e.preventDefault();
        if (lines.length > 0) onFinish();
      }
    };
    window.addEventListener("keydown", handler);
    return () => window.removeEventListener("keydown", handler);
  }, [lines.length, onFinish]);

  // ==========================================
  // Handlers — search phase
  // ==========================================
  const commitItem = () => {
    if (products.length === 0) return;
    const chosen = products[highlightedIdx] ?? products[0];
    onSelectItem(chosen);
    setKeyword("");
  };

  const handleSearchKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === "ArrowDown") {
      e.preventDefault();
      setHighlightedIdx((i) => Math.min(i + 1, products.length - 1));
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      setHighlightedIdx((i) => Math.max(i - 1, 0));
    } else if (e.key === "Enter") {
      e.preventDefault();
      commitItem();
    }
  };

  // ==========================================
  // Handlers — quantity phase
  // ==========================================
  const handleQtyKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === "Enter") {
      e.preventDefault();
      onConfirmQuantity();
    } else if (e.key === "Escape") {
      e.preventDefault();
      onCancelQuantity();
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      onIncreaseQty();
    } else if (e.key === "ArrowDown") {
      e.preventDefault();
      onDecreaseQty();
    }
  };

  // ==========================================
  // Derived
  // ==========================================
  const totalItems = lines.reduce((sum, l) => sum + l.quantity, 0);

  return (
    <div style={{ maxWidth: 900, margin: "0 auto" }}>
      {/* ====================================== */}
      {/* Header card — workplace + team + person */}
      {/* ====================================== */}
      <Card variant="borderless" style={{ borderRadius: 12, marginBottom: 16 }}>
        <Space direction="vertical" size={4} style={{ width: "100%" }}>
          <Space size={8}>
            <EnvironmentOutlined style={{ color: "#1677ff", fontSize: 20 }} />
            <Title level={3} style={{ margin: 0 }}>
              {workplace.name}
            </Title>
          </Space>
          <Space size={12} wrap>
            <Text type="secondary">
              <EnvironmentOutlined /> {workplace.name}
            </Text>
            <Text type="secondary">
              <TeamOutlined /> {team.name}
            </Text>
            <Text type="secondary">
              <UserOutlined /> {person.name}
            </Text>
          </Space>
        </Space>
      </Card>

      {/* ====================================== */}
      {/* Input card — search atau quantity       */}
      {/* ====================================== */}
      <Card
        variant="borderless"
        style={{ borderRadius: 12, marginBottom: 16 }}
        styles={{ body: { padding: 24 } }}
      >
        {phase === "search" && (
          <>
            <Text strong style={{ fontSize: 14 }}>
              Nama Barang
            </Text>
            <AutoComplete
              value={keyword}
              onChange={setKeyword}
              options={products.map((p, idx) => ({
                value: p.name,
                label: (
                  <div
                    style={{
                      display: "flex",
                      justifyContent: "space-between",
                      padding: "2px 0",
                      background:
                        idx === highlightedIdx ? "#e6f4ff" : "transparent",
                    }}
                  >
                    <span>
                      {p.name}{" "}
                      <Text type="secondary" style={{ fontSize: 12 }}>
                        · {p.code}
                      </Text>
                    </span>
                    <Text type="secondary" style={{ fontSize: 12 }}>
                      stok: {p.stock} {p.unit}
                    </Text>
                  </div>
                ),
                data: p,
              }))}
              onSelect={(_, opt) => {
                onSelectItem((opt as { data: Product }).data);
                setKeyword("");
              }}
              style={{ width: "100%", marginTop: 8 }}
            >
              <Input
                ref={searchRef as never}
                size="large"
                prefix={<SearchOutlined />}
                placeholder="Ketik nama barang, lalu Enter..."
                onKeyDown={handleSearchKeyDown}
                autoFocus
              />
            </AutoComplete>

            <Space style={{ marginTop: 12 }} size="small" wrap>
              <Tag>↑↓</Tag>
              <Text type="secondary" style={{ fontSize: 12 }}>
                navigasi
              </Text>
              <Tag>Enter</Tag>
              <Text type="secondary" style={{ fontSize: 12 }}>
                pilih & lanjut ke jumlah
              </Text>
              <Tag>Ctrl+Enter</Tag>
              <Text type="secondary" style={{ fontSize: 12 }}>
                selesai
              </Text>
            </Space>
          </>
        )}

        {phase === "quantity" && activeItem && (
          <>
            <Space
              style={{
                marginBottom: 12,
                width: "100%",
                justifyContent: "space-between",
              }}
            >
              <Text strong style={{ fontSize: 14 }}>
                Jumlah — {activeItem.name}
              </Text>
              <Tag color="blue">
                Stok: {activeItem.stock} {activeItem.unit}
              </Tag>
            </Space>

            <Space.Compact style={{ width: "100%" }}>
              <Button
                size="large"
                onClick={onDecreaseQty}
                style={{ width: 60, height: 48, fontSize: 20 }}
              >
                −
              </Button>

              <InputNumber
                ref={qtyRef as never}
                value={activeQty}
                size="large"
                min={1}
                max={activeItem.stock}
                controls={false}
                onKeyDown={handleQtyKeyDown}
                readOnly
                style={{
                  flex: 1,
                  textAlign: "center",
                  fontSize: 20,
                  fontWeight: 600,
                }}
              />

              <Button
                size="large"
                onClick={onIncreaseQty}
                style={{ width: 60, height: 48, fontSize: 20 }}
              >
                +
              </Button>
            </Space.Compact>

            <div style={{ marginTop: 8, textAlign: "right" }}>
              <Text type="secondary">
                {activeQty} {activeItem.unit}
              </Text>
            </div>

            <Space style={{ marginTop: 12 }} size="small" wrap>
              <Tag>↑↓ / + −</Tag>
              <Text type="secondary" style={{ fontSize: 12 }}>
                ubah jumlah
              </Text>
              <Tag>Enter</Tag>
              <Text type="secondary" style={{ fontSize: 12 }}>
                tambah & lanjut
              </Text>
              <Tag>Esc</Tag>
              <Text type="secondary" style={{ fontSize: 12 }}>
                batal
              </Text>
            </Space>
          </>
        )}
      </Card>

      {/* ====================================== */}
      {/* Lines list — barang yang sudah di-input */}
      {/* ====================================== */}
      <Card
        title={`Daftar Barang (${lines.length} jenis · ${formatNumber(totalItems)} total)`}
        variant="borderless"
        style={{ borderRadius: 12 }}

          <Button

            Selesai (Ctrl+Enter)

      >
        {lines.length === 0 ? (

                <List.Item.Meta

  );
};

        size="middle"
        scroll={{ x: 800 }}
        columns={type === "stock" ? stockColumns : outboundColumns}
      />
    </Card>
  );
};

// ============================================

export interface UseDashboardStatsResult {
  stats: DashboardStats;
  isLoading: boolean;
  refetch: () => void;
}

const EMPTY_STATS: DashboardStats = {
  totalProducts: 0,
  totalStock: 0,
  todayInbound: 0,
  todayOutbound: 0,
};

export const useDashboardStats = (): UseDashboardStatsResult => {
  const query = useQuery<DashboardStats, Error, DashboardStats>({
    queryKey: ["dashboard-stats"],
    refetchInterval: USE_MOCK ? false : 60_000,
    queryFn: async (): Promise<DashboardStats> => {
      // === MODE MOCK ===
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return getMockDashboardStats();
      }

      // === MODE SUPABASE ===
      const today = new Date().toISOString().split("T")[0];

      const [productsRes, inboundRes, outboundRes] = await Promise.all([
        supabaseClient.from(TABLES.products).select("stok"),
        supabaseClient
          .from(TABLES.stockEntries)
          .select("jumlah")
          .eq("tanggal", today),
        supabaseClient
          .from(TABLES.stockExits)
          .select("items")
          .eq("date", today),
      ]);

      if (productsRes.error) throw productsRes.error;
      if (inboundRes.error) throw inboundRes.error;
      if (outboundRes.error) throw outboundRes.error;

      const totalProducts = productsRes.data?.length ?? 0;
      const totalStock =
        productsRes.data?.reduce(
          (sum, p: { stok: number | null }) => sum + (p.stok ?? 0),
          0,
        ) ?? 0;
      const todayInbound =
        inboundRes.data?.reduce(
          (sum, e: { jumlah: number | null }) => sum + (e.jumlah ?? 0),
          0,
        ) ?? 0;

      // Hitung total quantity dari items[] di stock exits
      const todayOutbound =
        outboundRes.data?.reduce(
          (sum, exit: { items: Array<{ quantity: number }> }) =>
            sum + (exit.items?.reduce((s, i) => s + (i.quantity ?? 0), 0) ?? 0),
          0,
        ) ?? 0;

      return { totalProducts, totalStock, todayInbound, todayOutbound };
    },
  });

  return {
    stats: query.data ?? EMPTY_STATS,
    isLoading: query.isLoading,
    refetch: query.refetch,
  };
};

export const useEntryFlow = () => {

  const [workplace, setWorkplace] = useState<Workplace | null>(null);
  const [wizardOpen, setWizardOpen] = useState(false);

  const [phase, setPhase] = useState<EntryPhase>("search");
  const [activeItem, setActiveItem] = useState<Product | null>(null);
  const [activeQty, setActiveQty] = useState(1);
  const [lines, setLines] = useState<EntryLine[]>([]);

  // ✅ Ambil data pekerja langsung dari Database
  const { persons } = usePersons();

  const wizardData = {
    person: {
      value: person,
      options: persons, // Pakai data dari DB
      placeholder: "Ketik atau pilih nama pekerja...",
      title: "Pilih Nama Pekerja",
    },
    workplace: {

      title: "Pilih Lokasi Pekerjaan",
    },
  } as const;

  const currentWizardData =
    step === "done"
      ? { value: null, title: "", placeholder: "", options: [] }
      : wizardData[step as "person" | "workplace"];

  const openWizard = useCallback(() => {
    setStep("person");
    setWizardOpen(true);
  }, []);

  const closeWizard = useCallback(() => setWizardOpen(false), []);

  const selectWizardValue = useCallback(
    (value: Person | Workplace) => {
      if (step === "person") {
        setPerson(value as Person);
        setWorkplace(null);
        setStep("workplace");
      } else if (step === "workplace") {
        setWorkplace(value as Workplace);
        setStep("done");
        setWizardOpen(false);
      }
    },
    [step],
  );

  const resetSession = useCallback(() => {
    setStep("person");
    setPerson(null);
    setWorkplace(null);
    setPhase("search");
    setActiveItem(null);
    setActiveQty(1);
    setLines([]);
    setWizardOpen(false);
  }, []);

  const backWizard = useCallback(() => {
    if (step === "workplace") {
      setWorkplace(null);
      setStep("person");
    }
  }, [step]);

  const selectItem = useCallback((item: Product) => {
    setActiveItem(item);
    setActiveQty(1);
    setPhase("quantity");

    if (!activeItem) return;

        const copy = [...prev];

  }, [activeItem, activeQty]);

  const cancelQuantity = useCallback(() => {
    setActiveItem(null);
    setActiveQty(1);
    setPhase("search");
  }, []);

  const removeLine = useCallback((itemId: number) => {
    setLines((prev) => prev.filter((l) => l.itemId !== itemId));
  }, []);

  const increaseQty = useCallback(() => setActiveQty((q) => q + 1), []);
  const decreaseQty = useCallback(
    () => setActiveQty((q) => Math.max(1, q - 1)),
    [],
  );

    step,
    person,
    workplace,
    currentWizardData,
    selectWizardValue,
    backWizard,
    wizardOpen,
    openWizard,
    closeWizard,
    phase,
    activeItem,
    activeQty,
    lines,
    selectItem,
    confirmQuantity,
    cancelQuantity,
    increaseQty,
    decreaseQty,

    removeLine,
    resetSession,
    isReady: step === "done",
  };
};

import { useEffect } from "react";

interface UseKeyboardShortcutOptions {
  key: string;
  ctrlKey?: boolean;
  shiftKey?: boolean;
  altKey?: boolean;
  metaKey?: boolean;
  enabled?: boolean;
  preventDefault?: boolean;
  ignoreInputs?: boolean;
  handler: (event: KeyboardEvent) => void;
}

const isEditableElement = (el: EventTarget | null): boolean => {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName.toLowerCase();
  return (
    tag === "input" ||
    tag === "textarea" ||
    tag === "select" ||
    el.isContentEditable
  );
};

export const useKeyboardShortcut = ({
  key,
  ctrlKey = false,
  shiftKey = false,
  altKey = false,
  metaKey = false,
  enabled = true,
  preventDefault = true,
  ignoreInputs = false,
  handler,
}: UseKeyboardShortcutOptions) => {
  useEffect(() => {
    if (!enabled) return;

    const onKeyDown = (event: KeyboardEvent) => {
      // Normalisasi nama key (mengabaikan case untuk huruf 'm' dan 'M')
      const isKeyMatch = event.key.toLowerCase() === key.toLowerCase() || event.key === key;
      
      if (!isKeyMatch) return;
      if (event.ctrlKey !== ctrlKey) return;
      if (event.shiftKey !== shiftKey) return;
      if (event.altKey !== altKey) return;
      if (event.metaKey !== metaKey) return;
      if (ignoreInputs && isEditableElement(event.target)) return;

      if (preventDefault) event.preventDefault();
      handler(event);
    };

    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  }, [
    key,
    ctrlKey,
    shiftKey,
    altKey,
    metaKey,
    enabled,
    preventDefault,
    ignoreInputs,
    handler,
  ]);
};

export const useProductSearch = (keyword: string, limit = 8) => {
  const trimmed = keyword.trim();

  const { data, isLoading, refetch } = useQuery<Product[]>({
    queryKey: ["product-search", trimmed, limit],
    queryFn: async () => {
      // === MOCK ===
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 150));
        if (!trimmed) return mockProducts.slice(0, limit);
        const lower = trimmed.toLowerCase();
        return mockProducts
          .filter(
            (p) =>
              p.name.toLowerCase().includes(lower) ||
              p.code.toLowerCase().includes(lower),
          )
          .slice(0, limit);
      }

      // === SUPABASE ===
      let query = supabaseClient
        .from(TABLES.products) // "barang"
        .select("*")
        .limit(limit);

      if (trimmed) {
        query = query.or(`nama.ilike.%${trimmed}%,kode.ilike.%${trimmed}%`);
      }

      const { data, error } = await query;
      if (error) throw error;
      return (data ?? []).map(mapProduct);
    },
    staleTime: 10_000,
  });

  return {
    products: data ?? [],
    isLoading,
    refetch,
  };
};

import { useCallback, useEffect, useRef, useState } from "react";

// ============================================

export interface PendingLine {
  itemId: number;
  itemName: string;
  size: string | null; // ← TAMBAH
  unit: string;
  quantity: number;
}

interface UseQuickEntryParams {
  onFinish?: (exit: StockExit) => void;
}

export const useQuickEntry = ({ onFinish }: UseQuickEntryParams = {}) => {
  const { message } = App.useApp();
  const flow = useEntryFlow();

  const [keyword, setKeyword] = useState("");
  const [highlightedIdx, setHighlightedIdx] = useState(0);
  const [activeItem, setActiveItem] = useState<Product | null>(null);
  const [activeQty, setActiveQty] = useState(1);
  const [pendingLines, setPendingLines] = useState<PendingLine[]>([]);

  const searchRef = useRef<HTMLInputElement>(null);
  const qtyRef = useRef<HTMLInputElement>(null);

  const { products: searchResults } = useProductSearch(keyword, 8);
  const phase = activeItem ? "quantity" : "search";

  // ==========================================
  // Auto-focus
  // ==========================================
  useEffect(() => {
    if (!flow.isReady) return;
    if (phase === "search") {
      setTimeout(() => searchRef.current?.focus(), 100);
    } else {
      setTimeout(() => qtyRef.current?.focus(), 100);
    }
  }, [phase, flow.isReady]);

  useEffect(() => setHighlightedIdx(0), [keyword]);

  // ==========================================
  // Ctrl+Enter → selesai
  // ==========================================
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (e.ctrlKey && e.key === "Enter" && pendingLines.length > 0) {
        e.preventDefault();
        handleFinish();
      }
    };
    window.addEventListener("keydown", handler);
    return () => window.removeEventListener("keydown", handler);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [pendingLines.length]);

  // ==========================================
  // Search handlers
  // ==========================================
  const commitItem = useCallback(() => {
    if (searchResults.length === 0) return;
    const chosen = searchResults[highlightedIdx] ?? searchResults[0];
    setActiveItem(chosen);
    setActiveQty(1);
    setKeyword("");
  }, [searchResults, highlightedIdx]);

  const handleSearchKeyDown = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === "ArrowDown") {
        e.preventDefault();
        setHighlightedIdx((i) => Math.min(i + 1, searchResults.length - 1));
      } else if (e.key === "ArrowUp") {
        e.preventDefault();
        setHighlightedIdx((i) => Math.max(i - 1, 0));
      } else if (e.key === "Enter") {
        e.preventDefault();
        commitItem();
      }
    },
    [searchResults.length, commitItem],
  );

  // ==========================================
  // Quantity handlers
  // ==========================================
  const confirmQuantity = useCallback(() => {
    if (!activeItem) return;
    setPendingLines((prev) => {
      const idx = prev.findIndex((l) => l.itemId === activeItem.id);
      if (idx >= 0) {
        const copy = [...prev];
        copy[idx] = {
          ...copy[idx],
          quantity: copy[idx].quantity + activeQty,
        };
        return copy;
      }
      return [
        ...prev,
        {
          itemId: activeItem.id,
          itemName: activeItem.name,
          size: activeItem.size, // ← TAMBAH
          unit: activeItem.unit,
          quantity: activeQty,
        },
      ];
    });
    setActiveItem(null);
    setActiveQty(1);
  }, [activeItem, activeQty]);

  const handleQtyKeyDown = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === "Enter") {
        e.preventDefault();
        confirmQuantity();
      } else if (e.key === "Escape") {
        e.preventDefault();
        setActiveItem(null);
        setActiveQty(1);
      } else if (e.key === "ArrowUp") {
        e.preventDefault();
        setActiveQty((q) => q + 1);
      } else if (e.key === "ArrowDown") {
        e.preventDefault();
        setActiveQty((q) => Math.max(1, q - 1));
      }
    },
    [confirmQuantity],
  );

  const removeLine = useCallback((itemId: number) => {
    setPendingLines((prev) => prev.filter((l) => l.itemId !== itemId));
  }, []);

  const resetEntry = useCallback(() => {
    setPendingLines([]);
    setActiveItem(null);
    setActiveQty(1);
    setKeyword("");
  }, []);

  // ============================================
  // Handle Finish — build StockExit sesuai tipe baru
  // ============================================
  const handleFinish = useCallback(() => {
    if (pendingLines.length === 0) return;
    if (!flow.person || !flow.workplace) return;

    const now = dayjs();
    const exit: StockExit = {
      id: Date.now(),
      nomorTransaksi: "", // akan di-generate DB
      timestamp: now.toISOString(),
      date: now.format("YYYY-MM-DD"),
      time: now.format("HH:mm"),
      person: flow.person.name,
      destination: flow.workplace.name, // ← workplace jadi destination
      isBreakdown: true, // default breakdown
      items: pendingLines.map<StockExitItem>((l) => ({
        itemId: l.itemId,
        itemName: l.itemName,
        size: l.size, // ← TAMBAH
        unit: l.unit,
        quantity: l.quantity,
      })),
    };

    message.success(
      `Tersimpan: ${pendingLines.length} jenis barang di ${flow.workplace.name}`,
    );

    onFinish?.(exit);
    resetEntry();
    flow.resetSession();
  }, [pendingLines, flow, message, onFinish, resetEntry]);

  const handleCancel = useCallback(() => {
    resetEntry();
    flow.resetSession();
  }, [resetEntry, flow]);

  const totalPendingItems = pendingLines.reduce((s, l) => s + l.quantity, 0);

  return {
    flow,
    keyword,
    setKeyword,
    highlightedIdx,
    searchResults,
    searchRef,
    handleSearchKeyDown,
    commitItem,
    activeItem,
    activeQty,

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../utility/supabaseClient";
import { TABLES } from "../utility/constants";
import { USE_MOCK } from "../utility/config";
import { mockStockExits } from "../mocks/stockExitList";
import { mapStockExits } from "../utility/mappers";
import type { StockExit } from "../types";

interface Options {
  onlyToday?: boolean;
  breakdown?: boolean; // true = breakdown, false = non-breakdown, undefined = semua
}

export const useStockExits = ({
  onlyToday = false,
  breakdown,
}: Options = {}) => {
  const { data, isLoading, refetch } = useQuery<StockExit[]>({
    queryKey: ["stock-exits", onlyToday, breakdown],
    queryFn: async () => {
      // === MOCK ===
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return mockStockExits;
      }

      // === SUPABASE ===
      let query = supabaseClient
        .from(TABLES.stockExits)
        .select("*")
        .order("created_at", { ascending: false });

      if (onlyToday) {
        const startOfDay = dayjs().startOf("day").toISOString();
        const endOfDay = dayjs().endOf("day").toISOString();
        query = query.gte("created_at", startOfDay).lte("created_at", endOfDay);
      }

      if (breakdown !== undefined) {
        query = query.eq("is_breakdown", breakdown);
      }

      const { data, error } = await query;
      if (error) throw error;

      return mapStockExits(data ?? []);
    },
    refetchInterval: USE_MOCK ? false : 30_000,
  });

  return {
    exits: data ?? [],
    isLoading,
    refetch,
  };
};

import { useQuery } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../utility/supabaseClient";
import { TABLES } from "../utility/constants";
import { USE_MOCK } from "../utility/config";
import { mockStockInChart } from "../mocks/stockEntries";
import type { StockInChartItem } from "../types";

export const useStockInChart = (days = 30) => {
  const { data, isLoading, refetch } = useQuery<StockInChartItem[]>({
    queryKey: ["stock-in-chart", days],
    queryFn: async () => {
      // === MODE MOCK ===
      if (USE_MOCK) {
        await new Promise((r) => setTimeout(r, 300));
        return mockStockInChart.slice(-days);
      }

      // === MODE SUPABASE ===
      const fromDate = dayjs().subtract(days, "day").format("YYYY-MM-DD");

      const { data, error } = await supabaseClient
        .from(TABLES.stockEntries)
        .select("tanggal, jumlah")
        .gte("tanggal", fromDate)
        .order("tanggal", { ascending: true });

      if (error) throw error;

      // Group by date
      const grouped = (data ?? []).reduce<Record<string, number>>(
        (acc, row: { tanggal: string; jumlah: number }) => {
          acc[row.tanggal] = (acc[row.tanggal] ?? 0) + row.jumlah;
          return acc;
        },
        {},
      );

      return Object.entries(grouped).map(([date, total]) => ({
        date,
        total,
      }));
    },
  });

  return {
    items: data ?? [],
    isLoading,
    refetch,
  };
};

      setSearchParams({ person: personName, workplace: val.name });
    }
  };

  const commitItem = (item?: Product) => {
    const chosen = item ?? products[highlightedIdx] ?? products[0];
    if (!chosen) return;
    setActiveItem(chosen);
    setActiveQty(1);
    setKeyword("");
  };

  const confirmQuantity = () => {
    if (!activeItem) return;
    if (activeQty <= 0) {
      message.warning("Jumlah minimal 1");
      return;
    }
    setLines((prev) => {
      const idx = prev.findIndex((l) => l.itemId === activeItem.id);
      if (idx >= 0) {
        const copy = [...prev];
        copy[idx] = {
          ...copy[idx],
          quantity: copy[idx].quantity + activeQty,
        };
        return copy;
      }
      return [
        ...prev,
        {
          itemId: activeItem.id,
          itemName: activeItem.name,
          size: activeItem.size,
          unit: activeItem.unit,
          maxStock: activeItem.stock,
          quantity: activeQty,
        },
      ];
    });
    setActiveItem(null);
    setActiveQty(1);
  };

  const removeLine = (itemId: number) => {
    setLines((prev) => prev.filter((l) => l.itemId !== itemId));
  };

  const handleFinish = async () => {
    if (lines.length === 0) {
      message.warning("Tambahkan minimal 1 barang terlebih dahulu");
      return;
    }
    if (!personName || !workplaceName) {
      setWizardStep("person");
      setWizardOpen(true);
      return;
    }

    setIsSubmitting(true);
    try {
      const trxNumber = `OUT-${Date.now()}`;

      if (!USE_MOCK) {
        const payload = lines.map((l) => ({
          nomor_transaksi: trxNumber,
          barang_id: l.itemId,
          nama_pekerja: personName,
          nama_barang: l.itemName,
          ukuran_barang: l.size,
          satuan_barang: l.unit,
          jumlah_barang_keluar: l.quantity,
          tujuan: workplaceName,
          is_breakdown: isBreakdown,
        }));

        const { error: insertErr } = await supabaseClient
          .from(TABLES.stockExits)
          .insert(payload);

        if (insertErr) throw insertErr;

        // Update stok berkurang di tabel barang
        for (const line of lines) {
          const newStock = Math.max(0, line.maxStock - line.quantity);
          await supabaseClient
            .from(TABLES.products)
            .update({ stok: newStock })
            .eq("id", line.itemId);
        }
      }

      message.success(
        `Pengeluaran ${lines.length} jenis barang (${
          isBreakdown ? "Breakdown" : "Diluar Breakdown"
        }) berhasil disimpan!`,
      );

      await queryClient.invalidateQueries({ queryKey: ["products"] });
      await queryClient.invalidateQueries({ queryKey: ["stock-exits"] });
      await queryClient.invalidateQueries({ queryKey: ["dashboard-stats"] });

      navigate("/dashboard");
    } catch (err: any) {
      message.error(err?.message || "Gagal menyimpan pengeluaran barang");
    } finally {
      setIsSubmitting(false);
    }
  };

  useEffect(() => {
    const onKeyDown = (e: KeyboardEvent) => {
      if (e.ctrlKey && e.key === "Enter" && lines.length > 0 && !isSubmitting) {
        e.preventDefault();
        handleFinish();
      }
    };
    window.addEventListener("keydown", onKeyDown);
    return () => window.removeEventListener("keydown", onKeyDown);
  });

  const totalItems = lines.reduce((sum, l) => sum + l.quantity, 0);

  return (
    <div style={{ maxWidth: 920, margin: "0 auto", paddingBottom: 32 }}>
      <SetupWizard
        open={wizardOpen}
        step={wizardStep}
        title={
          wizardStep === "person"
            ? "Pilih Nama Pekerja"
            : "Pilih Lokasi Pekerjaan"
        }
        placeholder={
          wizardStep === "person"
            ? "Ketik nama pekerja..."
            : "Ketik lokasi pekerjaan..."
        }
        options={wizardStep === "person" ? persons : mockWorkplaces}
        onSelect={handleWizardSelect}
        onBack={
          wizardStep === "workplace"
            ? () => setWizardStep("person")
            : undefined
        }
        onClose={() => {
          if (!personName || !workplaceName) {
             navigate("/dashboard");
          } else {
             setWizardOpen(false);
          }
        }}
      />

      <div
        style={{
          marginBottom: 16,
          display: "flex",
          alignItems: "center",
          justifyContent: "space-between",
          flexWrap: "wrap",
          gap: 12,
        }}
      >
        <Space align="center">
          <Button
            icon={<ArrowLeftOutlined />}
            onClick={() => navigate("/dashboard")}
            type="text"
            size="large"
          />
          <div>
            <Title level={3} style={{ margin: 0 }}>
              Pengeluaran Barang
            </Title>
            <Text type="secondary">
              Pilih barang yang diambil dan tentukan jumlahnya
            </Text>
          </div>
        </Space>

        <Card
          size="small"
          style={{
            borderRadius: 10,
            background: isBreakdown ? "#e6f4ff" : "#fffbe6",
            borderColor: isBreakdown ? "#91caff" : "#ffe58f",
          }}
          styles={{ body: { padding: "8px 14px" } }}
        >
          <Space align="center" size={10}>
            <div>
              <Text strong style={{ fontSize: 13, display: "block" }}>
                Status: {isBreakdown ? "Masuk Breakdown" : "Diluar Breakdown"}
              </Text>
              <Text type="secondary" style={{ fontSize: 11 }}>
                {isBreakdown
                  ? "Otomatis tercatat di daftar Breakdown"
                  : "Tercatat sebagai Non-Breakdown"}
              </Text>
            </div>
            <Switch
              checked={isBreakdown}
              onChange={setIsBreakdown}
              checkedChildren="Breakdown"
              unCheckedChildren="Non"
            />
          </Space>
        </Card>
      </div>

      <Card
        variant="borderless"
        style={{ borderRadius: 12, marginBottom: 16 }}
        styles={{ body: { padding: 16 } }}
      >
        <Row justify="space-between" align="middle" gutter={[12, 12]}>
          <Col>
            <Space size={12} wrap>
              <Tag
                icon={<UserOutlined />}
                color="blue"
                style={{ fontSize: 14, padding: "6px 12px", borderRadius: 8 }}
              >
                Pekerja: <strong>{personName || "Belum dipilih"}</strong>
              </Tag>
              <Tag
                icon={<EnvironmentOutlined />}
                color="cyan"
                style={{ fontSize: 14, padding: "6px 12px", borderRadius: 8 }}
              >
                Lokasi: <strong>{workplaceName || "Belum dipilih"}</strong>
              </Tag>
            </Space>

          <Col>

              size="small"

                setWizardStep("person");
                setWizardOpen(true);

              Ubah Nama / Lokasi
            </Button>
          </Col>
        </Row>
      </Card>

        variant="borderless"
        style={{ borderRadius: 12, marginBottom: 16 }}
        styles={{ body: { padding: 20 } }}

              Cari Material / Alat
            </Text>
            <AutoComplete
              value={keyword}
              onChange={setKeyword}
              options={products.map((p, idx) => ({
                value: p.name,
                label: (
                  <div
                    style={{
                      display: "flex",
                      justifyContent: "space-between",
                      padding: "4px 0",

                        idx === highlightedIdx ? "#e6f4ff" : "transparent",

                    e.preventDefault();

        )}

        {phase === "quantity" && activeItem && (
          <div>
            <div
              style={{
                display: "flex",
                justifyContent: "space-between",
                alignItems: "center",
                marginBottom: 12,
                flexWrap: "wrap",

                    e.preventDefault();
                    setActiveItem(null);

              </Button>

                <Tag>Enter</Tag>
                <Text type="secondary" style={{ fontSize: 12 }}>
                  tambahkan ke daftar
                </Text>
                <Tag>Esc</Tag>
                <Text type="secondary" style={{ fontSize: 12 }}>
                  batal
                </Text>
              </Space>
              <Space>
                <Button onClick={() => setActiveItem(null)}>Batal</Button>
                <Button type="primary" onClick={confirmQuantity}>
                  Tambah Barang
                </Button>
              </Space>
            </Space>
          </div>
        )}
      </Card>

      <Card
        title={`📦 Daftar Pengeluaran (${lines.length} jenis · ${formatNumber(
          totalItems,
        )} item)`}
        variant="borderless"
        style={{ borderRadius: 12 }}
        extra={
          <Button
            type="primary"
            size="large"
            icon={<CheckCircleOutlined />}
            disabled={lines.length === 0}
            loading={isSubmitting}
            onClick={handleFinish}
          >
            Simpan (Ctrl+Enter)
          </Button>
        }
      >
        {lines.length === 0 ? (
          <Empty
            description="Belum ada barang yang dipilih"
            image={Empty.PRESENTED_IMAGE_SIMPLE}
          />
        ) : (
          <List
            dataSource={lines}
            renderItem={(line) => (
              <List.Item
                actions={[
                  <Button
                    key="del"

                    danger
                    icon={<CloseCircleOutlined />}

                  color="red"
                  style={{ fontSize: 15, padding: "4px 12px", fontWeight: 600 }}
                >
                  {formatNumber(line.quantity)} {line.unit}
                </Tag>
              </List.Item>
            )}
          />
        )}
      </Card>
    </div>
  );
};

export default EntryPage;

import { useState } from "react";
import {
  Button,

  Col,

  Space,

  AppstoreAddOutlined,
} from "@ant-design/icons";
import { useQueryClient } from "@tanstack/react-query";
import dayjs from "dayjs";
import { supabaseClient } from "../../utility/supabaseClient";
import { TABLES } from "../../utility/constants";
import { USE_MOCK } from "../../utility/config";
import { useProducts } from "../../hooks/useProducts";
import { formatNumber, formatDate } from "../../utility/format";
import type { Product } from "../../types";

const { Title, Text } = Typography;

interface StockEntryFormValues {
  productId: number;
  jumlah: number;
  tanggal_pengiriman: dayjs.Dayjs;
  spesifikasi?: string;
  merk?: string;
}

export const ProductCreatePage = () => {
  const [form] = Form.useForm<StockEntryFormValues>();
  const { message } = App.useApp();
  const queryClient = useQueryClient();

  const { products, isLoading } = useProducts();
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null);

  // Menyimpan riwayat lokal sementara setelah diinput
  const [recentEntries, setRecentEntries] = useState<any[]>([]);

  const handleProductSelect = (val: number) => {
    const prod = products.find((p) => p.id === val);
    setSelectedProduct(prod || null);
  };

  const handleSubmit = async (values: StockEntryFormValues) => {
    if (!selectedProduct) return;
    setIsSubmitting(true);
    
    try {
      if (!USE_MOCK) {
        // 1. Catat di tabel barang_masuk
        const payload = {
          kode_barang: selectedProduct.code,
          nama_barang: selectedProduct.name,
          ukuran_barang: selectedProduct.size,
          spesifikasi_barang: values.spesifikasi || null,
          satuan_barang: selectedProduct.unit,
          kategori: selectedProduct.category,
          jumlah_barang_masuk: values.jumlah,
          tanggal_pengiriman: values.tanggal_pengiriman.format("YYYY-MM-DD"),
          keterangan: values.merk ? `Merk: ${values.merk}` : null,
        };

        const { error: insertErr } = await supabaseClient
          .from("barang_masuk")
          .insert(payload);

        if (insertErr) throw insertErr;

        // 2. Tambah stok di tabel barang
        const newStock = selectedProduct.stock + values.jumlah;
        const { error: updateErr } = await supabaseClient
          .from(TABLES.products)
          .update({ stok: newStock })
          .eq("id", selectedProduct.id);

        if (updateErr) throw updateErr;
      }

      message.success(`Berhasil menambahkan ${values.jumlah} ${selectedProduct.unit} ke stok ${selectedProduct.name}!`);
      
      setRecentEntries((prev) => [
        {
          key: Date.now(),
          nama: selectedProduct.name,
          ukuran: selectedProduct.size,
          jumlah: values.jumlah,
          satuan: selectedProduct.unit,
          tanggal: values.tanggal_pengiriman.format("YYYY-MM-DD"),
        },
        ...prev,
      ]);

      form.resetFields(["jumlah", "spesifikasi", "merk"]);
      await queryClient.invalidateQueries({ queryKey: ["products"] });
      await queryClient.invalidateQueries({ queryKey: ["dashboard-stats"] });

    } catch (err: any) {
      message.error(err?.message || "Gagal menyimpan data barang masuk");
    } finally {
      setIsSubmitting(false);
    }

  const { products, isLoading: isSearching } = useProductSearch(searchKeyword);

  // Auto-focus ke product search saat halaman dibuka
  useEffect(() => {
    setTimeout(() => productSearchRef.current?.focus(), 100);
  }, []);

  // Jika ada productId di URL, langsung set

    if (preselectedId) {
      const found = products.find((p) => p.id === Number(preselectedId));
      if (found) {
        setSelectedProduct(found);
        form.setFieldsValue({
          productId: found.id,
          unitPrice: found.costPrice,
        });
      }
    }
  }, [preselectedId, products, form]);

  useKeyboardShortcut({
    key: "Enter",
    ctrlKey: true,

    key: "Escape",

      productId: product.id,
      unitPrice: product.costPrice,
    });

      form.getFieldInstance("quantity")?.focus?.();

      unitPrice: values.unitPrice,
      date: values.date.format("YYYY-MM-DD"),
      note: values.note,
    });

    form.resetFields();
    setSelectedProduct(null);
    // Balik ke search produk untuk entri berikutnya
    setTimeout(() => productSearchRef.current?.focus(), 100);

import type { Product, StockExit, StockExitItem } from "../types";

// ============================================
// Product — DB row → Product

export const mapProduct = (row: Record<string, unknown>): Product => ({
  id: row.id as number,
  code: row.kode as string,
  name: row.nama as string,
  size: (row.ukuran as string) ?? null,
  specification: (row.spesifikasi as string) ?? null,
  category: (row.kategori as string) ?? "Material",
  categoryId: null,
  categoryName: null,
  unit: (row.satuan as string) ?? "pcs",
  stock: (row.stok as number) ?? 0,
  minStock: 0,
  costPrice: 0,
  status: undefined,
  createdAt: (row.created_at as string) ?? new Date().toISOString(),
});

// ============================================
// Stock Exit — group by nomor_transaksi
// ============================================
interface StockExitRow {
  id: number;
  nomor_transaksi: string | null;
  barang_id: number;
  nama_pekerja: string | null;
  nama_barang: string;
  ukuran_barang: string | null;
  satuan_barang: string | null;
  jumlah_barang_keluar: number;
  tujuan: string | null;
  is_breakdown: boolean | null;
  created_at: string;
}

export const mapStockExits = (rows: StockExitRow[]): StockExit[] => {
  const grouped = new Map<string, StockExit>();

  for (const row of rows) {
    const key = row.nomor_transaksi ?? `temp-${row.id}`;

    if (!grouped.has(key)) {
      const created = new Date(row.created_at);
      grouped.set(key, {
        id: row.id,
        nomorTransaksi: key,
        timestamp: row.created_at,
        date: created.toISOString().split("T")[0],
        time: created.toISOString().slice(11, 16),
        person: row.nama_pekerja ?? "-",
        destination: row.tujuan,
        isBreakdown: row.is_breakdown ?? true,
        items: [],
      });
    }

    const item: StockExitItem = {
      itemId: row.barang_id,
      itemName: row.nama_barang,
      size: row.ukuran_barang,
      unit: row.satuan_barang ?? "pcs",
      quantity: row.jumlah_barang_keluar,
    };

    grouped.get(key)!.items.push(item);
  }

  return Array.from(grouped.values());
};
